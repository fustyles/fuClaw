============================================================
執行模型
============================================================
本系統為提示詞驅動的工具路由架構。

Gemini 不使用原生的函式呼叫 API。

取而代之的是：
  - Gemini 輸出結構化 JSON tool_call 回應
  - 本地韌體驗證所有工具呼叫
  - 無效 JSON 會被拒絕執行
  - 執行順序嚴格為循序執行
  - 硬體動作絕不模擬（必須實際執行）

原子執行規則：
  每次回應只能執行一個硬體動作：
    - 一個接腳（pin）
    - 一種操作模式
    - 一個數值

多步驟工作流程將逐步依序執行。

// 將 URL 百分比編碼解碼回原始字串（用於解析 HTTP GET 參數）
// 例如：%20→空格，%2B→+，+→空格
String urldecode(const String& input) {
    String result = "";
    result.reserve(input.length()); // 預先分配，避免多次 heap 重配置
    for (int i = 0; i < (int)input.length(); i++) {
        if (input[i] == '%' && i + 2 < (int)input.length()) {
            // 讀取後兩個十六進位字元並轉為 ASCII 字元
            char hex[3] = { input[i+1], input[i+2], '\0' };
            uint8_t val = (uint8_t)strtol(hex, nullptr, 16);
            result.concat((char)val);
            i += 2; // 跳過已處理的兩個字元
        } else if (input[i] == '+') {
            result += ' '; // HTML form 中的 + 代表空格
        } else {
            result += input[i]; // 一般字元直接附加
        }
    }
    return result;
}

// 將字串進行 URL 百分比編碼（用於組裝 HTTP GET 查詢字串）
// RFC 3986 非保留字元直接輸出，其餘字元轉為 %XX
String urlencode(String str) {
  const char *msg = str.c_str();
  const char *hex = "0123456789ABCDEF"; // 十六進位對照表（大寫）
  String encodedMsg = "";
  while (*msg != '\0') {
    // 字母 / 數字 / - _ . ~ 保持原樣（RFC 3986 非保留字元）
    if (('a' <= *msg && *msg <= 'z') || ('A' <= *msg && *msg <= 'Z') ||
        ('0' <= *msg && *msg <= '9') ||
        *msg == '-' || *msg == '_' || *msg == '.' || *msg == '~') {
      encodedMsg += *msg;
    } else {
      // 其他字元：輸出 %高4位低4位
      encodedMsg += '%';
      encodedMsg += hex[(unsigned char)*msg >> 4]; // 高 4 位元
      encodedMsg += hex[*msg & 0xf];               // 低 4 位元
    }
    msg++;
  }
  return encodedMsg;
}

// 向 Gemini 發送輕量請求，從 HTTP 回應標頭的 Date 欄位擷取 GMT 日期時間
// 此為 RTC 初始化的時間來源（Ameba 無 NTP 支援，改以此方式取得當前時間）
// 回傳值：HTTP 標頭中的 Date 字串（例如：Sat, 28 Jun 2026 10:00:00 GMT）
String getGeminiDatetime() {

  String contents = systemContent + buildGeminiMessage("user", "I am fuClaw!", true);

      client.print(request.substring(i, i + 1024));

    int waitTime = 5000;              // 最多等待 5 秒
    unsigned long startTime = millis();
    bool getStatus = false;            // true=已找到 Date 標頭，開始記錄

      }

// 取得本地時間的格式化字串（從 Ameba 硬體 RTC 讀取 epoch 秒數再轉換）
// filename=true  → 輸出適合檔名的格式：YYYYMMDDHHmmss（無分隔符）
// filename=false → 輸出標準顯示格式：YYYY/M/D HH:mm:ss
String getRtcTimeString(bool filename = false) {
  long long epoch = rtc.Read();        // 從 Ameba 硬體 RTC 讀取 epoch 秒數（64-bit）
  time_t rawtime = (time_t)epoch;      // 截斷為 32-bit（2038 年前安全）
  struct tm *timeinfo = localtime(&rawtime); // 轉換為本地時間結構
  char buffer[32];

  if (filename == true) {
	  sprintf(
		buffer,
		"%04d%d%d%02d%02d%02d",
		timeinfo->tm_year + 1900,
		timeinfo->tm_mon + 1,
		timeinfo->tm_mday,
		timeinfo->tm_hour,
		timeinfo->tm_min,
		timeinfo->tm_sec
	  );
  }
  else {
	  sprintf(
		buffer,
		"%04d/%d/%d %02d:%02d:%02d",
		timeinfo->tm_year + 1900,
		timeinfo->tm_mon + 1,
		timeinfo->tm_mday,
		timeinfo->tm_hour,
		timeinfo->tm_min,
		timeinfo->tm_sec
	  );
  }

  return String(buffer);
}

// 透過 Gemini API 取得當前 GMT 時間，轉換為指定時區後寫入 Ameba 硬體 RTC
// 流程：getGeminiDatetime() → Gemini 時區轉換 → JSON 解析 → rtc.Init()/Write()
// 注意：此函式與 /syncrtc 工具共用，啟動時及使用者主動同步時均會呼叫
void rtcInitialTime(String workName) {
  rtcUpdateStatus = true; // 標記 RTC 同步中（防止排程任務在同步期間誤判時間）
  
  // 組裝時間轉換提示詞：請 Gemini 將 GMT 時間轉換為指定時區的本地時間
  // 並額外加 4 秒補償 API 來回的時間延遲
  String prompt =
    "Convert this GMT datetime to " + timeZone + ".\n"
    "GMT datetime: " + getGeminiDatetime() + "\n\n"
	
    "Before generating the JSON output, add exactly 4 seconds to the converted local datetime.\n"
    "Handle minute, hour, day, month, and year rollovers correctly if needed.\n\n" 

    "Output requirements:\n"
    "- Return ONLY a raw JSON object.\n"
    "- Do NOT use markdown.\n"
    "- Do NOT use code fences.\n"
    "- Do NOT explain anything.\n"
    "- Do NOT add extra text.\n"
    "- First character must be {.\n"
    "- Last character must be }.\n\n"

    "Required JSON format:\n"
    "{\n"
    "\"rtcYear\":2026,\n"
    "\"rtcMonth\":5,\n"
    "\"rtcDay\":28,\n"
    "\"rtcHour\":11,\n"
    "\"rtcMinute\":35,\n"
    "\"rtcSecond\":0\n"
    "}";

  // 以 tools=-1（純系統脈絡，無歷史、無工具定義）呼叫 Gemini 取得 JSON 格式時間
  String message = geminiChatRequest(String(taskTags[1]), prompt, -1);

  if (message.startsWith("{") && message.endsWith("}")) {
    // 解析 Gemini 回傳的 JSON 時間物件
    DynamicJsonDocument doc(1024);
    DeserializationError error = deserializeJson(doc, message);
    if (error) {
      Serial.println("[DEBUG] JSON parse failed\n" + message);
      replyUserMessage(workName, "RTC time update failed.", "");
      rtcUpdateStatus = false;
      return;
    }
    JsonObject obj = doc.as<JsonObject>();
    // 從 JSON 讀取各時間欄位（若欄位缺失則預設為 0）
    rtcYear   = obj["rtcYear"]   | 0;
    rtcMonth  = obj["rtcMonth"]  | 0;
    rtcDay    = obj["rtcDay"]    | 0;
    rtcHour   = obj["rtcHour"]   | 0;
    rtcMinute = obj["rtcMinute"] | 0;
    rtcSecond = obj["rtcSecond"] | 0;
  } else {
    Serial.println("[DEBUG] JSON parse failed : (rtcInitialTime)\n" + message);
    replyUserMessage(workName, "RTC time update failed.", "");
  }
  // 初始化 RTC 硬體並將轉換後的本地時間寫入
  rtc.Init();
  long long initTime = rtc.SetEpoch(rtcYear, rtcMonth, rtcDay, rtcHour, rtcMinute, rtcSecond);
  rtc.Write(initTime); // 寫入後 rtc.Read() 即可取得正確時間
}

// 透過 Telegram Bot API sendMessage 傳送 HTML 格式文字訊息
// keyboard：JSON 格式的 reply_markup（傳空字串則不顯示鍵盤）
String telegramSendMessage(String token, String chatid, String text, String keyboard = "") {

  const char* myDomain = "api.telegram.org";
  String getAll = "", getBody = "";
  // 組裝 POST 表單資料（application/x-www-form-urlencoded）
  String request = "parse_mode=HTML&chat_id=" + chatid + "&text=" + text;
  if (keyboard != "")
    request += "&reply_markup=" + keyboard; // 附加自訂鍵盤（可選）

    client.println("POST /bot" + String(token) + "/sendMessage HTTP/1.1");

    client.println("Content-Type: application/x-www-form-urlencoded");
    client.println("Connection: close");
    client.println();
    client.print(request);
    int waitTime = 5000;               // 最多等待 5 秒
    unsigned long startTime = millis();
    bool state = false;                // true=已通過空白行，開始接收 body
    while ((startTime + waitTime) > millis()) {

      if (getBody.length() > 0) break; // 收到 body 即結束
    }
    client.stop();
  } else {
    getBody = "Connected to api.telegram.org failed.";
  }
  return getBody; // 回傳 Telegram API 的 JSON 回應
}

// 從相機拍攝靜態影像並上傳至 Telegram（JPEG 格式）
//
// 注意：整個「拍攝 + 讀取 + 傳送」流程期間持有 imageMutex，
// 防止其他任務在上傳途中呼叫 captureImage() 導致緩衝區被釋放或覆寫
String telegramSendCapturedImage(String token, String chat_id, bool frames) {
  const char* myDomain = "api.telegram.org";
  String getAll="", getBody = "";
  WiFiSSLClient client;

  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    return "Image buffer busy, please try again.";
  }

  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);
    else if (!frames && imageLength == 0) {
      client.stop();
      xSemaphoreGive(imageMutex);
      return "Previous image does not exist";
    }

    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    String head =
      "--Taiwan\r\nContent-Disposition: form-data; name=\"chat_id\"; \r\n\r\n"
      + chat_id +
      "\r\n--Taiwan\r\nContent-Disposition: form-data; name=\"photo\"; filename=\"capture.jpg\"\r\nContent-Type: image/jpeg\r\n\r\n";

    String tail = "\r\n--Taiwan--\r\n";

    size_t imageLen = imageLength;
    size_t extraLen = head.length() + tail.length();
    size_t totalLen = imageLen + extraLen;

    client.println("POST /bot"+token+"/sendPhoto HTTP/1.1");
    client.println("Host: " + String(myDomain));
    client.println("Content-Length: " + String(totalLen));
    client.println("Content-Type: multipart/form-data; boundary=Taiwan");

    client.print(head);

    // 分段傳送 JPEG 影像資料
    for (size_t n=0;n<fbLen;n=n+1024) {
      if (n+1024<fbLen) {
        client.write(fbBuf, 1024);
        fbBuf += 1024;
      }
      else {
        size_t remainder = fbLen - n;
        if (remainder > 0)
          client.write(fbBuf, remainder);
      }
    }

    client.print(tail);

    int waitTime = 10000;
    unsigned long startTime = millis();
    bool state = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (state)
          getBody += String(c);

        if (c == '\n') {
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);

        startTime = millis();
      }

      if (getBody.length()>0)
        break;
    }

    client.stop();
    Serial.println();

  } else {
    getBody="Connected to api.telegram.org failed.";
    Serial.println("Connected to api.telegram.org failed.");
  }

  xSemaphoreGive(imageMutex);

  return getBody;
}

// 清除文字字串中的時間戳記、工作 ID，並在任務標籤處截斷
// 若文字為空或明確標示為 none，則回傳 "NONE"
String removeTimestamps(String workId, String timestamps, String text) {

    // 若文字為空或已標示為 NONE，直接跳過處理
    if (text.length() == 0 || text.startsWith("NONE")) return "NONE";
	
    // 快速退出：若字串中完全沒有角括號，則無需進一步處理
    if (text.indexOf("&lt;") == -1 && text.indexOf("<") == -1) return text;	

    // 將 HTML 跳脫的角括號還原為原始字元
    text.replace("&lt;", "<");
    text.replace("&gt;", ">");

    // 從文字中移除時間戳記與工作 ID
    text.replace(timestamps, "");
    text.replace(workId, "");

    // 在第一個任務標籤處截斷文字
    // 標籤之後的所有內容均被丟棄
    for (const auto& tag : taskTags) {
        int pos = text.indexOf(tag);
        if (pos != -1)
            text = text.substring(0, pos); // 只保留標籤之前的部分
    }

    return text;
}

      for (size_t i = 0; i < fbLen; i += 3) {

// 對字串進行 JSON 跳脫處理（適用於 Ameba 版本，分兩步驟執行）
// Pass 1：將所有 " 替換為 \"（防止破壞 JSON 字串邊界）
// Pass 2：將多餘的雙反斜線 \\\\ 簡化為單反斜線 \\（修正 Pass 1 可能引入的過度跳脫）
// 注意：ESP32-S3 版本使用 message.replace() 直接處理，Ameba 版本需此獨立函式
String escapeForJson(const String &src) {
  unsigned int n = src.length();
  // Pass 1：跳脫雙引號 " → \" （預留約 12% 額外空間應對大量引號）
  String s1;
  s1.reserve(n + (n >> 3) + 16);
  for (unsigned int i = 0; i < n; i++) {
    char c = src.charAt(i);
    if (c == '"') s1 += '\\'; // 在 " 前插入反斜線
    s1 += c;
  }
  // Pass 2：簡化雙反斜線 \\\\ → \\（處理原始字串中已有的反斜線）
  String s2;
  s2.reserve(s1.length());
  unsigned int m = s1.length();
  for (unsigned int i = 0; i < m; ) {
    if (s1.charAt(i) == '\\' && i + 1 < m && s1.charAt(i + 1) == '\\') {
      s2 += '\\'; // 只保留一個反斜線
      i += 2;      // 跳過兩個字元
    } else {
      s2 += s1.charAt(i);
      i += 1;
    }
  }
  return s2;
}

// 將 role/content 組裝為 Gemini API contents 陣列中的訊息物件
// 格式：{"role":"user/model","parts":[{"text":"..."}]}
// comma=true（預設）：前置逗號（接在已有元素後）
// comma=false：無逗號（陣列第一個元素）
// Ameba 版本使用 escapeForJson() 處理跳脫（ESP32-S3 版本直接用 replace()）
String buildGeminiMessage(const String &role, const String &message, bool comma = true) {
  String esc = escapeForJson(message); // 跳脫特殊字元，防止破壞 JSON 結構
  String jsonMessage;
  jsonMessage.reserve(esc.length() + role.length() + 48); // 預先分配避免多次 heap 重配置
  jsonMessage  = comma ? ", {\"role\": \"" : "{\"role\": \""; // 非首元素加逗號
  jsonMessage += role;
  jsonMessage += "\", \"parts\":[{ \"text\": \"";
  jsonMessage += esc;
  jsonMessage += "\" }]}";
  return jsonMessage;
}

// 從 SD 卡讀取指定檔案的全部內容並回傳字串
// 使用 sdMutex 確保執行緒安全（AmebaFatFS 不支援可重入）
// 回傳空字串：Mutex 逾時、掛載失敗、檔案不存在或 malloc 失敗
String getStringFromFile(String fileNname) {
  String data = "";
  // 取得 SD 互斥鎖（10 秒逾時）
  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] getStringFromFile: failed to take sdMutex");
    return "";
  }
  // 掛載 SD 卡（AmebaFatFS 每次存取前需 begin/end）
  if (!fs.begin()) {
    Serial.println("Card Mount Failed"); // SD 卡未插入或格式不正確
    if (sdMutex) xSemaphoreGive(sdMutex);
    return "";
  }
  // 組合完整路徑（AmebaFatFS 根路徑 + 檔案名稱）
  String path = String(fs.getRootPath()) + "/" + fileNname;
  file = fs.open(path); // 以預設讀取模式開啟
  if (file) {
    uint32_t len = file.size();
    char *buf = (char*)malloc(len + 1); // +1 補 null terminator
    if (buf) {
      file.read(buf, len);  // 一次讀取全部內容
      buf[len] = '\0';     // 手動補 null（file.read 不補）
      data = String(buf);
      free(buf);            // 立即釋放暫存緩衝區
    }
    file.close();
  }
  fs.end(); // 卸載 SD 卡（釋放 FAT 驅動資源）
  if (sdMutex) xSemaphoreGive(sdMutex);
  return data;
}

// 將資料儲存至 SD 卡（寫入前先備份舊版本）
// timestamp=false：備份為 filename.bak（覆蓋舊備份）
// timestamp=true ：備份為 filename_YYYYMMDDHHmmss.bak（保留歷史，適用於 memory.md 重置）
void storeDataToFile(String filename, String data, bool timestamp = false) {
  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] storeDataToFile: failed to take sdMutex");
    return;
  }
  if (!fs.begin()) {
    Serial.println("Card Mount Failed");
    if (sdMutex) xSemaphoreGive(sdMutex);
    return;
  }
  String file_path = String(fs.getRootPath());
  String currentFile = file_path + "/" + filename;
  // 決定備份檔案名稱
  String backupFile = currentFile + ".bak";
  if (timestamp == true)
    backupFile = currentFile + "_" + getRtcTimeString(true) + ".bak"; // 帶時間戳備份
  // 若目標檔案已存在，先備份再寫入（防止寫入失敗導致資料遺失）
  if (fs.exists(currentFile)) {
    if (fs.exists(backupFile))
      fs.remove(backupFile); // 刪除舊備份（同名）
    delay(100);               // 短暫延遲確保刪除完成
    fs.rename(currentFile, backupFile); // 原檔案改名為備份
  }
  file = fs.open(currentFile); // 開啟（或建立）目標檔案（AmebaFatFS 預設為寫入模式）
  if (file) {
    file.println(data.c_str()); // 寫入資料（println 附加換行）
    file.close();
  } else
    Serial.println("[DEBUG] File open failed: " + currentFile);
  fs.end();
  if (sdMutex) xSemaphoreGive(sdMutex);
}

// 透過 TCP（Port 81）向另一台 fuClaw 裝置或代理傳送訊息
// 使用 HTTP GET /message? 路徑（對方的 task_getRequest 端點）
String tcpSendMessage(String workId, String domain, String request) {
  WiFiClient client; // 使用非加密 TCP（裝置間區域網路通訊）
  if (client.connect(domain.c_str(), 81)) { // 連接目標 Port 81
    client.setTimeout(20000); // 20 秒逾時（目標可能需要 Gemini 回應時間）
    // 傳送 HTTP GET 請求（request 已 URL 編碼）
    client.println("GET /message?" + urlencode(request) + " HTTP/1.1");
    client.println("Host: " + domain);
    client.println("Access-Control-Allow-Origin: *");
    client.println("Content-Length: 0");
    client.println();
    
    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }
    
    client.stop();  
    
    return 
      "{\"status\":\"success\","
      "\"method\":\"/tcpSendMessage\","
      "\"response\":\"" + body + "\","   
      "\"workId\":\"" + workId + "\"}"; 
  }

  return 
    "{\"status\":\"error\","
    "\"method\":\"/tcpSendMessage\","       
    "\"reason\":\"Connected to the device failed.\","
    "\"workId\":\"" + workId + "\"}";
      
}

// 向 Gemini generateContent API 發送請求並回傳文字回應
// tools 模式：
//   1（預設）= 含工具定義（主工作模式）
//   0        = 無工具（工作流程收尾對話）
//   2 / 其他 = 純角色提示 + 單一訊息（內部處理用，不帶歷史）
//  -1        = 純系統脈絡（無歷史；用於 RTC 初始化等輔助呼叫）
String geminiChatRequest(String workId, String message, int tools = 1) {
  // workId 注入訊息末尾作辨識標記（後續由 removeTimestamps 清除）
  String timestamps = "\n" + workId;
  // 附加 RTC 時間讓 Gemini 知道當前時刻（排程判斷必需）
  message = message + "\n\nRTC current time: " + getRtcTimeString();
  // 將使用者訊息寫入歷史（持有 stateMutex 確保執行緒安全）
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }
  // 依 tools 模式選擇系統提示詞並組裝 contents 陣列
  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    if (tools == 0)      contents = systemContentNoTools + historicalMessages; // 無工具
    else if (tools == 1) contents = systemContentTools   + historicalMessages; // 標準工具
    else if (tools == 2) contents = systemContent + buildGeminiMessage("user", message); // 輕量單次
    else                 contents = systemContent + buildGeminiMessage("user", message);
    xSemaphoreGive(stateMutex);
  }

  String request = "{\"contents\": [" + contents +
                   "],\"generationConfig\": {\"maxOutputTokens\": " +
                   geminiMaxOutputTokens +
                   ", \"temperature\": " + geminiTemperature + "}}";

  WiFiSSLClient client;
  String responseText = "";
  client.setTimeout(10000); // TCP 層逾時 10 秒
  if (client.connect("generativelanguage.googleapis.com", 443)) {
    // 傳送 HTTP/1.0 請求（不需處理 chunked 編碼，簡化回應解析）
    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();
    // 分 1024 位元組分段傳送（避免超過 TCP buffer）
    for (int i = 0; i < request.length(); i += 1024)
      client.print(request.substring(i, i + 1024));
    String body = "";
    unsigned long timeout = millis() + 20000; // 動態逾時（有資料時持續延展）
    bool headersEnded = false;
    String line = "";
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();
        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) headersEnded = true; // 空行=header 結束
            line = "";
          } else if (c != '\r') line += c;
        } else {
          body += c;
          timeout = millis() + 20000; // 有資料就延展逾時
        }
      }
      vTaskDelay(1); // 讓出 CPU tick
    }
    client.stop();
    // 找到第一個 { 跳過可能的 HTTP chunked 大小行
    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) body = body.substring(jsonStart);
    // 解析 Gemini JSON 回應
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);
    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (geminiChatRequest)\n" + body);
      responseText = "JSON parse failed (geminiChatRequest). Please try again.";
    } else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>(); // 正常回應
    } else if (doc["error"]) {
      // API 回傳錯誤（金鑰無效、配額耗盡等）
      responseText = "[DEBUG] Gemini API Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Gemini API Error";
    } else {
      responseText = "Unexpected response from Gemini.";
      Serial.println("Unknown response format.");
    }
  } else {
    Serial.println("Failed to connect to Gemini API");
    responseText = "Connection failed";
  }
  if (responseText == "") responseText = "Gemini did not respond. Please try again.";
  // 清除 workId/timestamp 標記後存入歷史（model 角色）
  responseText = removeTimestamps(workId, timestamps, responseText);
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }
  return responseText;
}

// 拍攝相機畫面並傳送至 Gemini Vision 進行多模態分析
//
// 注意：整個「拍攝 + Base64 編碼 + HTTP 傳送」流程期間持有 imageMutex，
// 防止其他任務的 captureImage() 在處理期間釋放或覆寫緩衝區
String geminiVisionRequest(String workId, String message, bool frames = true) {
  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  WiFiSSLClient client;
  String responseText = "";
  const char* myDomain = "generativelanguage.googleapis.com";
	  
  client.setTimeout(10000);

  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    responseText = "Image buffer busy, please try again.";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      historicalMessages += buildGeminiMessage("model", responseText + timestamps);
      xSemaphoreGive(stateMutex);
    }
    return responseText;
  }
	
  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);
    else if (!frames && imageLength == 0) {
      client.stop();
      xSemaphoreGive(imageMutex);

      responseText = "Previous image does not exist";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("model", responseText + timestamps);
        xSemaphoreGive(stateMutex);
      }

      return responseText;
    }
    
    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    char *input = (char *)fbBuf;
    char output[base64_enc_len(3)];
    String imageFile = "";

    // 每次 base64_encode() 呼叫推進 3 個位元組
    // 當 fbLen 不是 3 的倍數時，base64_encode() 自動處理填補
    for (size_t i = 0; i < fbLen; i += 3) {
      base64_encode(output, input, 3);
      input += 3;
      imageFile += String(output);
    }

    String Data = "{\"contents\": [{\"parts\": [{\"text\": \"" + message + 
                  "\"}, {\"inline_data\": {\"mime_type\":\"image/jpeg\",\"data\":\"" + 
                  imageFile + "\"}}]}]}";

    // 影像緩衝區已完整編碼為 Data 字串
    // 因此可在網路往返前安全釋放 imageMutex，讓其他任務拍攝/編碼新畫面
    xSemaphoreGive(imageMutex);

    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.0");
    client.println("Host: " + String(myDomain));
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(Data.length()));
    client.println("Connection: close");
    client.println();
    
    for (size_t i = 0; i < Data.length(); i += 1024) {
      client.print(Data.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }
    
    client.stop();   

    int jsonStart = body.indexOf('{'); 
    if (jsonStart != -1) { 
      body = body.substring(jsonStart);
    }

    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed (geminiSearchRequest):\n" + body);
      responseText = "JSON parse failed (geminiSearchRequest). Please try again.";
    } 
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    } 
    else if (doc["error"]) {
      responseText = "[DEBUG] Gemini API (Vision) Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
	  responseText = "Gemini API (Vision) Error";
    } 
    else {
      responseText = "Unexpected response from Gemini Vision.";
    }

  } else {
    Serial.println("Failed to connect to Gemini API (Vision)");
    responseText = "Connection failed";
    xSemaphoreGive(imageMutex);
  }

  if (responseText == "") {
    responseText = "Gemini Vision did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// GPIO 輸出控制（/digitalwrite 與 /analogwrite 工具的實作）
// mode="digitalwrite"：數位輸出，value 只能是 0（LOW）或 1（HIGH）
// mode="analogwrite" ：PWM 輸出，value 範圍 0~255
String toolPinOutput(int pin, String mode, int value, String workId) {
    pinMode(pin, OUTPUT); // 設定腳位為輸出模式
    mode.toLowerCase();   // 統一轉小寫（容錯處理）

    if (mode == "digitalwrite") {

        if (value != 0 && value != 1) {
            return 
				"{\"status\":\"error\","
				"\"method\":\"/digitalwrite\","				
				"\"reason\":\"invalid_digital_value\","
				"\"workId\":\"" + workId + "\"}";
        }

        digitalWrite(pin, value);

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalwrite\","
			"\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogwrite") {

        if (value < 0 || value > 255) {
            return 
				"{\"status\":\"error\","
				"\"method\":\"/analogwrite\","				
				"\"reason\":\"invalid_analog_value\","
				"\"workId\":\"" + workId + "\"}";
        }

        analogWrite(pin, value);

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogwrite\","
			"\"workId\":\"" + workId + "\"}";

    }

    return
        "{\"status\":\"error\","
        "\"method\":\"/analogwrite\","		
        "\"reason\":\"invalid_output_mode\","
		"\"workId\":\"" + workId + "\"}";
}

// GPIO 輸入讀取（/digitalread 與 /analogread 工具的實作）
// mode="digitalread"：讀取數位值（0=LOW, 1=HIGH）
// mode="analogread" ：讀取 ADC 值（Ameba 為 0~1023，10-bit）
String toolPinInput(int pin, String mode, String workId) {
    pinMode(pin, INPUT); // 設定腳位為輸入模式
    mode.toLowerCase();

    if (mode == "digitalread") {

        int value = digitalRead(pin);

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalread\","
            "\"value\":" + String(value) + ","
			"\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogread") {

        int value = analogRead(pin);

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogread\","
            "\"value\":" + String(value) + ","
			"\"workId\":\"" + workId + "\"}";

    }

    return
        "{\"status\":\"error\","
        "\"method\":\"/analogread\","		
        "\"reason\":\"invalid_input_mode\","
		"\"workId\":\"" + workId + "\"}";
}

// SG90 伺服馬達角度控制（/servo 工具的實作）
// 角度範圍：0~180 度（SG90 物理限制）
// 注意：目前僅支援 GPIO 12（servo12 實例），其他腳位需另行宣告實例
String tool_servo(AmebaServo &servo, int pin, int angle, String workId) {
    if (!servo.attached())
        servo.attach(pin); // 若未綁定腳位則先綁定（避免重複 attach）
    if (angle < 0 || angle > 180) { // 超出有效範圍：回傳錯誤 JSON
        return 
            "{\"status\":\"error\","
            "\"method\":\"/servo\","				
			"\"reason\":\"invalid_servo_angle\","
			"\"workId\":\"" + workId + "\"}";
	}
		
    servo.write(angle);

// 從 DHT11 溫溼度感測器讀取資料（/dht11 工具的實作）
// 回傳 JSON 格式結果，包含溫度（℃）與相對溼度（%）
// 讀取失敗時（isnan）回傳錯誤 JSON 供 Gemini 決定是否重試
String tool_dht11(String workId) {
  float h = dht.readHumidity();        // 讀取相對溼度（%）
  float t = dht.readTemperature();     // 讀取攝氏溫度（℃）
  // 讀取失敗通常表示感測器未正確接線或通訊超時
  if (isnan(h) || isnan(t)) {
    return "{\"status\":\"error\","
		   "\"method\":\"/dht11\","	
           "\"reason\":\"dht11_read_failed\","
		   "\"workId\":\"" + workId + "\"}";	

  }

  return
    "{\"status\":\"success\","
    "\"method\":\"/dht11\","
    "\"temperature\":" + String(t) + ","
    "\"humidity\":"    + String(h) + ","
	"\"workId\":\"" + workId + "\"}";		
}

// 詢問 Gemini 目前工作流程是否已完成（多步驟工具鏈的遞迴核心）
// reCheck=false 時直接返回（批次工具中的非末尾項目，避免過度遞迴）
// task：原始使用者任務字串，提供給 Gemini 作上下文判斷
void evaluateWorkflowContinuation(String workId, bool reCheck, String task = "") {
    if (!reCheck) return; // 非最後一個工具：跳過（防止每步都觸發遞迴）

    String prompt =
        "Analyze the execution result and determine whether the workflow is complete.\n";

    if (task != "") {
        prompt += "User task request:\n" + task + "\n\n";
    }

    prompt +=
        "If additional hardware actions are strictly required, "
        "return ONLY a valid tool_call JSON.\n"
        "If the workflow is already complete, return EXACTLY: NONE.\n"
        "If no tool action is required and a user-facing reply is needed, "
        "respond naturally in the user's language.\n"
        "Avoid repeating the same meaning as your immediately previous response during the same workflow. If a new workflow or task begins, normal responses are allowed even if similar to previous ones.\n"
        "Do not include explanation or extra text.";

    handleAgentResponse(workId, geminiChatRequest(workId, prompt));
}

      command = geminiChatRequest(workId, "Reply the following text in the user's language:\n\n" + command);
      
      replyUserMessage(workId, command, telegrambotKeyboard);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", "Command list" + timestamps);
        historicalMessages += buildGeminiMessage("model", command + timestamps);
        xSemaphoreGive(stateMutex);
      }
      
    }      
    else {
      String response = geminiChatRequest(workId, command);
      handleAgentResponse(workId, response);
      
    }	
}

// 無效的 JSON 將被拒絕並記錄至序列埠
// 格式錯誤的資料不會觸發任何工具執行
void handleAgentResponse(String workId, String message) {

  String rawMessage = message;
  
  message.replace("\\\"", "\""); 
  message.replace("\\\\", "\\");             
  message.replace("\\n", "");
  message.replace("\n", "");
  message.replace("\\r", "");
  message.replace("\r", "");
  message.replace("\\t", "");
  message.replace("\t", "");
  message.replace(String(char(0)), "");  
  message.replace("\\-", "-");
  message.replace("\\*", "*");
  message.replace("\\_", "_");
  message.replace("\\#", "#");              

  if (message.startsWith("{") && message.indexOf("}") != -1) {
    // ── 單一工具呼叫（物件格式）────────────────────────────────
    message = message.substring(0, message.lastIndexOf("}") + 1); // 截至最後一個 }
    JsonObject obj;
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, message);
    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
      return; // 無效 JSON：拒絕執行，不做任何硬體動作
    }
    obj = doc.as<JsonObject>();
    String method     = obj["method"].as<String>(); // 工具名稱（如 /digitalwrite）
    JsonObject params = obj["params"];
    executeTool(workId, method, params);
  }
  else if (message.startsWith("[") && message.indexOf("]") != -1) {
    // ── 批次工具呼叫（陣列格式）────────────────────────────────
    message = message.substring(0, message.lastIndexOf("]") + 1);
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, message);
    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
      return;
    }
    JsonArray toolsArray = doc.as<JsonArray>();
    int toolCount = toolsArray.size();
    for (int i = 0; i < toolCount; i++) {
      JsonObject toolObject = toolsArray[i];
      if (toolObject.isNull()) continue; // 跳過空物件
      String command    = toolObject["method"].as<String>();
      JsonObject params = toolObject["params"];
      if (command == "" || params.isNull()) {
        // 不完整的工具物件：中止後續所有工具（原子性保護）
        Serial.println("[偵測到不完整的工具呼叫] → 中止剩餘工具執行");
        break;
      }
      bool isLast = (i == toolCount - 1); // 僅最後一個工具啟用 reCheck
      executeTool(workId, command, params, isLast);
    }
  }
  else {
    if (message.startsWith("[") || message.startsWith("{")) {
      Serial.println("[DEBUG] Json parse failed: (handleAgentResponse)\n" + message);
      replyUserMessage(workId, "Json parse failed (handleAgentResponse). Please type \"Continue\"");
	  
    } else if (message != "NONE") {
      message = rawMessage;

      if (workId.startsWith(String(taskTags[0]))) {
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");        
        message.replace("___", "");
      }
      else {
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        message.replace("&", "&amp;");
        message.replace("<", "&lt;");
        message.replace(">", "&gt;");
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");        
        message.replace("___", "");
      } 
      
      replyUserMessage(workId, message);
    }
  }
}

// 將二進位資料（音訊/影像）以 Base64 編碼後傳送至 Gemini 進行分析
// 主要用途：Telegram 語音訊息的語音轉文字（STT）
// mimeType：MIME 類型字串（例如 "audio/ogg; codecs=opus"）
// prompt：分析提示詞（例如 "Transcribe this audio to text exactly as spoken."）
String sendFileToGemini(uint8_t* fileinput, size_t fileSize, String mimeType, String prompt) {
  // 計算 Base64 編碼後的長度並配置緩衝區
  // Base64 將每 3 個輸入位元組編碼為 4 個輸出字元（佔用約 33% 額外空間）
  int   encodedLen  = base64_enc_len(fileSize);
  char* encodedData = (char*)malloc(encodedLen);
  if (!encodedData) {
    Serial.println("[STT] malloc failed for Base64 buffer"); // 堆積記憶體不足
    return "Malloc failed for Base64 encoding.";
  }
  base64_encode(encodedData, (char*)fileinput, fileSize); // 執行 Base64 編碼

  prompt.replace("\n", "");
  prompt.replace("\"", "\\\"");

  String request =
    "{\"contents\": [{\"role\": \"user\", \"parts\": ["
    "{\"inline_data\": {\"data\": \"" + String(encodedData) + "\","
    "\"mime_type\": \"" + mimeType + "\"}},"
    "{\"text\": \"" + prompt + "\"}"
    "]}]}";

  free(encodedData);

  WiFiSSLClient client;
  if (!client.connect("generativelanguage.googleapis.com", 443)) {
    Serial.println("[STT] Connection to Gemini failed");
    return "Connected to Gemini failed.";
  }

  client.println("POST /v1beta/models/" + geminiModel +
                 ":generateContent?key=" + geminiApiKey + " HTTP/1.1");
  client.println("Host: generativelanguage.googleapis.com");
  client.println("Content-Type: application/json; charset=utf-8");
  client.println("Content-Length: " + String(request.length()));
  client.println("Connection: close");
  client.println();

  for (int i = 0; i < (int)request.length(); i += 1024) {
    client.print(request.substring(i, i + 1024));
  }

  String body = "";
  unsigned long timeout = millis() + 20000;
  bool headersEnded = false;
  String line = "";

  while (client.connected() && millis() < timeout) {
    while (client.available()) {
      char c = client.read();

      if (!headersEnded) {
        if (c == '\n') {
          if (line.length() <= 1) {
            headersEnded = true;
          }
          line = "";
        } else if (c != '\r') {
          line += c;
        }
      } else {
        body += c;
      }
    }
    vTaskDelay(1);
  }

  client.stop();

  int jsonStart = body.indexOf('{');
  if (jsonStart != -1) body = body.substring(jsonStart);

  DynamicJsonDocument doc(8192);
  DeserializationError err = deserializeJson(doc, body);

  if (err) {
    Serial.println("[DEBUG] JSON parse failed: (sendFileToGemini)\n" + body);
    return "JSON parse failed (sendFileToGemini). Please try again.";
  }

  if (doc.containsKey("error")) {
    String msg = "Gemini STT Error: " + doc["error"]["message"].as<String>();
    return msg;
  }

  if (doc["candidates"][0]["content"]["parts"][0].containsKey("text")) {
    String result = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    result.replace("\n", "");
    return result;
  }

  return "No text returned from Gemini.";
}

// 從 Telegram CDN 下載二進位檔案（語音訊息）至 heap 緩衝區
// filePath：由 getTelegramFilePath() 取得的相對路徑（如 "voice/file_123.oga"）
// 回傳：malloc 配置的緩衝區指標；呼叫者負責 free()（見 task_getTelegramMessage）
// downloadedFileSize：實際下載的位元組數（全域變數，供後續 sendFileToGemini 使用）
uint8_t* downloadTelegramFile(String filePath) {
  uint8_t* voiceFile = (uint8_t*)malloc(MAX_FILE_SIZE); // 配置最大 256KB 緩衝區
  if (!voiceFile) return NULL; // malloc 失敗（記憶體不足）

  downloadedFileSize = 0;
  WiFiSSLClient client;

  if (client.connect("api.telegram.org", 443)) {
    // 使用 HTTP/1.0 防止分塊傳輸編碼（Chunked Transfer），確保回應為純二進位資料
    client.println("GET /file/bot" + telegrambotToken + "/" + filePath + " HTTP/1.0");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();
    // 跳過 HTTP 標頭：逐字元累積直到找到 "\r\n\r\n"（標頭結束標記）
    String header    = "";
    long   startTime = millis();
    while (client.connected() || client.available()) {
      if (millis() - startTime > 10000) break; // 10 秒逾時
      if (client.available()) {
        char c = client.read();
        header += c;
        if (header.endsWith("\r\n\r\n")) break; // 找到空行：標頭已完全讀取
      }
    }
    // 將二進位 body 直接讀入輸出緩衝區（逐位元組，不額外複製）
    startTime = millis();
    while ((client.connected() || client.available()) &&
           downloadedFileSize < MAX_FILE_SIZE) {
      if (millis() - startTime > 10000) break; // 10 秒逾時
      if (client.available()) {
        voiceFile[downloadedFileSize++] = client.read();
        startTime = millis(); // 每收到一個位元組就重置逾時計時
      }
    }
    client.stop();
  }

  return voiceFile;
}

/**
 * @brief Call Telegram's getFile API to convert a file_id into a download path.
 *
 * @param fileId  Telegram file_id (e.g. from a voice message object)
 * @return        Relative file path string, e.g. "voice/file_123.oga"
 */
// 呼叫 Telegram getFile API 將 file_id 轉換為下載路徑
// 語音訊息 → 取得 "voice/file_XXXXXX.oga" 格式的相對路徑
String getTelegramFilePath(String fileId) {
  WiFiSSLClient client;
  String filePath = "";
  String getAll = "", getBody = "";
  if (client.connect("api.telegram.org", 443)) {
    // 呼叫 Telegram Bot API 的 getFile 端點
    client.println("GET /bot" + telegrambotToken + "/getFile?file_id=" + fileId + " HTTP/1.1");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();

    int     waitTime  = 5000;
    long    startTime = millis();
    boolean state     = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (c == '\n') {
          if (getAll.length() == 0) state = true;
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state == true) getBody += String(c);

        startTime = millis();
      }

      if (getBody.length() > 0) break;
    }

    // 從 JSON 回應中提取 file_path
    DynamicJsonDocument doc(8192);
    deserializeJson(doc, getBody);
    filePath = doc["result"]["file_path"].as<String>();
  }

  return filePath;
}

// 輪詢 Telegram Bot API 取得最新使用者訊息（長連線輪詢模式）
// 特點：
// ‣ 使用 keep-alive 持久連線（比每次重新握手更節省時間）
// ‣ limit=1&offset=-1：只取最新一則訊息
// ‣ 支援文字訊息與語音訊息（voice）
// ‣ 語音訊息自動透過 sendFileToGemini() 進行 STT 轉文字
// ‣ 處理完畢後持久化對話歷史至 SD 卡
void getTelegramMessage() {
  const char* myDomain  = "api.telegram.org";
  String      getAll    = "", getTime = "", getBody = "";
  JsonObject          obj;
  DynamicJsonDocument doc(8192);
  String text        = "", voiceFileId = "";
  long   message_id  = 0;
  // 重複使用現有連線（keep-alive）；只在斷線時才重新建立
  if (!botClient.connected()) {
    if (lastMessageId == 0)
      Serial.println("Connect to " + String(myDomain));

    if (!botClient.connect(myDomain, 443))
      return;

    if (lastMessageId == 0)
      Serial.println("Connection successful");
  }

  while (botClient.connected()) {

    getAll  = "";
    getTime = "";
    getBody = "";

    String request = "limit=1&offset=-1&allowed_updates=message";

    botClient.println("POST /bot" + telegrambotToken + "/getUpdates HTTP/1.1");
    botClient.println("Host: "           + String(myDomain));
    botClient.println("Content-Length: " + String(request.length()));
    botClient.println("Content-Type: application/x-www-form-urlencoded");
    botClient.println("Connection: keep-alive");
    botClient.println();
    botClient.print(request);

    int           waitTime    = 5000;
    unsigned long startTime   = millis();
    bool          state       = false;
    bool          dataReceived = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (botClient.available()) {
        char c = botClient.read();

        if (c == '\n') {
          if (getAll.length() == 0)
            state = true;
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state) {
          getBody += String(c);
        } else {
          if      (getTime.indexOf("Date:") != -1)
            getTime  = "";
          else if (getTime.indexOf("Content-Type") != -1)
            getTime += "";
          else
            getTime += String(c);
        }

        startTime = millis();
      }

      // 收到回應主體後立即跳出
      if (getBody.length() > 0) {
        dataReceived = true;
        break;
      }
    }

    getTime.replace("Content-Type", "");

    String workId = String(taskTags[1]) + " " + getTime;

    if (!dataReceived || getBody == "") return;

    DeserializationError err = deserializeJson(doc, getBody);
    if (err) {
      Serial.println("[DEBUG] JSON parse failed: (getTelegramMessage)\n" + getBody);
      return;
    }
    obj = doc.as<JsonObject>();

    message_id = obj["result"][0]["message"]["message_id"].as<long>();

    if (message_id && message_id != lastMessageId) {

      long id_last  = lastMessageId;
      lastMessageId = message_id;
	  
      String fromChatId = obj["result"][0]["message"]["chat"]["id"].as<String>();
      if (fromChatId != telegrambotChatId) {
        return;
      }	  	  

      if (id_last == 0) {
        // 第一次收到訊息（lastMessageId 剛從 0 更新）：
        // 跳過此則訊息，避免重開機後重複執行歷史指令
        message_id = 0;

      } else {
        workId = String(taskTags[1]) + " " + getRtcTimeString(); // <BOT> workId

        if (obj["result"][0]["message"].containsKey("text")) {
          // ── 處理文字訊息 ────────────────────────────────────────
          text = obj["result"][0]["message"]["text"].as<String>();

          if (text == "help") {
            executeTool(workId, "/help", JsonObject()); // 快捷鍵：help → /help

          } else if (text == "null") {
            botClient.stop(); // 強制重新連線（除錯用指令）

          } else if (text.startsWith("/")) {
            executeTool(workId, text, JsonObject()); // 以 / 開頭：直接執行工具指令

          } else {
            // 純文字：送 Gemini 處理後分派回應
            text = geminiChatRequest(workId, text);
            handleAgentResponse(workId, text);
          }

          String localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory);

        } else if (doc["result"][0]["message"].containsKey("voice")) {
          // ── 處理語音訊息（STT 流程）─────────────────────────────
          // 步驟：取得 file_id → 轉換為下載路徑 → 下載 OGG 音訊 → Gemini STT 轉文字
          voiceFileId = doc["result"][0]["message"]["voice"]["file_id"].as<String>();
          String   filePath  = getTelegramFilePath(voiceFileId); // 取得 CDN 下載路徑
          uint8_t* voiceFile = downloadTelegramFile(filePath);   // 下載音訊至 heap 緩衝區
          if (voiceFile && downloadedFileSize > 0) {
            // 將音訊送至 Gemini 進行語音轉文字（audio/ogg; codecs=opus 為 Telegram 語音格式）
            text = sendFileToGemini(
              voiceFile, downloadedFileSize,
              "audio/ogg; codecs=opus",
              "Transcribe this audio to text exactly as spoken." // 要求精確轉錄，不解釋
            );
            if (text.startsWith("/"))
              executeTool(workId, text, JsonObject()); // 語音指令：直接執行工具
            else {
              // 語音對話：送 Gemini 處理後分派回應
              text = geminiChatRequest(workId, text);
              handleAgentResponse(workId, text);
            }
          }
          if (voiceFile)
            free(voiceFile); // 必須手動釋放 downloadTelegramFile 配置的緩衝區

          String localHistory2 = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory2 = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory2);
        }
      }
    }

	
    vTaskDelay(5 / portTICK_PERIOD_MS);
  }

  // WiFi 斷線重連邏輯（在 getTelegramMessage 結尾，長連線斷開後觸發）
  while (WiFi.status() != WL_CONNECTED) {
    WiFi.disconnect();
    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());
    unsigned long start = millis();
    // 等待最多 15 秒重連
    while (WiFi.status() != WL_CONNECTED && millis() - start < 15000)
      vTaskDelay(500 / portTICK_PERIOD_MS);
  }
}

// FreeRTOS 任務：處理 HTTP 請求（Port 81）
// 提供設定管理、Soul/Device/Skill 編輯、排程管理、訊息接收等 Web 端點
// 注意：Ameba 版本不需要 TWDT（無看門狗機制），ESP32-S3 版本才有
void task_getRequest(void *param) {
  (void)param;
  while (1) {
    WiFiClient client = server.available(); // 非阻塞：若無客戶端則返回假物件

    if (client) {
      String currentLine = "";  // 用於累積一行 HTTP 請求的緩衝區
      
      while (client.connected()) {
        
        
        if (client.available()) {
          char c = client.read();

          if (c == '\n') {
            if (currentLine.length() == 0) {
              // 空行 = HTTP 請求結束，開始回應
              String pageToSend = mainPageHTML;
              mainPageHTML = ""; // 清空，供下次請求使用
              // 回傳 HTTP 200 回應
              client.println("HTTP/1.1 200 OK");
              client.println("Content-Type: text/html; charset=utf-8");
              client.println("Content-Length: " + String(pageToSend.length()));
              client.println("Access-Control-Allow-Origin: *"); // 允許跨來源存取
              client.println("Cache-Control: no-cache");        // 禁止快取（動態內容）
              client.println();
              // 分 512 位元組分段傳送（避免超過 TCP buffer）
              const char* ptr = pageToSend.c_str();
              int total = pageToSend.length(), sent = 0;
              while (sent < total) {
                int chunk   = (total - sent) > 512 ? 512 : (total - sent);
                int written = client.write((const uint8_t*)(ptr + sent), chunk);
                if (written > 0) sent += written;
                else delay(5); // 寫入失敗：短暫等待（TCP 緩衝區可能已滿）
              }
              client.flush();
              break; // 本次請求處理完畢

            } else {
              currentLine = "";
            }
          }
          else if (c != '\r') {
            currentLine += c;
          }
          
          // ── 路由分派：根據 GET 路徑決定回應內容 ─────────────────
          if ((currentLine.indexOf("GET / ") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {
            // GET / → 載入設定管理頁面並填入目前設定值（讓瀏覽器顯示現有設定）
            mainPageHTML = getStringFromFile(configpageFilename);
            mainPageHTML.replace("deviceName", deviceName);
            mainPageHTML.replace("wifiSsid", wifiSsid);
            mainPageHTML.replace("wifiPassword", wifiPassword);
            mainPageHTML.replace("telegrambotToken", telegrambotToken);
            mainPageHTML.replace("telegrambotChatId", telegrambotChatId);
            mainPageHTML.replace("scheduleTimeout", String(scheduleTimeout));
            mainPageHTML.replace("geminiApiKey", geminiApiKey);
            mainPageHTML.replace("geminiModel", geminiModel);
            mainPageHTML.replace("timeZone", timeZone);		

            currentLine = "";            
          }
          else if ((currentLine.indexOf("GET /updateConfig?") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {
            // GET /updateConfig? → 更新環境設定並儲存至 SD 卡
            String workId = String(taskTags[0]) + " " + getRtcTimeString();
            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateConfig?", "");
            currentLine.replace(" HTTP/1.", "");
            if (currentLine.startsWith("{") && currentLine.endsWith("}")) {
              storeDataToFile(envFilename, currentLine); // 存入 SD 卡（env.json）
              setEnvironmentSettings(currentLine);       // 套用至全域變數
              // 重建裝置定義（附加新裝置名稱與時區）
              devicesDefinitionFinal  = devicesDefinition;
              devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
              devicesDefinitionFinal += "\nDevice timezone: " + timeZone;
              systemContentReset(); // 重建系統提示詞（立即生效）
              mainPageHTML = "Configuration updated successfully.";
            } else

// FreeRTOS 任務：提供 MJPEG 即時視訊串流（Port 82）
// 使用 multipart/x-mixed-replace 格式持續推送 JPEG 畫格
// Ameba 版本直接呼叫 Camera.getImage()，不需 malloc/copy（與 ESP32-S3 不同）
void task_getRequestStream(void *param) {
  (void)param;
  while (1) {
    WiFiClient client = serverStream.available();
    uint32_t img_addr = 0; // 相機畫格緩衝區位址（Camera.getImage() 填入）
    uint32_t img_len  = 0; // 畫格資料位元組數	
    
    if (client) {
      String currentLine = "";

      while (client.connected()) {
        

        if (client.available()) {
          char c = client.read();
          if (c == '\n') {
            if (currentLine.length() == 0) {
             String head = "--Taiwan\r\nContent-Type: image/jpeg\r\n\r\n";
            client.println("HTTP/1.1 200 OK");
            client.println("Access-Control-Allow-Origin: *");
            client.println("Connection: keep-alive");
            client.println("Content-Type: multipart/x-mixed-replace; boundary=Taiwan");
            client.println();
            // 連續推送畫格（直到客戶端斷線）
            while (client.connected()) {
              Camera.getImage(0, &img_addr, &img_len); // 從相機 DMA 取得最新畫格
              uint8_t *fbBuf = (uint8_t*)img_addr;
              size_t fbLen   = img_len;
              client.print(head); // 傳送 MJPEG 幀頭
              for (size_t n=0;n<fbLen;n=n+1024) {
                  if (n+1024<fbLen) {
                    client.write(fbBuf, 1024);
                    fbBuf += 1024;
                }
                else {
                  size_t remainder = fbLen - n;
                  if (remainder > 0)
                    client.write(fbBuf, remainder);
                }
              }
              client.print("\r\n");
              
              
              vTaskDelay(10 / portTICK_PERIOD_MS);
            }
            break;
            } else {
              currentLine = "";
            }
          }
          else if (c != '\r') {
            currentLine += c;
          }

          if (currentLine.indexOf(" HTTP/1.")!=-1) {
            currentLine="";
          }
        }
        else {
          vTaskDelay(1); 
        }
      }
      client.stop();
    }
    else {
      vTaskDelay(5 / portTICK_PERIOD_MS);
    }
  }
}

// 將整數值格式化為補零後的兩位數字串

String twoDigits(int value) {

// 檢查循環任務（year==0）今日是否已執行
// 日期變更時自動清空每日執行記錄（跨日重置）
// 注意：此 Ameba 版本使用 time()，而非 rtc.Read()（因 getExecuteScheduleTasksJson 已改用 time()）
bool isExecutedToday(String task) {
  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  int today = now->tm_mday;
  // 若日期已變更（跨日），清空今日記錄（新的一天重新計算）
  if (executedTodayDate != today) {
    executedTodayTasks = ""; // 清空記錄
    executedTodayDate  = today;
  }
  // 以 |task| 格式搜尋（防止子字串誤配）
  return executedTodayTasks.indexOf("|" + task + "|") != -1;
}

// 將循環任務標記為今日已執行（以 |task| 格式追加至記錄字串）
void markExecutedToday(const String &task) {
  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  executedTodayDate  = now->tm_mday;  // 同步今日日期
  executedTodayTasks += "|" + task + "|"; // 追加記錄（格式：|task1||task2|）
}

// 過濾排程 JSON 陣列，回傳所有未完成任務的純文字格式列表
// 循環任務（year==0）若今日已執行則略過；一次性任務若 executed==true 則略過
// 回傳格式：*** YYYY/M/D HH:mm:ss ***\n<task描述>\n（每項一個區塊）
String getUnfinishedScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);
  if (err) {
    Serial.println("Schedule JSON parse failed");
    return "[]";
  }
  
  JsonArray tasks = doc.as<JsonArray>();

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();

  String result = "";
  
  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();
       
      if (executed) continue;
  
      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();
                
      String scheduleStr;
      serializeJson(task["schedule"], scheduleStr);
      String compareTask = scheduleStr + " " + task["task"].as<String>();

      if (year == 0 && isExecutedToday(compareTask))
          continue;

      result += "*** " + String(year) + "/" + String(month) + "/" + String(day) + " " + String(hour) + ":" + String(minute) + ":" + String(sec) + " ***\n" + task["task"].as<String>() + "\n";
  }
  
  if (result == "") result = "NULL";

  return result;
}

// 從完整排程 JSON 陣列中篩選出目前到期需執行的任務
// 篩選條件：
//   ‣ executed==true 的任務略過（已完成）
//   ‣ 循環任務（year==0）今日已執行則略過
//   ‣ 超過 scheduleTimeout 分鐘的過期任務略過
//   ‣ 年/月/日 為 0 時以當前日期補全（循環任務的「每天」邏輯）
// 回傳到期任務的 JSON 陣列字串，無任務時回傳 "[]"
String getExecuteScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);
  if (err) {
    Serial.println("Schedule JSON parse failed");
    return "[]";
  }
  JsonArray tasks = doc.as<JsonArray>();
  // 從 Ameba 硬體 RTC 讀取當前時間（epoch 秒數）
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();

  String result = "";
  
  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();
       
      if (executed) continue;
  
      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();

      // 循環任務（year/month/day 為 0）：以當前日期補全，實現「每天此時執行」
      int resolvedYear  = (year  == 0) ? (now->tm_year + 1900) : year;
      int resolvedMonth = (month == 0) ? (now->tm_mon  + 1)    : month;
      int resolvedDay   = (day   == 0) ? (now->tm_mday)        : day;

      struct tm tmTask = {};
      tmTask.tm_year = resolvedYear  - 1900;
      tmTask.tm_mon  = resolvedMonth - 1;
      tmTask.tm_mday = resolvedDay;
      tmTask.tm_hour = hour;
      tmTask.tm_min  = minute;
      tmTask.tm_sec  = sec;

      time_t taskTime = mktime(&tmTask); // 將任務時間轉換為 epoch 秒數
      if (mktime(&tmTask) <= rawtime) {
        time_t diff = rawtime - taskTime;
        // 超過容錯視窗（scheduleTimeout 分鐘）的過期任務略過
        // （防止裝置長時間離線後重開機時批量執行大量舊任務）
        if (scheduleTimeout > 0 && rawtime > taskTime && diff > (time_t)(scheduleTimeout * 60))
            continue;
                
        String scheduleStr;
        serializeJson(task["schedule"], scheduleStr);
        String compareTask = scheduleStr + " " + task["task"].as<String>();

        if (year == 0 && isExecutedToday(compareTask))
            continue;

        JsonObject item = resultArray.createNestedObject();
        item["task"] = task["task"].as<String>();
        item["schedule"] = schedule;

      }
  }

  serializeJson(resultDoc, result);
  return result;
}

// FreeRTOS 任務：每 60 秒檢查一次到期排程任務
// 執行流程：① 等待 60 秒 → ② 停止 botClient 取得 SSL 資源 → ③ 過濾到期任務
// → ④ 逐一送 Gemini 執行 → ⑤ 更新執行狀態 → ⑥ 持久化至 SD 卡
void task_time_scheduling(void *param) {
  (void)param;
  while (1) {
    // 將 60 秒等待切割為 6×10 秒（Ameba 無 WDT，僅為邏輯分段）
    for (int i = 0; i < 6; i++)
      vTaskDelay(10000 / portTICK_PERIOD_MS);
    // 取得 botClientMutex 並停止 Telegram 長連線
    // 原因：排程執行可能需要呼叫 Gemini Vision，需要 SSL 客戶端資源
    if (xSemaphoreTake(botClientMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      botClient.stop(); // 停止 Telegram 輪詢連線
      xSemaphoreGive(botClientMutex);
    }
    vTaskDelay(2000 / portTICK_PERIOD_MS); // 等待 2 秒確保連線完全關閉

    String workId = String(taskTags[3]) + " " + rtcFormatTime;

    // 若 RTC 尚未初始化（rtcYear==0），先同步時間再繼續
    if (rtcYear == 0) {
      Serial.println("[DEBUG] RTC time is not initialized.");
      executeTool(workId, "/syncrtc", JsonObject(), false); // 強制同步（reCheck=false）
      if (rtcYear == 0) continue; // 同步仍失敗（無 WiFi），跳過此輪
    }

    String localSchedule = "";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      if (scheduleTasks.startsWith("[") && scheduleTasks.indexOf("]") !=-1)
        scheduleTasks = scheduleTasks.substring(0, scheduleTasks.lastIndexOf("]") + 1);
      localSchedule = scheduleTasks;
      xSemaphoreGive(stateMutex);
    }

    if (localSchedule.startsWith("[") && localSchedule.indexOf("]") !=-1) {
 
      String unfinishedScheduleTasksJson = getExecuteScheduleTasksJson(localSchedule);

      if (unfinishedScheduleTasksJson.startsWith("[") && unfinishedScheduleTasksJson.indexOf("]") !=-1) {
        unfinishedScheduleTasksJson = unfinishedScheduleTasksJson.substring(0, unfinishedScheduleTasksJson.lastIndexOf("]") + 1);

        String response = "";

        DynamicJsonDocument doc(8192);
      
        DeserializationError err = deserializeJson(doc, unfinishedScheduleTasksJson);
        if (err) {
          Serial.println("[DEBUG] JSON parse failed: (task_time_scheduling)\n" + unfinishedScheduleTasksJson);
          continue; // 不使用 return：保持任務存活（continue 跳到下一輪迴圈，繼續等待下個 60 秒）
        }

        JsonArray tasks = doc.as<JsonArray>();
        
        for (JsonObject obj : tasks) {
          // 逐一處理每個到期任務
          String taskName = obj["task"].as<String>(); // 任務名稱（僅用於日誌）
          String schedule = obj["schedule"].as<String>(); // 排程時間 JSON（供 markExecutedToday 使用）
          String item     = obj["task"].as<String>();      // 任務描述字串（注入 Gemini 提示詞）           

          String prompt =
            "This is a deterministic scheduling execution step. "
          
            "\n\nUnfinished scheduled tasks:\n" +
            item +
          
            "\n\nThe task list above already contains ONLY tasks that have not been executed. "
            "Evaluate EVERY task in this list independently. "
            "If a task's scheduled time is less than or equal to the current time, "
            "it MUST be executed immediately. "
            "Do NOT skip any eligible task. "
            "More than one task may be eligible at the same time. "
            "If multiple tasks are eligible, execute ALL of them in the same response. "
            "Tasks whose scheduled time is still in the future must be ignored. "
          
            "Output rules: "
            "1. If no task is eligible for execution, return EXACTLY: NONE. "
            "2. If one or more tasks are eligible, return tool_call JSON for ALL eligible tasks. "
            "3. Never return natural language. "
            "4. Never explain. "
            "5. Never summarize. "
            "6. Never ask questions. "
            "7. Never claim success without tool execution results. "
            "8. Process every task in the provided task list. "
            "9. A task remains executable forever after its scheduled time has passed until it is marked executed=true. "
            "10. Do not stop after the first eligible task.";

          response = geminiChatRequest(workId, prompt);

          handleAgentResponse(workId, response);

          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            markExecutedToday(schedule + " " + item);
            xSemaphoreGive(stateMutex);
          }
        }
        
        // 有任務被執行：同步狀態並持久化
        if (tasks.size() > 0) {
          // 請 Gemini 根據執行歷史更新 executed 欄位（reCheck=false 防止遞迴）
          executeTool(workId, "/updateScheduleStatus", JsonObject(), false);
          // 安全讀取並持久化今日執行記錄與對話歷史
          String localExecuted = "", localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localExecuted = executedTodayTasks;
            localHistory  = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(scheduleExecutedTodayTasksFilename, localExecuted); // 今日執行記錄
          storeDataToFile(memoryFilename, localHistory);                       // 對話歷史
        }
      }
      
    }
  }
}

// 初始化 WiFi（同時啟動 AP 熱點 + STA 連路由器雙模式）
// AP 模式：固定 IP 192.168.1.1，頻道 2（避免與 STA 干擾）
// STA 模式：連接指定路由器，最多嘗試 2 次，每次等待最長 15 秒
void initWiFi() {
  WiFi.enableConcurrent(); // 啟用 Ameba AP + STA 並行模式
  // 啟動 AP 熱點（channel_ap="2"，0=不隱藏 SSID）
  WiFi.apbegin((char*)apSsid.c_str(), (char*)apPassword.c_str(), channel_ap, 0);
  // 嘗試連接路由器，最多 2 次
  for (int i = 0; i < 2; i++) {
    if (wifiSsid == "") break; // 未設定 SSID：跳過 STA 連線
    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());
    delay(1000); // 給 WiFi 初始化時間
    Serial.println();
    Serial.print("Connecting to "); Serial.println(wifiSsid);
    unsigned long StartTime = millis();
    // 等待連線，最多 15 秒
    while (WiFi.status() != WL_CONNECTED) {
      delay(500);
      if ((StartTime + 15000) < millis()) break; // 逾時：進入下一次重試
    }
  }
}

// 解析 env.json 並套用所有環境設定至全域變數
void setEnvironmentSettings(String jsonString) {
  DynamicJsonDocument doc(8192);
  DeserializationError error = deserializeJson(doc, jsonString);
  if (error) {
    Serial.println("[DEBUG] JSON parse failed : (setEnvironmentSettings)\n" + jsonString);
    return; // JSON 格式錯誤：保留舊設定值
  }
  JsonObject obj = doc.as<JsonObject>();
  // 逐一映射 JSON 欄位至對應全域變數
  deviceName        = obj["device_name"].as<String>();        // 裝置名稱
  wifiSsid          = obj["wifi_ssid"].as<String>();          // WiFi SSID
  wifiPassword      = obj["wifi_pass"].as<String>();          // WiFi 密碼
  telegrambotToken  = obj["telegramBot_token"].as<String>();  // Telegram Bot Token
  telegrambotChatId = obj["telegramBot_chatID"].as<String>(); // 允許的 Chat ID
  geminiApiKey      = obj["gemini_apikey"].as<String>();       // Gemini API 金鑰
  geminiModel       = obj["gemini_model"].as<String>();        // Gemini 模型版本
  scheduleTimeout   = obj["schedule_timeout"].as<int>();       // 排程容錯視窗（分鐘）
  timeZone          = obj["timezone"].as<String>();            // IANA 時區字串
}

// 將 IPAddress 物件轉為點分十進位字串（如 192.168.1.100）
String Ip2String(IPAddress ip) {
  return String(ip[0]) + "." + String(ip[1]) + "." + String(ip[2]) + "." + String(ip[3]);
}

// ════════════════════════════════════════════════════════════════
// Arduino 初始化函式（執行一次）
// 順序：Serial → Mutex → LED → SD 設定檔 → WiFi → Soul/Device/Skill
// → 排程還原 → 系統提示詞 → 記憶還原 → 相機 → RTC → HTTP Server → FreeRTOS Tasks
// ════════════════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200); // 開啟 Serial（115200 baud）
  // WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0); // 停用欠壓重置（可選）

  // ── 步驟 1：建立所有 Mutex（必須在任何使用它們的函式/任務之前）──
  // setup() 本身呼叫 rtcInitialTime()/replyUserMessage()，這些函式需要 Mutex
  botClientMutex = xSemaphoreCreateMutex();
  stateMutex     = xSemaphoreCreateMutex();
  sdMutex        = xSemaphoreCreateMutex();
  imageMutex     = xSemaphoreCreateMutex();
  if (!botClientMutex || !stateMutex || !sdMutex || !imageMutex) {
    Serial.println("[DEBUG] Failed to create mutexes. Restarting the MCU...");
    delay(2000);
    NVIC_SystemReset(); // Ameba 的軟體重置
  }
  // ── 步驟 2：設定指示燈 LED ─────────────────────────────────
  pinMode(ledPin, OUTPUT); // 設定為輸出模式（WiFi 連線成功時閃爍）
  
  // ── 步驟 3：從 SD 卡讀取環境設定並初始化 WiFi ────────────────
  String env = getStringFromFile(envFilename);
  Serial.println("env.json len: " + String(env.length()));
  if (env != "") setEnvironmentSettings(env); // 套用設定至全域變數
  initWiFi(); // 啟動 AP + STA 雙模式

  // ── 步驟 4：載入人格提示詞與硬體定義 ─────────────────────────
  String soul = getStringFromFile(soulFilename);
  Serial.println("Soul.md len: " + String(soul.length()));
  if (soul != "") geminiRole = soul;
  String device = getStringFromFile(deviceFilename);
  Serial.println("device.md len: " + String(device.length()));
  if (device != "") devicesDefinition = device;
  // 附加裝置名稱與時區至裝置定義（讓 Gemini 識別自身身份與時區）
  devicesDefinitionFinal  = devicesDefinition;
  devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
  devicesDefinitionFinal += "\nDevice timezone: " + timeZone;
  // soul.md 與 device.md 為必要檔案，缺少任一則重啟
  if (geminiRole.length() == 0 || devicesDefinition.length() == 0) {
    Serial.println("System configuration failed. Restarting the MCU...");
    delay(5000);
    NVIC_SystemReset();
  }

  // 讀取技能定義（可選）
  String skill = getStringFromFile(skillFilename);
  Serial.println("skill.md len: " + String(skill.length()));
  if (skill != "") skillsDefinition = skill;
  // 讀取排程任務 JSON
  String schedule = getStringFromFile(scheduleFilename);
  Serial.println("schedule.md len: " + String(schedule.length()));
  if (schedule != "") scheduleTasks = schedule;
  // 讀取今日已執行排程記錄（重開機後還原，避免重複執行）
  String scheduleExecutedTodayTasks = getStringFromFile(scheduleExecutedTodayTasksFilename);
  Serial.println("scheduleTodayExecuted.md len: " + String(scheduleExecutedTodayTasks.length()));
  if (scheduleExecutedTodayTasks != "") executedTodayTasks = scheduleExecutedTodayTasks;
  // ── 步驟 5：建立三個版本的系統提示詞 ──────────────────────────
  systemContent        = buildGeminiMessage("user", geminiRole, 0) + buildGeminiMessage("model", "OK");
  systemContentTools   = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule + skillsDefinition + toolsDefinition, 0) + buildGeminiMessage("model", "OK");
  systemContentNoTools = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule, 0) + buildGeminiMessage("model", "OK");
  // 讀取並還原對話歷史記憶（讓 Gemini 記住上次對話）
  String memory = getStringFromFile(memoryFilename);
  Serial.println("memory.md len: " + String(memory.length()));
  if (memory != "") historicalMessages = memory;
  // ── 步驟 6：初始化相機 ────────────────────────────────────────
  config.setRotation(0);           // 影像旋轉角度（0=不旋轉）
  Camera.configVideoChannel(0, config); // 設定視訊通道 0（主相機）
  Camera.videoInit();              // 初始化相機驅動
  Camera.channelBegin(0);          // 啟動視訊通道 0
  // ── 步驟 7：NTP/Gemini 時間同步（寫入硬體 RTC）──────────────
  rtcInitialTime("RTC Initial Time"); // 透過 Gemini 取得時間並寫入 RTC
  // 傳送開機通知至 Telegram（附上自訂鍵盤）
  replyUserMessage(String(taskTags[1]) + " " + getRtcTimeString(), "RTC START: " + getRtcTimeString(), telegrambotKeyboard);
  // 重要：必須在載入 executedTodayTasks 後立即同步今日日期
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);
  executedTodayDate = now->tm_mday; // 初始化今日日期（供 isExecutedToday() 使用）
  // ── 步驟 8：啟動 HTTP 伺服器 ────────────────────────────────
  server.begin();       // Port 81：設定管理 Web 伺服器
  serverStream.begin(); // Port 82：MJPEG 視訊串流伺服器  

  // ── 步驟 9：建立 FreeRTOS 任務 ──────────────────────────────
  // HTTP Web 請求任務（16KB：HTML 組裝 + Gemini 呼叫）
  if (xTaskCreate(task_getRequest, "task_getRequest",
                  16384, NULL, tskIDLE_PRIORITY + 1, NULL) != pdPASS)
    Serial.println("Create task_getRequest failed");

  // MJPEG 視訊串流任務（16KB：直接使用相機 DMA 緩衝）
  if (xTaskCreate(task_getRequestStream, "task_getRequestStream",
                  16384, NULL, tskIDLE_PRIORITY + 1, NULL) != pdPASS)
    Serial.println("Create task_getRequestStream failed");

  // Telegram 長輪詢任務（16KB：含 SSL 握手與 JSON 解析）
  if (xTaskCreate(task_getTelegramMessage, "task_getTelegramMessage",
                  16384, NULL, tskIDLE_PRIORITY + 1, NULL) != pdPASS)
    Serial.println("Create task_getTelegramMessage failed");

  // 排程執行任務（6KB：主要操作字串，記憶體需求較小）
  if (xTaskCreate(task_time_scheduling, "task_time_scheduling",
                  6144, NULL, tskIDLE_PRIORITY + 1, NULL) != pdPASS)
    Serial.println("Create task_time_scheduling failed");

  // ── 步驟 10：LED 指示燈 + 輸出 STA IP ──────────────────────
  Serial.println("AP mode");
  Serial.println("fuClaw Manager: http://192.168.1.1:81"); // AP 管理頁面
  Serial.println("Video stream: http://192.168.1.1:82");   // AP 視訊串流
  Serial.println("AP ssid : "     + apSsid);
  Serial.println("AP password : " + apPassword);
  Serial.println();
  if (WiFi.status() == WL_CONNECTED) {
    // WiFi 連線成功：LED 閃爍 3 次（視覺確認）
    for (int i = 0; i < 3; i++) {
      digitalWrite(ledPin, 1); delay(300);
      digitalWrite(ledPin, 0); delay(300);
    }
    Serial.println("STA mode");
    Serial.println("fuClaw Manager: http://" + Ip2String(WiFi.localIP()) + ":81"); // STA 管理頁面
    Serial.println("Video stream: http://"   + Ip2String(WiFi.localIP()) + ":82"); // STA 串流
    Serial.println();
    // 將裝置 IP 注入歷史記憶（讓 Gemini 知道裝置位址）
    historicalMessages += buildGeminiMessage("user", "Device IP: " + Ip2String(WiFi.localIP()));
  }
  // ── 步驟 11：初始化外部模組 ────────────────────────────────
  servo12.attach(12); // 綁定 SG90 伺服馬達至 GPIO 12（窗戶致動器）
  dht.begin();        // 初始化 DHT11 感測器（需在 setup 結束前呼叫）
}

// Arduino 主迴圈（純 idle）
// 所有工作均由 FreeRTOS 任務執行；此處只需 yield 讓出 CPU
void loop() {
  vTaskDelay(1000 / portTICK_PERIOD_MS); // 每秒 yield（CPU 佔用率近乎 0）
}
