系統概述
============================================================
fuClaw 是一套執行於 Realtek Ameba Pro2 開發板上的
嵌入式多模態 AI 代理框架，支援以下兩款硬體：
  - AMB82-mini
  - HUB 8735 Ultra

整合功能包含：
  - Telegram Bot API（HTTPS 長輪詢）
  - Gemini 網頁聊天介面
  - Google Gemini generateContent API
  - Gemini 接地式（grounded）網路搜尋
  - Gemini 多模態視覺推理（Vision）
  - 提示詞驅動的 JSON 工具路由機制
  - GPIO 數位 / 類比輸出入控制
  - 相機拍照與影像上傳
  - 持久化對話記憶（SD 卡）
  - FreeRTOS 並發多工排程

整個系統作為混合式自主代理運行：
對話 + 推理 + 工具 + 視覺 + 記憶 + 硬體控制

執行架構（Runtime Architecture）
============================================================
Telegram / 網頁聊天使用者
  ↓
通訊任務（Telegram 長輪詢 / 網頁聊天）
  ↓

  ↓

  ↓

============================================================
執行模型說明
============================================================
本系統採用「提示詞編排工具路由」機制。

Gemini 不使用原生 Function Calling API，而是：
  - 由 Gemini 輸出結構化 JSON 的 tool_call 回應
  - 本地韌體驗證所有工具呼叫
  - 格式錯誤的 JSON 一律拒絕執行
  - 執行順序嚴格為循序（sequential）
  - 硬體動作絕不模擬，只執行實際操作

原子執行原則：
每次回應只能執行一個硬體動作：
  - 一個腳位
  - 一種操作
  - 一個數值

多步驟工作流程以逐步方式依序執行。

  - 綠色 LED：GPIO 24
  - 藍色 LED：GPIO 23

HUB 8735 Ultra
  - 按鈕：GPIO 12（僅輸入，低電位觸發）
  - 可用 GPIO：0~13, 15~26
  - ADC 腳位：0, 1, 2, 9, 10
  - PWM 腳位：11, 12, 13, 19, 20, 21, 22, 23, 24
  - 綠色 LED：GPIO 25
  - 藍色 LED：GPIO 26
  - 補光 LED：GPIO 13

外接模組（已確認）
  - 緊急按鈕      ：GPIO 1（高電位觸發）
  - 光線感測器    ：GPIO 2（類比輸入，0-1023）
  - 警示燈        ：GPIO 11（PWM 輸出，0-255）
  - 窗戶致動器 SG90：AMB82-mini → GPIO 5 / HUB 8735 → GPIO 12（伺服，0-180°）
  - DHT11 感測器  ：AMB82-mini → GPIO 8 / HUB 8735 → GPIO 20

只有上述列出的外部硬體對應關係已確認。
請勿假設其他 GPIO 腳位已連接任何額外的硬體。
未知的硬體對應關係需先確認後才能使用。
所有 GPIO 值在執行前都會進行嚴格驗證。

// ============================================================
// URL 解碼函式（urldecode）
// 將 URL 百分比編碼（%XX）還原為原始字元。
// 例：%20 → 空格，%2F → /，+ → 空格
// 用於解析網頁伺服器收到的 GET 請求參數。
// ============================================================
String urldecode(const String& input) {
    String result = "";
    result.reserve(input.length());  // 預先分配記憶體，避免頻繁重新分配
    for (int i = 0; i < (int)input.length(); i++) {
        if (input[i] == '%' && i + 2 < (int)input.length()) {
            // 讀取 %XX 後面兩個十六進位字元，轉換為對應 ASCII 字元
            char hex[3] = { input[i+1], input[i+2], '\0' };
            uint8_t val = (uint8_t)strtol(hex, nullptr, 16);
            result.concat((char)val);
            i += 2;  // 跳過已處理的兩個十六進位字元
        } else if (input[i] == '+') {
            result += ' ';  // + 號代表空格（application/x-www-form-urlencoded 規範）
        } else {
            result += input[i];  // 其他字元直接保留
        }
    }
    return result;
}

// ============================================================
// URL 編碼函式（urlencode）
// 將字串轉換為 URL 百分比編碼格式。
// 英文字母、數字及 -_.~ 保留不編碼，其餘字元編碼為 %XX。
// 用於組合 Telegram API 的 GET 請求 URL。
// ============================================================
String urlencode(String str) {
  const char *msg = str.c_str();
  const char *hex = "0123456789ABCDEF";
  String encodedMsg = "";
  while (*msg != '\0') {
    // 不需編碼的安全字元：字母、數字、-、_、.、~
    if (('a' <= *msg && *msg <= 'z') || ('A' <= *msg && *msg <= 'Z') ||
        ('0' <= *msg && *msg <= '9') || *msg == '-' || *msg == '_' ||
        *msg == '.' || *msg == '~') {
      encodedMsg += *msg;
    } else {
      // 其他字元轉為 %XX 格式（高 4 位元 + 低 4 位元）
      encodedMsg += '%';
      encodedMsg += hex[(unsigned char)*msg >> 4];
      encodedMsg += hex[*msg & 0xf];
    }
    msg++;
  }
  return encodedMsg;
}

// ============================================================
// 向 Gemini API 發送簡短請求，取得目前 GMT 日期時間字串
// 這個函式用於 RTC 初始時間同步：
// 先取得 HTTP 回應標頭中的 Date 欄位（GMT 時間），
// 再交由 rtcInitialTime() 透過另一次 Gemini 請求轉換為本地時間。
//
// 回傳值：GMT 日期時間字串（來自 HTTP 標頭的 Date 欄位）
// 若連線失敗，回傳固定提示字串，讓後續的 Gemini 請求改用搜尋功能取得。
// ============================================================
String getGeminiDatetime() {

  // 建立一個簡單的 Gemini 對話請求（只包含基本角色提示詞，不帶工具定義）
  String contents = systemContent + buildGeminiMessage("user", "I am fuClaw!", true);

  // 組合完整的 Gemini API 請求 JSON 主體
  String request = "{\"contents\": [" + contents +
                   "],\"generationConfig\": {\"maxOutputTokens\": " +
                   geminiMaxOutputTokens +
                   ", \"temperature\": " + geminiTemperature + "}}";

  WiFiSSLClient client;
  String getDatetime = "";

  if (client.connect("generativelanguage.googleapis.com", 443)) {
    // 傳送 HTTP POST 請求至 Gemini generateContent 端點
    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.1");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    // 以 1024 位元組為單位分批傳送請求主體（避免大型 JSON 一次傳送失敗）
    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    int waitTime = 5000;              // 最多等待 5 秒
    unsigned long startTime = millis();
    bool getStatus = false;           // 標記是否已找到 Date 欄位並開始讀取

    // 讀取 HTTP 回應標頭，目標是擷取 "Date:" 欄位的值
    while ((startTime + waitTime) > millis()){
      vTaskDelay(100 / portTICK_PERIOD_MS);  // 讓出 CPU 給其他 FreeRTOS 任務

      while (client.available()){
        char c = client.read();

        // 一旦開始讀取 Date 值（getStatus = true），遇到換行即停止
        if (getStatus == true && c == '\n') {
          waitTime = 0;
          break;
        }
        // 找到 "Date:" 字樣後，重置緩衝區並開始記錄後續內容
        if (getDatetime.indexOf("Date:")!=-1) {
          getDatetime = "";
          getStatus = true;
        }
        else
          getDatetime += String(c);

        startTime = millis();  // 每收到一個字元就重置逾時計時器
      }
    }

    client.stop();

  } else {
    // 連線失敗：回傳固定提示，讓 Gemini 改用 Google Search 取得時間
    getDatetime = "Use grounded search to retrieve the current GMT date and time.";
  }

  if (getDatetime == "") {
    getDatetime = "Use grounded search to retrieve the current GMT date and time.";
  }

  return getDatetime;
}

// ============================================================
// 取得 RTC 目前時間的格式化字串
// filename 參數決定輸出格式：
//   filename = true  → 適合當檔名的格式：YYYYMDDHHMM SS
//   filename = false → 人類可讀格式：YYYY/M/D HH:MM:SS（預設）
// 此函式被廣泛用於：日誌記錄、工作 ID 生成、排程比對等。
// ============================================================
String getRtcTimeString(bool filename = false) {

  // 從硬體 RTC 讀取 Unix epoch 時間
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;

  // 轉換為本地時間結構
  struct tm *timeinfo = localtime(&rawtime);

  char buffer[32];

  if (filename == true) {
    // 檔名格式：不含分隔符號，適合嵌入檔名中（例如備份檔）
    sprintf(
      buffer,
      "%04d%d%d%02d%02d%02d",
      timeinfo->tm_year + 1900,
      timeinfo->tm_mon + 1,
      timeinfo->tm_mday,
      timeinfo->tm_hour,
      timeinfo->tm_min,
      timeinfo->tm_sec
    );
  }
  else {
    // 人類可讀格式：YYYY/M/D HH:MM:SS
    sprintf(
      buffer,
      "%04d/%d/%d %02d:%02d:%02d",
      timeinfo->tm_year + 1900,
      timeinfo->tm_mon + 1,
      timeinfo->tm_mday,
      timeinfo->tm_hour,
      timeinfo->tm_min,
      timeinfo->tm_sec
    );
  }

  return String(buffer);
}

// ============================================================
// RTC 初始時間同步（透過 Gemini API）
// 流程：
//   1. 呼叫 getGeminiDatetime() 取得目前 GMT 時間（從 HTTP 標頭）
//   2. 組合提示詞，請 Gemini 將 GMT 轉換為指定時區的本地時間
//   3. Gemini 回傳 JSON，解析後設定 rtcYear/Month/Day/Hour/Minute/Second
//   4. 呼叫 rtc.Init() 與 rtc.Write() 設定硬體 RTC
//
// 注意：提示詞中要求 Gemini 在轉換後加 4 秒，補償網路延遲。
// ============================================================
void rtcInitialTime(String workName) {

  rtcUpdateStatus = true;  // 標記 RTC 正在更新中

  // 建立給 Gemini 的時間轉換提示詞
  String prompt =
    "Convert this GMT datetime to " + timeZone + ".\n"
    "GMT datetime: " + getGeminiDatetime() + "\n\n"

    // 加 4 秒補償網路延遲，並要求正確處理進位（例如 59 秒 + 4 → 下一分鐘）
    "Before generating the JSON output, add exactly 4 seconds to the converted local datetime.\n"
    "Handle minute, hour, day, month, and year rollovers correctly if needed.\n\n"

    // 嚴格規定輸出格式：純 JSON，不可有任何說明文字或 Markdown
    "Output requirements:\n"
    "- Return ONLY a raw JSON object.\n"
    "- Do NOT use markdown.\n"
    "- Do NOT use code fences.\n"
    "- Do NOT explain anything.\n"
    "- Do NOT add extra text.\n"
    "- First character must be {.\n"
    "- Last character must be }.\n\n"

    // 期望的 JSON 格式範例
    "Required JSON format:\n"
    "{\n"
    "\"rtcYear\":2026,\n"
    "\"rtcMonth\":5,\n"
    "\"rtcDay\":28,\n"
    "\"rtcHour\":11,\n"
    "\"rtcMinute\":35,\n"
    "\"rtcSecond\":0\n"
    "}";

  // 傳送請求給 Gemini（tools = -1 代表不帶工具定義，純文字對話模式）
  String message = geminiChatRequest(String(taskTags[1]), prompt, -1);

  // 驗證回傳值是否為合法的 JSON 物件（以 { 開頭、以 } 結尾）
  if (message.startsWith("{") && message.endsWith("}")) {

    DynamicJsonDocument doc(1024);
    DeserializationError error = deserializeJson(doc, message);

    if (error) {
      // JSON 解析失敗：回報錯誤並提前返回
      Serial.println("[DEBUG] JSON parse failed\n" + message);
      replyUserMessage(workName, "RTC time update failed.", "");
      rtcUpdateStatus = false;
      return;
    }

    JsonObject obj = doc.as<JsonObject>();

    // 從 JSON 中提取時間各欄位（若欄位不存在則預設為 0）
    rtcYear   = obj["rtcYear"]   | 0;
    rtcMonth  = obj["rtcMonth"]  | 0;
    rtcDay    = obj["rtcDay"]    | 0;
    rtcHour   = obj["rtcHour"]   | 0;
    rtcMinute = obj["rtcMinute"] | 0;
    rtcSecond = obj["rtcSecond"] | 0;

  } else {
    Serial.println("[DEBUG] JSON parse failed : (rtcInitialTime)\n" + message);
    replyUserMessage(workName, "RTC time update failed.", "");
  }

  // 初始化 RTC 硬體並寫入時間
  rtc.Init();
  long long initTime = rtc.SetEpoch(rtcYear, rtcMonth, rtcDay, rtcHour, rtcMinute, rtcSecond);
  rtc.Write(initTime);
}

// ============================================================
// 傳送文字訊息至 Telegram Bot
// 使用 HTTPS POST 呼叫 Telegram 的 sendMessage API。
// 支援 HTML 格式（parse_mode=HTML）與自訂鍵盤（reply_markup）。
//
// 參數：
//   token    → Telegram Bot Token
//   chatid   → 目標對話的 Chat ID
//   text     → 要傳送的訊息內容（支援 HTML 標籤）
//   keyboard → （可選）JSON 格式的自訂鍵盤字串
//
// 回傳值：Telegram API 的 HTTP 回應主體（JSON 字串）
// ============================================================
String telegramSendMessage(String token, String chatid, String text, String keyboard = "") {
  text.replace("\\n", "%0A");  // 將換行符號轉為 URL 編碼，Telegram 才能正確顯示
  const char* myDomain = "api.telegram.org";
  String getAll="", getBody = "";
  String request = "parse_mode=HTML&chat_id="+chatid+"&text="+text;

  // 如果有自訂鍵盤，附加到請求參數中
  if (keyboard!="")
    request += "&reply_markup="+keyboard;

  WiFiSSLClient client;

  if (client.connect(myDomain, 443)) {
    // 傳送 HTTP POST 請求（使用 application/x-www-form-urlencoded 格式）
    client.println("POST /bot"+token+"/sendMessage HTTP/1.1");
    client.println("Host: " + String(myDomain));
    client.println("Content-Length: " + String(request.length()));
    client.println("Content-Type: application/x-www-form-urlencoded");
    client.println("Connection: close");
    client.println();
    client.print(request);

    int waitTime = 5000;              // 最多等待 5 秒
    unsigned long startTime = millis();
    bool state = false;               // 標記是否已略過 HTTP 標頭，開始讀取主體

    // 讀取 HTTP 回應：略過標頭，只記錄主體（Body）
    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available())  {
        char c = client.read();

        if (state)
          getBody += String(c);    // 正在讀取主體

        if (c == '\n')  {
          // 遇到空白行（HTTP 標頭結束標記）時開始讀取主體
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);

        startTime = millis();  // 每收到字元重置逾時
      }

      if (getBody.length()>0)
        break;  // 已收到主體資料，提前結束等待
    }
    client.stop();
  }
  else {
    getBody="Connected to api.telegram.org failed.";
  }

  return getBody;
}

// ============================================================
// 傳送文字訊息至 LINE Bot
// 使用 LINE Messaging API 的 push message 端點（HTTPS POST）。
//
// 參數：
//   token    → LINE Channel Access Token（Bearer 認證）
//   targetId → 目標使用者或群組的 Line ID
//   message  → 要傳送的純文字訊息
//
// 回傳值：LINE API 的 HTTP 回應主體
// ============================================================
String lineSendMessage(String token, String targetId, String message) {
  String getAll="", getBody="";

  // 組合 LINE Messaging API 的 JSON 請求主體
  String request = "{\"to\":\""+targetId+"\",\"messages\":[{\"type\":\"text\",\"text\":\""+message+"\"}]}";

  const char* myDomain = "api.line.me";
  WiFiSSLClient client;

  if (client.connect(myDomain, 443)) {
    // 傳送 HTTP POST 至 LINE 推播訊息端點
    client.println("POST /v2/bot/message/push HTTP/1.1");
    client.println("Connection: close");
    client.println("Host: api.line.me");
    client.println("Authorization: Bearer " + token);  // Bearer Token 認證
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();
    client.println(request);
    client.println();

    boolean state = false;
    long startTime = millis();

    // 讀取 HTTP 回應主體（略過標頭）
    while ((startTime + 3000) > millis()) {
      while (client.available()) {
        char c = client.read();
        if (c == '\n') {
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);
        if (state==true)
          getBody += String(c);
        startTime = millis();
      }
      if (getBody.length()!= 0)
        break;
    }
    client.stop();
  }
  else {
    getBody="Connected to api.line.me failed.";
  }

  return getBody;
}

// ============================================================
// 捕獲相機影像並透過 Telegram 傳送 JPEG 照片
// 使用 multipart/form-data 格式上傳到 Telegram sendPhoto API。
//
// 重要安全性說明：
//   整個「取得影像 → 讀取緩衝區 → 傳送」的序列在持有 imageMutex 期間完成，
//   防止另一個任務在傳送過程中呼叫 Camera.getImage() 覆蓋緩衝區。
//
// 參數：
//   token   → Telegram Bot Token
//   chat_id → 目標 Chat ID
//   frames  → true = 拍攝新照片；false = 使用上一張快照（需確認緩衝區非空）
//
// 回傳值：Telegram API 的 HTTP 回應主體
// ============================================================
String telegramSendCapturedImage(String token, String chat_id, bool frames) {
  const char* myDomain = "api.telegram.org";
  String getAll="", getBody = "";
  WiFiSSLClient client;

  // 嘗試取得影像互斥鎖（最多等待 MUTEX_TIMEOUT_TICKS）
  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    return "Image buffer busy, please try again.";
  }

  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);  // 拍攝新的影像到緩衝區
    else if (!frames && imageLength == 0) {
      // 要求使用舊快照但緩衝區是空的
      client.stop();
      xSemaphoreGive(imageMutex);
      return "Previous image does not exist";
    }

    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    // 組合 multipart/form-data 標頭（boundary = "Taiwan"）
    String head =
      "--Taiwan\r\nContent-Disposition: form-data; name=\"chat_id\"; \r\n\r\n"
      + chat_id +
      "\r\n--Taiwan\r\nContent-Disposition: form-data; name=\"photo\"; filename=\"capture.jpg\"\r\nContent-Type: image/jpeg\r\n\r\n";

    String tail = "\r\n--Taiwan--\r\n";  // multipart 結束標記

    // 計算完整請求的總長度（標頭 + 影像資料 + 尾部）
    size_t imageLen = imageLength;
    size_t extraLen = head.length() + tail.length();
    size_t totalLen = imageLen + extraLen;

    // 傳送 HTTP 標頭
    client.println("POST /bot"+token+"/sendPhoto HTTP/1.1");
    client.println("Host: " + String(myDomain));
    client.println("Content-Length: " + String(totalLen));
    client.println("Content-Type: multipart/form-data; boundary=Taiwan");
    client.println();

    client.print(head);

    // 以 1024 位元組為單位分批傳送 JPEG 影像資料，避免 SSL 緩衝區溢位
    for (size_t n=0; n<fbLen; n=n+1024) {
      if (n+1024<fbLen) {
        client.write(fbBuf, 1024);
        fbBuf += 1024;
      }
      else {
        size_t remainder = fbLen - n;
        if (remainder > 0)
          client.write(fbBuf, remainder);
      }
    }

    client.print(tail);  // 傳送 multipart 結束標記

    int waitTime = 10000;             // 影像上傳最多等待 10 秒
    unsigned long startTime = millis();
    bool state = false;

    // 讀取 Telegram API 回應（略過標頭，只記錄主體）
    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (state)
          getBody += String(c);

        if (c == '\n') {
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);

        startTime = millis();
      }

      if (getBody.length()>0)
        break;
    }

    client.stop();
    Serial.println();

  } else {
    getBody="Connected to api.telegram.org failed.";
    Serial.println("Connected to api.telegram.org failed.");
  }

  xSemaphoreGive(imageMutex);  // 釋放影像互斥鎖

  return getBody;
}

// ============================================================
// 清理回應文字中的時間戳記、workId 與任務標籤
// Gemini 回應中可能殘留輸入時附加的時間戳記與 workId，
// 此函式負責移除這些內部標記，只留下給使用者看的純文字。
//
// 特殊情況：
//   - 若輸入為空或以 "NONE" 開頭，直接回傳 "NONE"
//   - 若遇到任何任務標籤（如 <BOT>、<PAGE>），截斷標籤之後的所有內容
//
// 回傳值：清理後的純文字，或 "NONE"
// ============================================================
String removeTimestamps(String workId, String timestamps, String text) {

    if (text.length() == 0 || text.startsWith("NONE")) return "NONE";

    // 如果文字中完全沒有角括號，則沒有需要處理的標記，直接回傳
    if (text.indexOf("&lt;") == -1 && text.indexOf("<") == -1) return text;

    // 還原 HTML 轉義的角括號（Telegram 傳來的文字可能已 HTML 編碼）
    text.replace("&lt;", "<");
    text.replace("&gt;", ">");

    // 移除時間戳記與工作 ID 字串
    text.replace(timestamps, "");
    text.replace(workId, "");

    // 找到任意任務標籤的出現位置，截斷標籤後的所有內容
    // 防止不同工作流程的回應互相污染
    for (const auto& tag : taskTags) {
        int pos = text.indexOf(tag);
        if (pos != -1)
            text = text.substring(0, pos);
    }

    return text;
}

// ============================================================
// 根據 workId 判斷回應的目標並傳送訊息
// 若 workId 以 <PAGE> 開頭，則回應給網頁介面（mainPageHTML）；
// 否則透過 Telegram 傳送文字訊息。
//
// 參數：
//   workId   → 工作 ID（決定回應管道）
//   text     → 要傳送的文字
//   keyboard → （可選）Telegram 自訂鍵盤 JSON 字串
// ============================================================
void replyUserMessage(String workId, String text, String keyboard = "") {
  if (text.length() == 0 || text.startsWith("NONE")) return;

  if (workId.startsWith(String(taskTags[0])))
    mainPageHTML += text +"\n";          // 附加到網頁回應緩衝區
  else
    telegramSendMessage(telegrambotToken, telegrambotChatId, text, keyboard);
}

// ============================================================
// 拍照並將影像回傳給使用者
// 根據 workId 決定輸出方式：
//   - 若為網頁介面請求（<PAGE>）：將影像轉為 base64 data URI，
//     嵌入到 mainPageHTML 中作為 <img> 標籤顯示。
//   - 若為 Telegram 請求：直接上傳 JPEG 到 Telegram Bot。
//
// 注意：imageMutex 在整個「取得影像 → base64 編碼 → 發布」過程中持有，
// 防止另一個任務同時呼叫 Camera.getImage() 覆蓋緩衝區。
//
// 參數：
//   workId → 工作 ID（決定回應管道）
//   frames → true = 拍攝新照片；false = 使用現有緩衝區
// ============================================================
String replyUserImage(String workId, bool frames) {
  if (workId.startsWith(String(taskTags[0]))) {

      // 取得影像互斥鎖
      if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
        return "Image buffer busy, please try again.";
      }

      if (frames)
          Camera.getImage(0, &imageAddress, &imageLength);  // 拍新照片

      if (imageLength == 0) {
        xSemaphoreGive(imageMutex);
        return "Previous image does not exist";
      }

      uint8_t* fbBuf = (uint8_t*)imageAddress;
      size_t   fbLen = imageLength;

      char *input = (char *)fbBuf;
      char output[base64_enc_len(3)];  // 每次 base64 編碼 3 個位元組的輸出緩衝區

      // 預估 base64 字串長度並預先保留記憶體（約為原始大小的 4/3）
      size_t estimatedSize = 23 + ((fbLen + 2) / 3) * 4 + 1;
      String imageFile = "<img src='data:image/jpeg;base64,";
      imageFile.reserve(estimatedSize);

      // 以每次 3 位元組的方式迴圈進行 base64 編碼
      // base64_encode() 函式內部會自動處理末尾不足 3 位元組的填充（padding）
      for (size_t i = 0; i < fbLen; i += 3) {
          base64_encode(output, input, 3);
          input += 3;
          imageFile += String(output);
      }

      // 將完整的 <img> HTML 標籤儲存到 mainPageHTML（持有 stateMutex）
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        mainPageHTML = imageFile + "' style='max-width:240px; height:auto; border-radius:8px;'><br>";
        xSemaphoreGive(stateMutex);
      }

      xSemaphoreGive(imageMutex);  // 釋放影像互斥鎖

      return "Image file created.";
  }
  else
    // Telegram 請求：直接上傳 JPEG 影像
    return telegramSendCapturedImage(telegrambotToken, telegrambotChatId, frames);

  return "";
}

// ============================================================
// JSON 字串跳脫處理函式（escapeForJson）
// 在將任意文字嵌入 JSON 字串值之前，必須先跳脫特殊字元：
//   第一步：將所有裸露的 " 轉為 \"
//   第二步：將連續的 \\ 折疊為單一 \（防止雙重跳脫）
//
// 使用場景：buildGeminiMessage() 在組合對話 JSON 前調用此函式，
// 確保訊息內容中的特殊字元不會破壞 JSON 結構。
// ============================================================
String escapeForJson(const String &src) {
  unsigned int n = src.length();

  // 第一步：將所有未跳脫的雙引號 " 改為 \"
  String s1;
  s1.reserve(n + (n >> 3) + 16);
  for (unsigned int i = 0; i < n; i++) {
    char c = src.charAt(i);
    if (c == '"') s1 += '\\';
    s1 += c;
  }

  // 第二步：將連續的 \\ 折疊為單一 \
  // 防止第一步產生的 \" 再被誤處理為 \\"
  String s2;
  s2.reserve(s1.length());
  unsigned int m = s1.length();
  for (unsigned int i = 0; i < m; ) {
    if (s1.charAt(i) == '\\' && i + 1 < m && s1.charAt(i + 1) == '\\') {
      s2 += '\\';
      i += 2;
    } else {
      s2 += s1.charAt(i);
      i += 1;
    }
  }
  return s2;
}

// ============================================================
// 建構 Gemini API 相容的對話訊息 JSON 物件
// Gemini API 的 contents 陣列格式為：
//   {"role": "user", "parts":[{"text": "..."}]}
//
// 參數：
//   role    → "user" 或 "model"
//   message → 訊息內容（會自動進行 JSON 跳脫）
//   comma   → true = 在前面加逗號（陣列中間項目）；false = 不加（第一項）
//
// 此函式是整個對話歷史記憶系統的核心：
// historicalMessages 就是由一連串此函式的輸出串接而成。
// ============================================================
String buildGeminiMessage(const String &role, const String &message, bool comma = true) {
  String esc = escapeForJson(message);  // 先跳脫特殊字元
  String jsonMessage;
  jsonMessage.reserve(esc.length() + role.length() + 48);
  jsonMessage  = comma ? ", {\"role\": \"" : "{\"role\": \"";
  jsonMessage += role;
  jsonMessage += "\", \"parts\":[{ \"text\": \"";
  jsonMessage += esc;
  jsonMessage += "\" }]}";
  return jsonMessage;
}

// ============================================================
// 從 SD 卡讀取文字檔案內容
// 使用 sdMutex 序列化所有 SD 卡存取，確保多工環境下的安全性。
//
// 流程：
//   1. 取得 sdMutex
//   2. 掛載 FAT 檔案系統（fs.begin()）
//   3. 組合完整檔案路徑並開啟檔案
//   4. 配置記憶體緩衝區，讀取所有內容
//   5. 關閉檔案，解除掛載，釋放互斥鎖
//
// 回傳值：檔案內容字串；若失敗則回傳空字串 ""
// ============================================================
String getStringFromFile(String fileNname) {
  String data = "";

  // 嘗試取得 SD 卡互斥鎖
  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] getStringFromFile: failed to take sdMutex");
    return "";
  }

  if (!fs.begin()) {
    Serial.println("Card Mount Failed");
    if (sdMutex) xSemaphoreGive(sdMutex);
    return "";
  }

  // 組合完整路徑（getRootPath() 回傳 SD 卡根目錄，例如 "/SD"）
  String path = String(fs.getRootPath()) + "/" + fileNname;

  file = fs.open(path);

  if (file) {
    uint32_t len = file.size();
    char *buf = (char*)malloc(len + 1);  // 動態配置讀取緩衝區

    if (buf) {
      file.read(buf, len);
      buf[len] = '\0';  // 確保字串以 null 結尾
      data = String(buf);
      free(buf);        // 立即釋放臨時緩衝區
    }

    file.close();
  }

  fs.end();  // 解除掛載 FAT 檔案系統

  if (sdMutex) xSemaphoreGive(sdMutex);

  return data;
}

// ============================================================
// 將資料儲存到 SD 卡上的文字檔案
// 在寫入新檔案之前，會先將現有檔案改名為 .bak 備份。
// 若 timestamp 參數為 true，備份檔名包含當前時間戳記（例如：memory.md_20260627123045.bak）。
//
// 使用 sdMutex 序列化存取，確保多工安全。
// 注意：fs.open() 在此用於建立並寫入新檔案（AmebaFatFS 的開啟模式）。
// ============================================================
void storeDataToFile(String filename, String data, bool timestamp = false) {

  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] storeDataToFile: failed to take sdMutex");
    return;
  }

  if (!fs.begin()) {
    Serial.println("Card Mount Failed");
    if (sdMutex) xSemaphoreGive(sdMutex);
    return;
  }

  String file_path = String(fs.getRootPath());
  String currentFile = file_path + "/" + filename;

  // 建立備份檔案路徑（一般 .bak 或帶時間戳記的 .bak）
  String backupFile = currentFile + ".bak";
  if (timestamp == true)
    backupFile = currentFile + "_" + getRtcTimeString(true) + ".bak";

  // 如果目標檔案已存在，先備份再刪除
  if (fs.exists(currentFile)) {

    if (fs.exists(backupFile)) {
      fs.remove(backupFile);  // 刪除舊備份
    }
    delay(100);  // 等待檔案系統操作完成

    fs.rename(currentFile, backupFile);  // 將現有檔案改名為備份
  }

  // 建立新檔案並寫入資料
  file = fs.open(currentFile);

  if (file) {
    file.println(data.c_str());
    file.close();
  }
  else
    Serial.println("[DEBUG] File open failed: " + currentFile);

  fs.end();

  if (sdMutex) xSemaphoreGive(sdMutex);
}

// ============================================================
// 透過 TCP 將訊息傳送給另一台裝置或代理（fuClaw 裝置間通訊）
// 連線到目標裝置的 HTTP 埠 81（fuClaw 管理介面），
// 以 GET /message?<訊息> 的格式傳遞請求。
//
// 參數：
//   workId  → 工作 ID（用於回應中的識別）
//   domain  → 目標裝置的 IP 位址或主機名稱
//   request → 要傳送的訊息內容（將進行 URL 編碼）
//
// 回傳值：包含 status 與 response 的 JSON 字串
// ============================================================
String tcpSendMessage(String workId, String domain, String request) {

  WiFiClient client;  // 使用一般 TCP 客戶端（無 SSL，裝置內網通訊）

  if (client.connect(domain.c_str(), 81)) {

    client.setTimeout(20000);  // 設定 TCP 連線逾時為 20 秒

    // 傳送 HTTP GET 請求，訊息內容進行 URL 編碼附在查詢字串中
    client.println("GET /message?" + urlencode(request) + " HTTP/1.1");
    client.println("Host: " + domain);
    client.println("Access-Control-Allow-Origin: *");
    client.println("Content-Length: 0");
    client.println();

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    // 略過 HTTP 標頭，讀取回應主體
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;  // 空白行代表標頭結束
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;  // 每收到資料就重置逾時
        }
      }
      vTaskDelay(1);
    }

    client.stop();

    return
      "{\"status\":\"success\","
      "\"method\":\"/tcpSendMessage\","
      "\"response\":\"" + body + "\","
      "\"workId\":\"" + workId + "\"}";
  }

  return
    "{\"status\":\"error\","
    "\"method\":\"/tcpSendMessage\","
    "\"reason\":\"Connected to the device failed.\","
    "\"workId\":\"" + workId + "\"}";
}

// ============================================================
// 重置 Gemini 對話狀態
// 清除所有對話歷史與工具執行紀錄，讓助理從全新狀態開始對話。
// 同時備份現有 memory.md（帶時間戳記），防止資料意外遺失。
//
// 需持有 stateMutex 才能修改共享字串狀態。
// ============================================================
void geminiChatReset() {

  storeDataToFile(memoryFilename, "", true);  // 備份現有記憶（帶時間戳記）

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages = "";    // 清除對話歷史
    executeToolHistory = "";    // 清除工具執行紀錄

    // 重建三個系統提示詞預組裝字串
    systemContent = buildGeminiMessage("user", geminiRole, false) + buildGeminiMessage("model", "OK");
    systemContentTools = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule + skillsDefinition + toolsDefinition, false) + buildGeminiMessage("model", "OK");
    systemContentNoTools = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule, false) + buildGeminiMessage("model", "OK");
    xSemaphoreGive(stateMutex);
  }
}

    xSemaphoreGive(stateMutex);
  }
}

// ============================================================
// 向 Gemini API 發送對話請求並取得回應文字
// 這是整個系統最核心的 API 呼叫函式，所有其他工具最終都透過此函式與 Gemini 溝通。
//
// 工具模式（tools 參數）：
//   1  → 使用完整工具定義（systemContentTools + 歷史記錄）→ 硬體控制模式
//   0  → 不含工具定義（systemContentNoTools + 歷史記錄）→ 對話模式（無工具）
//  -1  → 無歷史記錄模式（systemContent + 當前訊息）→ 獨立請求（如 RTC 同步）
//  其他→ 同 -1
//
// 流程：
//   1. 將時間戳記和 workId 附加到訊息中（讓 Gemini 知道目前時間）
//   2. 將使用者訊息存入 historicalMessages
//   3. 根據 tools 參數選擇系統提示詞
//   4. 組合 HTTP POST 請求發送至 Gemini API
//   5. 解析回應 JSON，提取 text 內容
//   6. 清理時間戳記，將 Gemini 回應存入 historicalMessages
//   7. 回傳純文字回應
// ============================================================
String geminiChatRequest(String workId, String message, int tools = 1) {
  String timestamps = "\n" + workId;  // 時間戳記字串（附在訊息末尾）

  // 附加 RTC 時間到訊息，讓 Gemini 知道目前裝置時間（用於排程判斷等）
  message = message + "\n\nRTC current time: " + getRtcTimeString();

  // 將含時間戳記的使用者訊息存入對話歷史（持有 stateMutex）
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  // 根據工具模式選擇對應的系統提示詞前綴
  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    if (tools == 0)
      contents = systemContentNoTools + historicalMessages;       // 對話模式（無工具）
    else if (tools == 1)
      contents = systemContentTools + historicalMessages;         // 硬體控制模式（含工具）
    else if (tools == 2)
      contents = systemContent + buildGeminiMessage("user", message);  // 獨立請求模式
    else
      contents = systemContent + buildGeminiMessage("user", message);  // 同上（tools = -1 等）
    xSemaphoreGive(stateMutex);
  }

  // 組合完整的 Gemini API 請求 JSON 主體
  String request = "{\"contents\": [" + contents +
                   "],\"generationConfig\": {\"maxOutputTokens\": " +
                   geminiMaxOutputTokens +
                   ", \"temperature\": " + geminiTemperature + "}}";

  WiFiSSLClient client;
  String responseText = "";

  client.setTimeout(10000);  // 連線逾時 10 秒

  if (client.connect("generativelanguage.googleapis.com", 443)) {

    // 傳送 HTTP POST 請求至 Gemini generateContent 端點（HTTP/1.0 避免 chunked 編碼）
    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    // 分批傳送請求主體（每次 1024 位元組）
    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;  // 等待回應最多 20 秒
    bool headersEnded = false;
    String line = "";

    // 讀取 HTTP 回應：略過標頭，收集主體
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;  // 遇到空白行，標頭結束
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        }
        else {
          body += c;
          timeout = millis() + 20000;  // 每收到資料重置逾時（防止大型回應超時）
        }
      }
      // [看門狗修復] 讓出 CPU 防止 WDT（看門狗計時器）在長時間 Gemini 回應期間觸發
      vTaskDelay(1);
    }

    client.stop();

    // 找到 JSON 的起始位置（跳過任何前置非 JSON 文字）
    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }

    // 解析 Gemini API 回應 JSON
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (geminiChatRequest)\n" + body);
      responseText = "JSON parse failed (geminiChatRequest). Please try again.";
    }
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      // 正常情況：提取 Gemini 的文字回應
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    }
    else if (doc["error"]) {
      // Gemini API 回傳錯誤（例如 API Key 無效、配額超限等）
      responseText = "[DEBUG] Gemini API Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Gemini API Error";
    }
    else {
      responseText = "Unexpected response from Gemini.";
      Serial.println("Unknown response format.");
    }

  } else {
    Serial.println("Failed to connect to Gemini API");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "Gemini did not respond. Please try again.";
  }

  // 移除回應中殘留的時間戳記與工作 ID
  responseText = removeTimestamps(workId, timestamps, responseText);

  // 將 Gemini 回應存入對話歷史記憶
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// ============================================================
// 向 Gemini API 發送「接地式搜尋」請求
// 與 geminiChatRequest 類似，但在請求中額外加入 Google Search 工具，
// 讓 Gemini 可以即時搜尋網路，回答需要最新資訊的問題。
//
// 請求格式差異：
//   普通請求：{"contents": [...], "generationConfig": {...}}
//   搜尋請求：{"contents": [...], "tools": [{"google_search": {}}], "generationConfig": {...}}
//
// 使用時機：使用者詢問即時新聞、最新股價、天氣等需要網路搜尋的問題。
// ============================================================
String geminiSearchRequest(String workId, String message, int tools = 1) {
  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  // 將使用者訊息存入對話歷史
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  // 根據工具模式選擇系統提示詞
  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    contents = systemContent + buildGeminiMessage("user", message);  // 預設：無歷史記錄
    if (tools == 1)
      contents = systemContentTools + historicalMessages;
    else if (tools == 0)
      contents = systemContentNoTools + historicalMessages;
    xSemaphoreGive(stateMutex);
  }

  // 關鍵差異：加入 "tools": [{"google_search": {}}] 啟用接地式搜尋
  String request = "{\"contents\": [" + contents +
                   "],\"tools\": [{\"google_search\": {}}],\"generationConfig\": {\"maxOutputTokens\": " +
                   geminiMaxOutputTokens +
                   ", \"temperature\": " + geminiTemperature + "}}";

  WiFiSSLClient client;
  String responseText = "";

  client.setTimeout(10000);

  if (client.connect("generativelanguage.googleapis.com", 443)) {

    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      // [看門狗修復] 讓出 CPU 防止搜尋等待期間 WDT 觸發
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }

    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (geminiChatRequest)\n" + body);
      responseText = "JSON parse failed (geminiChatRequest). Please try again.";
    }
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    }
    else if (doc["error"]) {
      responseText = "[DEBUG] Gemini API (Search) Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Gemini API (Search) Error";
    }
    else {
      responseText = "Unexpected response from Gemini Search.";
    }

  } else {
    Serial.println("Failed to connect to Gemini API (Search)");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "Gemini Search did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// ============================================================
// 捕獲相機影像並傳送給 Gemini Vision 進行多模態分析
// 流程：
//   1. 持有 imageMutex 防止並發影像緩衝區競爭
//   2. 拍照（或使用現有快照）
//   3. 將 JPEG 影像進行 base64 編碼
//   4. 將 base64 影像與問題文字組合成 Gemini multimodal 請求
//   5. 影像編碼完成後立即釋放 imageMutex（網路 round-trip 期間不持鎖）
//   6. 傳送請求，解析回應，存入對話歷史
//
// 注意：imageMutex 只在取得影像與編碼期間持有，
// 釋放後才進行耗時的 HTTPS 傳輸，讓其他任務可以拍新照片。
//
// 參數：
//   workId  → 工作 ID
//   message → 傳給 Gemini 的視覺分析問題（例如「描述圖片中的場景」）
//   frames  → true = 拍新照片；false = 使用現有緩衝區
// ============================================================
String geminiVisionRequest(String workId, String message, bool frames = true) {
  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  // 將使用者的視覺分析請求存入對話歷史
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  WiFiSSLClient client;
  String responseText = "";
  const char* myDomain = "generativelanguage.googleapis.com";

  client.setTimeout(10000);

  // 取得影像互斥鎖（確保整個取像+編碼序列的原子性）
  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    responseText = "Image buffer busy, please try again.";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      historicalMessages += buildGeminiMessage("model", responseText + timestamps);
      xSemaphoreGive(stateMutex);
    }
    return responseText;
  }

  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);  // 拍攝新照片
    else if (!frames && imageLength == 0) {
      // 要求舊快照但緩衝區是空的
      client.stop();
      xSemaphoreGive(imageMutex);

      responseText = "Previous image does not exist";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("model", responseText + timestamps);
        xSemaphoreGive(stateMutex);
      }

      return responseText;
    }

    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    char *input = (char *)fbBuf;
    char output[base64_enc_len(3)];
    String imageFile = "";

    // 以每次 3 位元組的方式進行 base64 編碼（影像資料 → base64 字串）
    for (size_t i = 0; i < fbLen; i += 3) {
      base64_encode(output, input, 3);
      input += 3;
      imageFile += String(output);
    }

    // 組合 Gemini multimodal 請求 JSON
    // 包含文字問題（text part）和 base64 影像（inline_data part）
    String Data = "{\"contents\": [{\"parts\": [{\"text\": \"" + message +
                  "\"}, {\"inline_data\": {\"mime_type\":\"image/jpeg\",\"data\":\"" +
                  imageFile + "\"}}]}]}";

    // 影像已完整編碼到 Data 字串，可以安全釋放 imageMutex
    // 讓其他任務在長時間網路傳輸期間可以拍攝新的影像
    xSemaphoreGive(imageMutex);

    // 傳送 HTTP POST 請求（影像請求通常較大，需要更長的等待時間）
    client.println("POST /v1beta/models/"+geminiModel+":generateContent?key="+geminiApiKey+" HTTP/1.0");
    client.println("Host: " + String(myDomain));
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(Data.length()));
    client.println("Connection: close");
    client.println();

    for (size_t i = 0; i < Data.length(); i += 1024) {
      client.print(Data.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      // [看門狗修復] 讓出 CPU 防止 Vision 回應期間 WDT 觸發
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }

    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed (geminiSearchRequest):\n" + body);
      responseText = "JSON parse failed (geminiSearchRequest). Please try again.";
    }
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    }
    else if (doc["error"]) {
      responseText = "[DEBUG] Gemini API (Vision) Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Gemini API (Vision) Error";
    }
    else {
      responseText = "Unexpected response from Gemini Vision.";
    }

  } else {
    Serial.println("Failed to connect to Gemini API (Vision)");
    responseText = "Connection failed";
    xSemaphoreGive(imageMutex);  // 連線失敗時也需要釋放互斥鎖
  }

  if (responseText == "") {
    responseText = "Gemini Vision did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  // 將 Gemini Vision 分析結果存入對話歷史
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildGeminiMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// ============================================================
// GPIO 輸出控制工具函式（數位寫入 / 類比寫入）
// 支援以下兩種輸出模式：
//   digitalwrite → 數位輸出，value 只能是 0 或 1（低/高電位）
//   analogwrite  → 類比（PWM）輸出，value 範圍 0~255
//
// 在執行前會進行數值邊界驗證，無效數值一律回傳錯誤 JSON。
// 成功執行後回傳 JSON 成功狀態，供 Gemini 確認執行結果。
//
// 參數：
//   pin    → GPIO 腳位號碼（必須是已確認的合法腳位）
//   mode   → "digitalwrite" 或 "analogwrite"（不區分大小寫）
//   value  → 輸出數值
//   workId → 工作 ID（含在回傳 JSON 中以便追蹤）
// ============================================================
String toolPinOutput(int pin, String mode, int value, String workId) {

    pinMode(pin, OUTPUT);  // 設定腳位為輸出模式

    mode.toLowerCase();    // 統一轉為小寫，方便比對

    if (mode == "digitalwrite") {

        // 數位輸出只接受 0 或 1
        if (value != 0 && value != 1) {
            return
              "{\"status\":\"error\","
              "\"method\":\"/digitalwrite\","
              "\"reason\":\"invalid_digital_value\","
              "\"workId\":\"" + workId + "\"}";
        }

        digitalWrite(pin, value);  // 執行數位輸出

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalwrite\","
            "\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogwrite") {

        // PWM 輸出範圍 0~255
        if (value < 0 || value > 255) {
            return
              "{\"status\":\"error\","
              "\"method\":\"/analogwrite\","
              "\"reason\":\"invalid_analog_value\","
              "\"workId\":\"" + workId + "\"}";
        }

        analogWrite(pin, value);  // 執行 PWM 輸出

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogwrite\","
            "\"workId\":\"" + workId + "\"}";

    }

    // 未知的輸出模式
    return
        "{\"status\":\"error\","
        "\"method\":\"/analogwrite\","
        "\"reason\":\"invalid_output_mode\","
        "\"workId\":\"" + workId + "\"}";
}

// ============================================================
// GPIO 輸入讀取工具函式（數位讀取 / 類比讀取）
// 支援以下兩種輸入模式：
//   digitalread → 讀取數位輸入（回傳 0 或 1）
//   analogread  → 讀取類比輸入（回傳 0~1023）
//
// 回傳包含讀取值的 JSON 字串，供 Gemini 解析並決定後續動作。
//
// 參數：
//   pin    → GPIO 腳位號碼
//   mode   → "digitalread" 或 "analogread"（不區分大小寫）
//   workId → 工作 ID
// ============================================================
String toolPinInput(int pin, String mode, String workId) {

    pinMode(pin, INPUT);  // 設定腳位為輸入模式

    mode.toLowerCase();

    if (mode == "digitalread") {

        int value = digitalRead(pin);  // 讀取數位輸入值（0 或 1）

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalread\","
            "\"value\":" + String(value) + ","
            "\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogread") {

        int value = analogRead(pin);  // 讀取類比輸入值（0~1023）

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogread\","
            "\"value\":" + String(value) + ","
            "\"workId\":\"" + workId + "\"}";

    }

    return
        "{\"status\":\"error\","
        "\"method\":\"/analogread\","
        "\"reason\":\"invalid_input_mode\","
        "\"workId\":\"" + workId + "\"}";
}

// ============================================================
// 伺服馬達（SG90）角度控制工具函式
// 透過 AmebaServo 函式庫控制 SG90 伺服馬達的旋轉角度。
// 使用場景：控制窗戶開合角度、機械臂等。
//
// 注意：servos[] 陣列以腳位號碼為索引，首次使用時自動 attach。
// 角度範圍嚴格限制在 0~180 度，超出範圍一律回傳錯誤。
//
// 參數：
//   servo  → 對應 GPIO 腳位的 AmebaServo 物件（來自 servos[pin]）
//   pin    → GPIO 腳位號碼
//   angle  → 目標角度（0~180 度）
//   workId → 工作 ID
// ============================================================
String tool_servo(AmebaServo &servo, int pin, int angle, String workId) {
    if (!servo.attached())
        servo.attach(pin);  // 若尚未 attach，先連結到指定腳位

    if (angle < 0 || angle > 180) {
        return
          "{\"status\":\"error\","
          "\"method\":\"/servo\","
          "\"reason\":\"invalid_servo_angle\","
          "\"workId\":\"" + workId + "\"}";
    }

    servo.write(angle);  // 設定伺服馬達目標角度

    return
        "{\"status\":\"success\","
        "\"method\":\"/servo\","
        "\"workId\":\"" + workId + "\"}";
}

// ============================================================
// DHT11 溫濕度感測器讀取工具函式
// 每次呼叫時建立 DHT 實例並初始化，讀取溫度（°C）與相對濕度（%）。
// 若讀取失敗（isnan 檢查）則回傳錯誤 JSON，讓 Gemini 知道感測器異常。
//
// 注意：DHT11 讀取速度較慢，每次初始化需要一點時間穩定。
//
// 參數：
//   pin    → DHT11 資料腳位（AMB82-mini: GPIO 8 / HUB 8735: GPIO 20）
//   workId → 工作 ID
//
// 回傳值：包含 temperature 和 humidity 的 JSON 字串
// ============================================================
String tool_dht11(int pin, String workId) {
  DHT dht(pin, DHT11);  // 建立 DHT11 感測器實例
  dht.begin();          // 初始化感測器
  float h = dht.readHumidity();     // 讀取相對濕度（%）
  float t = dht.readTemperature();  // 讀取溫度（攝氏）

  // 若任一數值為 NaN（讀取失敗），回傳錯誤
  if (isnan(h) || isnan(t)) {
    return "{\"status\":\"error\","
           "\"method\":\"/dht11\","
           "\"reason\":\"dht11_read_failed\","
           "\"workId\":\"" + workId + "\"}";
  }

  return
    "{\"status\":\"success\","
    "\"method\":\"/dht11\","
    "\"temperature\":" + String(t) + ","
    "\"humidity\":"    + String(h) + ","
    "\"workId\":\"" + workId + "\"}";
}

// ============================================================
// 工作流程完成性評估（Workflow Continuation Check）
// 每次工具執行完畢後，詢問 Gemini 工作流程是否已完成。
// 若 Gemini 認為還需要更多工具操作，它會回傳下一步的 tool_call JSON，
// 由 handleAgentResponse() 繼續執行；若已完成，回傳 "NONE" 結束鏈。
//
// 參數：
//   workId   → 工作 ID
//   reCheck  → 是否需要繼續評估（false = 跳過此函式）
//   task     → （可選）原始使用者任務描述，讓 Gemini 有更多上下文判斷
//
// 這是實現「多步驟工作流程」的核心機制：
// Gemini 以迭代方式確認每個步驟後的狀態，直到任務完成。
// ============================================================
void evaluateWorkflowContinuation(String workId, bool reCheck, String task = "") {

    if (!reCheck) return;  // 不需要繼續評估時直接返回

    // 建立提示詞，請 Gemini 判斷工作流程是否完成
    String prompt =
        "Analyze the execution result and determine whether the workflow is complete.\n";

    if (task != "") {
        prompt += "User task request:\n" + task + "\n\n";
    }

    prompt +=
        "If additional hardware actions are strictly required, "
        "return ONLY a valid tool_call JSON.\n"
        "If the workflow is already complete, return EXACTLY: NONE.\n"
        "If no tool action is required and a user-facing reply is needed, "
        "respond naturally in the user's language.\n"
        "Avoid repeating the same meaning as your immediately previous response during the same workflow. If a new workflow or task begins, normal responses are allowed even if similar to previous ones.\n"
        "Do not include explanation or extra text.";

    // 傳送評估請求並處理 Gemini 回應（可能觸發更多工具呼叫）
    handleAgentResponse(workId, geminiChatRequest(workId, prompt));
}

// ============================================================
// 工具指令派發器（Tool Command Dispatcher）
// 這是整個系統的「執行引擎」，負責將解析出的工具指令路由到對應的實作函式。
// Gemini 的 tool_call JSON 由 handleAgentResponse() 解析後傳入此函式。
//
// 執行原子性原則：
//   每次只執行一個硬體動作（一個腳位、一種操作、一個數值）。
//   多步驟工作流程透過 evaluateWorkflowContinuation() 逐步完成。
//
// 安全機制：
//   - 所有對話歷史更新均持有 stateMutex
//   - 硬體動作前均做參數驗證
//   - 未知指令回退到 Gemini 對話模式
//
// 參數：
//   workId  → 工作 ID（標識此次任務來源與時間）
//   command → 指令名稱（例如 "/digitalwrite"、"/servo"、"/chat" 等）
//   params  → 指令參數 JSON 物件（從 tool_call JSON 解析）
//   reCheck → 是否在執行後呼叫 evaluateWorkflowContinuation()
// ============================================================
void executeTool(String workId, String command, JsonObject params, bool reCheck = true) {
    String timestamps = "\n" + workId;

    // ---------- GPIO 數位/類比輸出 ----------
    if (command == "/digitalwrite" || command == "/analogwrite") {
      int pin = params["pin"].as<int>();
      String pinmode = params["pinmode"].as<String>();
      int value = params["value"].as<int>();

      String response = toolPinOutput(pin, pinmode, value, workId);

      // 記錄到對話歷史與執行日誌
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+String(pin)+" | "+pinmode+" | "+String(value)+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);  // 評估工作流程是否完成

    }
    // ---------- GPIO 數位/類比輸入讀取 ----------
    else if (command == "/digitalread" || command == "/analogread") {
      int pin = params["pin"].as<int>();
      String pinmode = params["pinmode"].as<String>();

      String response = toolPinInput(pin, pinmode, workId);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+String(pin)+" | "+pinmode+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 相機拍照並傳送給使用者 ----------
    else if (command == "/still") {
      bool frames = params.containsKey("frames") ? params["frames"].as<bool>() : true;
      String task = params.containsKey("task") ? params["task"].as<String>() : "NONE";
      String res = replyUserImage(workId, frames);  // 拍照並傳送（依 workId 決定管道）

      // 跳脫特殊字元，防止嵌入 JSON 時破壞格式
      res.replace("\\", "\\\\");
      res.replace("\"", "\\\"");

      String response =
        "{\"method\":\"/still\","
        "\"result\":\"" + res + "\","
        "\"workId\":\"" + workId + "\"}";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+frames+" | "+task+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck, task);

    }
    // ---------- RTC 時間同步 ----------
    else if (command == "/syncrtc") {
      rtcInitialTime(workId);  // 向 Gemini 請求目前時間並更新硬體 RTC
      String rtcTimeResponse = "RTC START: " + getRtcTimeString();
      replyUserMessage(workId, rtcTimeResponse);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", rtcTimeResponse + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 取得 RTC 目前時間 ----------
    else if (command == "/getrtc") {
      String rtcTime = getRtcTimeString();
      replyUserMessage(workId, rtcTime);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", rtcTime + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 新增排程任務 ----------
    else if (command == "/schedule") {
      String task = params["task"].as<String>();

      String response = "";
      String localSchedule = "";

      // 讀取目前的排程任務清單（持有 stateMutex 確保執行緒安全）
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      // 驗證任務參數是否為合法 JSON 陣列格式（以 [ 開頭、] 結尾）
      if (task.startsWith("[") && task.indexOf("]") !=-1) {
          task = task.substring(0, task.lastIndexOf("]") + 1);

          if (localSchedule == "") {
            localSchedule = task;  // 首次新增，直接使用
          } else {
            // 已有排程：使用 Gemini 合併兩個 JSON 陣列
            // 避免手動合併可能導致的 JSON 格式問題
            localSchedule += ", " + task;

            String prompt =
              "Merge all given JSON arrays into a single valid JSON array. "
              "Output ONLY the merged array. "
              "Ensure the result is valid JSON starting with [ and ending with ]. "
              "For every object in the arrays, keep all fields unchanged. "
              "The value of the task field MUST remain exactly as provided. "
              "Never translate, rewrite, summarize, localize, or modify task descriptions. "
              "Task descriptions MUST remain in the original user language.\n\n"
              + localSchedule;

            String jsonArray = geminiChatRequest(workId, prompt, -1);

            if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
              jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);
              localSchedule = jsonArray;
            }
          }

          // 更新共享排程狀態並寫入 SD 卡
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            scheduleTasks = localSchedule;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(scheduleFilename, localSchedule);  // 持久化到 SD 卡

          response =
            "{\"status\":\"success\","
            "\"method\":\"/schedule\","
            "\"workId\":\""+workId+"\"}";
      }
      else {
        // 任務格式不合法
        response =
          "{\"status\":\"error\","
          "\"method\":\"/schedule\","
          "\"reason\":\"Invalid JSON array format.\","
          "\"workId\":\""+workId+"\"}";
      }

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 修改或刪除排程任務 ----------
    else if (command == "/modifySchedule") {
      String task = params["task"].as<String>();

      String response = "";
      String localSchedule = "";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      // 建立提示詞，請 Gemini 根據使用者要求修改排程任務 JSON
      // 規則詳細說明了如何修改、刪除，以及不得更動未受影響的任務
      String prompt =
          "You are given a JSON array of scheduled tasks and a user-approved schedule modification request. "
          "Apply the requested modification or deletion to the scheduled tasks. "
          "Rules: "
          "- Match tasks using both schedule time and task description. "
          "- If the request is to modify a task, update only the requested fields. "
          "- Modifying a task includes changing the task description, schedule time, or recurrence settings. "
          "- Any modified task MUST have its executed field set to false. "
          "- If the request is to delete a task, remove the matching task from the array. "
          "- Do NOT modify unrelated tasks. "
          "- Preserve all fields of unaffected tasks. "
          "- Preserve the executed field of unaffected tasks. "
          "- Do NOT add new fields. "
          "- Do NOT remove existing fields except when deleting a task. "
          "- Preserve the original JSON schema. "
          "- If no matching task exists, return the original array unchanged. "
          "- Output ONLY the updated JSON array. "
          "- The result MUST start with [ and end with ]. "
          "- Do NOT output explanations, markdown, code fences, or natural language.\n\n"
          "Current scheduled tasks:\n" +
          localSchedule +
          "\n\nUser-approved modification request:\n" +
          task;

      String jsonArray = geminiChatRequest(workId, prompt);

      if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
        jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);

        if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
          scheduleTasks = jsonArray;
          xSemaphoreGive(stateMutex);
        }
        storeDataToFile(scheduleFilename, jsonArray);

        response =
          "{\"status\":\"success\","
          "\"method\":\"/modifySchedule\","
          "\"workId\":\""+workId+"\"}";
      }
      else {
        response =
          "{\"status\":\"error\","
          "\"method\":\"/modifySchedule\","
          "\"reason\":\"Invalid JSON array format.\","
          "\"workId\":\""+workId+"\"}";
      }

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 更新排程任務的執行狀態 ----------
    else if (command == "/updateScheduleStatus") {
      String response = "";
      String localSchedule = "";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      // 請 Gemini 根據工具執行歷史，更新排程任務的 "executed" 欄位
      // 重複性任務（year == 0）永遠不更改 executed 欄位
      // 一次性任務若在歷史中找到對應的成功執行記錄，才將 executed 設為 true
      String prompt =
          "You are given a JSON array of scheduled tasks and a tool execution history. "
          "For each task: "
          "- If the task's schedule has \"year\" equal to 0, it is a recurring task. Do NOT change its \"executed\" field. "
          "- Otherwise, set \"executed\" to true ONLY if the task's corresponding action appears in the execution history as successfully completed, otherwise Do NOT change its \"executed\" field. "
          "Output ONLY the updated JSON array. "
          "The result MUST start with [ and end with ]. "
          "Do NOT change any other fields.\n\n"
          + localSchedule;

      String jsonArray = geminiChatRequest(workId, prompt);

      if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
        jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);

        if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
          scheduleTasks = jsonArray;
          xSemaphoreGive(stateMutex);
        }
        storeDataToFile(scheduleFilename, jsonArray);

        response =
          "{\"status\":\"success\","
          "\"method\":\"/updateScheduleStatus\","
          "\"workId\":\""+workId+"\"}";
      }
      else {
        response =
          "{\"status\":\"error\","
          "\"method\":\"/updateScheduleStatus\","
          "\"reason\":\"Invalid JSON array format.\","
          "\"workId\":\""+workId+"\"}";
      }

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 取得所有排程任務（格式化後回傳給使用者） ----------
    else if (command == "/getSchedule") {
      String localSchedule = "";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      // 請 Gemini 將排程 JSON 整理成易讀的條列格式，並以使用者語言回覆
      String prompt =
        "Please organize the following scheduled tasks and respond in the user's current language. "
        "Present the information in a clear and well-structured bullet-point format for better readability: "
        + localSchedule;

      String response = geminiChatRequest(workId, prompt);
      replyUserMessage(workId, response);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 取得未完成的排程任務（純文字條列） ----------
    else if (command == "/getUnfinishedSchedule") {
      String localSchedule = "";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        // 確保 JSON 陣列格式完整（以 ] 結尾）
        if (scheduleTasks.startsWith("[") && scheduleTasks.indexOf("]") !=-1)
              scheduleTasks = scheduleTasks.substring(0, scheduleTasks.lastIndexOf("]") + 1);
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      // 篩選未完成任務並格式化為純文字
      String response = getUnfinishedScheduleTasksJson(localSchedule);
      replyUserMessage(workId, response);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 清除所有排程任務 ----------
    else if (command == "/clearSchedule") {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        scheduleTasks = "";        // 清除記憶體中的排程清單
        executedTodayTasks = "";   // 清除今日已執行記錄
        xSemaphoreGive(stateMutex);
      }

      // 清除 SD 卡上的排程資料
      storeDataToFile(scheduleFilename, "");
      storeDataToFile(scheduleExecutedTodayTasksFilename, "");

      String response = "Scheduled tasks have been cleared.";
      replyUserMessage(workId, response);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 重置對話（清除歷史記憶，重新開始） ----------
    else if (command == "/reset") {
      geminiChatReset();  // 清除對話歷史並重建系統提示詞

      String response = "New chat started.";
      replyUserMessage(workId, response);

    }
    // ---------- 取得系統記憶體使用資訊 ----------
    else if (command == "/getMemory") {
      String msg = getMemoryInfo();
      replyUserMessage(workId, msg);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", msg + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 顯示工具執行歷史紀錄 ----------
    else if (command == "/getLog") {
      String localHistory = "";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localHistory = executeToolHistory;
        executeToolHistory += workId + " " + command + "\n";  // 記錄此次 /getLog 本身
        xSemaphoreGive(stateMutex);
      }
      Serial.println("\n\nExecute tools history:\n\n"+localHistory+"\n\n");
      replyUserMessage(workId, "Please check the serial monitor to view the tool execution log.");

    }
    // ---------- 自然語言回覆（/chat 工具） ----------
    else if (command == "/chat") {
      String reply = params["reply"].as<String>();
      replyUserMessage(workId, reply);  // 直接傳回 Gemini 提供的回覆文字

    }
    // ---------- Gemini 網路搜尋 ----------
    else if (command == "/search") {
      String query = params["query"].as<String>();
      String task = params["task"].as<String>();

      // 使用接地式搜尋（附帶 Google Search 工具）
      String response = geminiSearchRequest(workId, query, false);
      handleAgentResponse(workId, response);  // 搜尋結果可能觸發更多工具

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        executeToolHistory += workId + " " + command + " [ "+query+" | "+task+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck, task);

    }
    // ---------- 延遲指定時間（毫秒） ----------
    else if (command == "/delay") {
      long milliseconds = params["milliseconds"].as<long>();
      milliseconds = constrain(milliseconds, 0, 10000);  // 最大延遲 10 秒

      unsigned long start = millis();

      // 使用 vTaskDelay 而非 delay()，讓 FreeRTOS 排程器可以切換其他任務
      while (millis() - start < milliseconds) {
          vTaskDelay(10 / portTICK_PERIOD_MS);
      }

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        executeToolHistory += workId + " " + command + " [ " + String(milliseconds) + " ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 相機拍照 + Gemini Vision 多模態分析 ----------
    else if (command == "/vision") {
      String query = params.containsKey("query") ? params["query"].as<String>() :
                     "Describe the image in detail in the user's language. Do not return bounding boxes or coordinates. Respond in natural language only.";
      bool frames = params.containsKey("frames") ? params["frames"].as<bool>() : true;
      String task = params.containsKey("task") ? params["task"].as<String>() : "NONE";

      String response = geminiVisionRequest(workId, query, frames);
      handleAgentResponse(workId, response);  // Vision 結果可能觸發更多工具

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        executeToolHistory += workId + " " + command + " [ "+query+" | "+frames+" | "+task+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck, task);

    }
    // ---------- 重新啟動裝置 ----------
    else if (command == "/reboot") {
      replyUserMessage(workId, "Rebooting the device, please wait...");

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      Serial.println("User requested reboot the device.");
      vTaskDelay(2000 / portTICK_PERIOD_MS);  // 等待 Telegram 訊息傳出

      NVIC_SystemReset();  // 透過 ARM Cortex-M 系統重置指令重啟

    }
    // ---------- TCP 裝置間訊息傳遞 ----------
    else if (command == "/tcpSendMessage") {
      String device = params["device"].as<String>();
      String message = params["message"].as<String>();

      String response = tcpSendMessage(workId, device, message);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+device+" | "+message+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 傳送 Telegram 文字訊息（可指定其他 Bot 和對話） ----------
    else if (command == "/telegramSendMessage") {
      String token = params["token"].as<String>();
      String chatId = params["chatId"].as<String>();
      String message = params["message"].as<String>();

      String response = telegramSendMessage(token, chatId, message);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        // 日誌中只顯示 Token 前 5 個字元，保護敏感資訊
        executeToolHistory += workId + " " + command + " [ "+token.substring(0, 5)+"... | "+chatId+" | "+message+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 傳送相機快照至指定 Telegram Bot ----------
    else if (command == "/telegramSendImage") {
      String token = params["token"].as<String>();
      String chatId = params["chatId"].as<String>();
      bool frames = params["frames"].as<bool>();

      String response = telegramSendCapturedImage(token, chatId, frames);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+token.substring(0, 5)+"... | "+chatId+" | "+frames+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 傳送 LINE 文字訊息 ----------
    else if (command == "/lineSendMessage") {
      String token = params["token"].as<String>();
      String targetId = params["targetId"].as<String>();
      String message = params["message"].as<String>();

      String response = lineSendMessage(token, targetId, message);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+token.substring(0, 5)+"... | "+targetId+" | "+message+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- SG90 伺服馬達角度控制 ----------
    else if (command == "/servo") {
        int pin = params["pin"].as<int>();
        int angle = params["angle"].as<int>();

        String response = tool_servo(servos[pin], pin, angle, workId);

        if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
          historicalMessages += buildGeminiMessage("user", command + timestamps);
          historicalMessages += buildGeminiMessage("model", response + timestamps);
          executeToolHistory += workId + " " + command + " [ " + String(pin) + " | " + String(angle) + " ]\n";
          xSemaphoreGive(stateMutex);
        }

        evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- DHT11 溫濕度讀取 ----------
    else if (command == "/dht11") {
      int pin = params["pin"].as<int>();

      String response = tool_dht11(pin, workId);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", command + timestamps);
        historicalMessages += buildGeminiMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ " + response  + " ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);

    }
    // ---------- 說明指令（/help 或 /start） ----------
    else if (command == "/help" || command == "/start") {

      // 動態替換記憶體資訊佔位符
      String mem = getMemoryInfo();
      String command = systemCommand;
      command.replace("<memory>", mem);

      // 請 Gemini 翻譯說明文字為使用者語言
      command = geminiChatRequest(workId, "Reply the following text in the user's language:\n\n" + command);

      replyUserMessage(workId, command, telegrambotKeyboard);  // 顯示快捷鍵盤

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildGeminiMessage("user", "Command list" + timestamps);
        historicalMessages += buildGeminiMessage("model", command + timestamps);
        xSemaphoreGive(stateMutex);
      }

    }
    // ---------- 未知指令：回退到 Gemini 對話模式 ----------
    else {
      // 若指令不在已知清單中，將其傳給 Gemini 作為自然語言請求處理
      String response = geminiChatRequest(workId, command);
      handleAgentResponse(workId, response);
    }
}

// ============================================================
// AI 代理回應處理器（Agent Response Handler）
// 這是「Gemini 回應 → 工具執行」的橋接層，負責：
//   1. 清理回應中的跳脫序列（\\n, \\"  等）
//   2. 判斷回應格式：
//      - 以 { 開頭 → 單一工具呼叫（tool_call JSON 物件）
//      - 以 [ 開頭 → 批次工具呼叫（tool_call JSON 陣列）
//      - 其他 → 自然語言回覆或 "NONE"（工作流程結束哨兵）
//   3. 解析並驗證 JSON，格式錯誤的一律拒絕執行並記錄到 Serial
//   4. 對自然語言回覆進行格式清理（移除 Markdown 語法等），
//      依目標管道（網頁/Telegram）做不同的 HTML 處理
//
// 批次模式的安全保護：
//   若批次中某個工具的 method 或 params 為空，立即停止，
//   不再執行後續工具（防止部分無效的批次命令繼續執行）。
//   只有最後一個工具才傳入 reCheck = true，觸發工作流程完成性評估。
// ============================================================
void handleAgentResponse(String workId, String message) {

  String rawMessage = message;  // 保留原始訊息，用於自然語言回覆時還原格式

  // 清理 JSON 跳脫序列，使字串可被 ArduinoJson 正確解析
  message.replace("\\\"", "\"");    // \" → "
  message.replace("\\\\", "\\");    // \\ → \
  message.replace("\\n", "");       // \n → 移除
  message.replace("\n", "");
  message.replace("\\r", "");
  message.replace("\r", "");
  message.replace("\\t", "");
  message.replace("\t", "");
  message.replace(String(char(0)), "");  // 移除 null 字元
  message.replace("\\-", "-");
  message.replace("\\*", "*");
  message.replace("\\_", "_");
  message.replace("\\#", "#");

  // ---------- 單一工具呼叫（JSON 物件） ----------
  if (message.startsWith("{") && message.indexOf("}") !=-1) {

    message = message.substring(0, message.lastIndexOf("}") + 1);  // 截取有效 JSON

    JsonObject obj;
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, message);
    if (error) {
        Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
        return;
    }

    obj = doc.as<JsonObject>();
    String method =  obj["method"].as<String>();
    JsonObject params = obj["params"];
    executeTool(workId, method, params);  // 執行工具

  }
  // ---------- 批次工具呼叫（JSON 陣列） ----------
  else if (message.startsWith("[") && message.indexOf("]") !=-1) {

    message = message.substring(0, message.lastIndexOf("]") + 1);  // 截取有效 JSON

    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, message);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
      return;
    }

    JsonArray toolsArray = doc.as<JsonArray>();
    int toolCount = toolsArray.size();

    // 循序執行批次中的每個工具
    for (int i = 0; i < toolCount; i++) {
      JsonObject toolObject = toolsArray[i];

      if (toolObject.isNull()) continue;

      String command = toolObject["method"].as<String>();
      JsonObject params = toolObject["params"];

      // 安全檢查：若工具物件不完整，立即中止整個批次
      if (command == "" || params.isNull()) {
        Serial.println("Incomplete tool detected → abort remaining tools");
        break;
      }

      // 只有最後一個工具才觸發工作流程完成性評估
      bool isLast = (i == toolCount - 1);

      executeTool(workId, command, params, isLast);
    }

  }
  // ---------- 自然語言回覆或 NONE（工作流程結束） ----------
  else {
    if (message.startsWith("[") || message.startsWith("{")) {
      // 以 JSON 符號開頭但解析失敗（不完整或格式錯誤）
      Serial.println("[DEBUG] Json parse failed: (handleAgentResponse)\n" + message);
      replyUserMessage(workId, "Json parse failed (handleAgentResponse). Please type \"Continue\"");

    } else if (message != "NONE") {
      // 非 NONE 且非 JSON：視為自然語言回覆，傳送給使用者
      message = rawMessage;  // 還原原始格式（保留換行等）

      if (workId.startsWith(String(taskTags[0]))) {
        // 網頁介面請求：進行較寬鬆的格式清理（不需 HTML 跳脫）
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        // 移除 Markdown 標記符號
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");
        message.replace("___", "");
      }
      else {
        // Telegram 請求：需要額外對 HTML 特殊字元進行跳脫
        // 避免 Telegram HTML 模式解析錯誤
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        message.replace("&", "&amp;");   // & 必須先跳脫
        message.replace("<", "&lt;");    // < 跳脫為 HTML 實體
        message.replace(">", "&gt;");    // > 跳脫為 HTML 實體
        // 移除 Markdown 標記符號
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");
        message.replace("___", "");
      }

      replyUserMessage(workId, message);
    }
    // message == "NONE"：工作流程結束哨兵，不傳送任何訊息
  }
}

// ============================================================
// 將音訊（或任意二進位）檔案以 base64 編碼後傳送至 Gemini 進行分析
// 主要用途：Telegram 語音訊息的語音轉文字（STT）
// 流程：
//   1. 將音訊資料進行 base64 編碼（動態配置緩衝區）
//   2. 組合 multimodal Gemini 請求（包含音訊 data 和提示詞）
//   3. 傳送至 Gemini，解析回應取得轉錄文字
//   4. 釋放 base64 緩衝區
//
// 參數：
//   fileinput → 音訊資料的位元組指標
//   fileSize  → 音訊資料大小（位元組）
//   mimeType  → MIME 類型（例如 "audio/ogg; codecs=opus"）
//   prompt    → 給 Gemini 的提示（例如 "請逐字轉錄這段語音"）
//
// 回傳值：Gemini 轉錄的文字，或錯誤訊息
// ============================================================
String sendFileToGemini(uint8_t* fileinput, size_t fileSize, String mimeType, String prompt) {

  // 計算 base64 編碼後的長度並配置緩衝區
  int   encodedLen  = base64_enc_len(fileSize);
  char* encodedData = (char*)malloc(encodedLen);
  if (!encodedData) {
    Serial.println("[STT] malloc failed for Base64 buffer");
    return "Malloc failed for Base64 encoding.";
  }
  base64_encode(encodedData, (char*)fileinput, fileSize);  // 執行 base64 編碼

  // 清理提示詞中可能破壞 JSON 格式的字元
  prompt.replace("\n", "");
  prompt.replace("\"", "\\\"");

  // 組合 multimodal 請求 JSON（含 inline 音訊資料和文字提示）
  String request =
    "{\"contents\": [{\"role\": \"user\", \"parts\": ["
    "{\"inline_data\": {\"data\": \"" + String(encodedData) + "\","
    "\"mime_type\": \"" + mimeType + "\"}},"
    "{\"text\": \"" + prompt + "\"}"
    "]}]}";

  free(encodedData);  // 請求字串已複製資料，立即釋放 base64 緩衝區

  WiFiSSLClient client;
  if (!client.connect("generativelanguage.googleapis.com", 443)) {
    Serial.println("[STT] Connection to Gemini failed");
    return "Connected to Gemini failed.";
  }

  // 傳送 HTTP POST 請求（音訊請求通常較大）
  client.println("POST /v1beta/models/" + geminiModel +
                 ":generateContent?key=" + geminiApiKey + " HTTP/1.1");
  client.println("Host: generativelanguage.googleapis.com");
  client.println("Content-Type: application/json; charset=utf-8");
  client.println("Content-Length: " + String(request.length()));
  client.println("Connection: close");
  client.println();

  for (int i = 0; i < (int)request.length(); i += 1024) {
    client.print(request.substring(i, i + 1024));
  }

  String body = "";
  unsigned long timeout = millis() + 20000;
  bool headersEnded = false;
  String line = "";

  while ((client.connected() || client.available()) && millis() < timeout) {
    while (client.available()) {
      char c = client.read();

      if (!headersEnded) {
        if (c == '\n') {
          if (line.length() <= 1) {
            headersEnded = true;
          }
          line = "";
        } else if (c != '\r') {
          line += c;
        }
      } else {
        body += c;
      }
    }
    // [看門狗修復] 音訊檔案較大，Gemini 處理時間較長，讓出 CPU 防止 WDT 觸發
    vTaskDelay(1);
  }

  client.stop();

  int jsonStart = body.indexOf('{');
  if (jsonStart != -1) body = body.substring(jsonStart);

  DynamicJsonDocument doc(8192);
  DeserializationError err = deserializeJson(doc, body);

  if (err) {
    Serial.println("[DEBUG] JSON parse failed: (sendFileToGemini)\n" + body);
    return "JSON parse failed (sendFileToGemini). Please try again.";
  }

  if (doc.containsKey("error")) {
    String msg = "Gemini STT Error: " + doc["error"]["message"].as<String>();
    return msg;
  }

  if (doc["candidates"][0]["content"]["parts"][0].containsKey("text")) {
    String result = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    result.replace("\n", "");  // 移除轉錄文字中多餘的換行
    return result;
  }

  return "No text returned from Gemini.";
}

// ============================================================
// 從 Telegram CDN 伺服器下載檔案（語音訊息等）
// 使用 HTTP/1.0 防止分塊傳輸編碼（chunked transfer encoding），
// 確保接收到的是純二進位資料。
//
// 流程：
//   1. 連線到 api.telegram.org
//   2. 發送 GET /file/bot{token}/{filePath} 請求
//   3. 略過 HTTP 標頭（等待 \r\n\r\n 分隔符）
//   4. 以位元組方式讀取二進位主體，儲存到動態配置的緩衝區
//   5. 更新 downloadedFileSize 記錄實際下載大小
//
// 回傳值：指向下載資料的堆積記憶體指標（呼叫者負責 free()），
//          若配置失敗則回傳 NULL
// ============================================================
uint8_t* downloadTelegramFile(String filePath) {

  uint8_t* voiceFile = (uint8_t*)malloc(MAX_FILE_SIZE);  // 配置最大 256KB 的緩衝區
  if (!voiceFile) return NULL;

  downloadedFileSize = 0;
  WiFiSSLClient client;

  if (client.connect("api.telegram.org", 443)) {

    // HTTP/1.0 避免 chunked 傳輸編碼，直接接收純二進位資料
    client.println("GET /file/bot" + telegrambotToken + "/" + filePath + " HTTP/1.0");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();

    // 略過 HTTP 標頭，等待 \r\n\r\n（標頭結束標記）
    String header    = "";
    long   startTime = millis();

    while (client.connected() || client.available()) {
      if (millis() - startTime > 10000) break;  // 逾時 10 秒
      if (client.available()) {
        char c = client.read();
        header += c;
        if (header.endsWith("\r\n\r\n")) break;  // 找到標頭結束標記
      }
    }

    // 逐位元組讀取二進位主體到緩衝區
    startTime = millis();
    while ((client.connected() || client.available()) &&
           downloadedFileSize < MAX_FILE_SIZE) {
      if (millis() - startTime > 10000) break;
      if (client.available()) {
        voiceFile[downloadedFileSize++] = client.read();
        startTime = millis();  // 每收到一個位元組重置逾時（防止資料中途超時）
      }
    }

    client.stop();
  }

  return voiceFile;
}

// ============================================================
// 透過 Telegram getFile API 將 file_id 轉換為下載路徑
// Telegram 的語音訊息等媒體不直接提供下載 URL，
// 需要先呼叫 getFile API 取得 file_path，再組合成下載 URL。
//
// 參數：
//   fileId → Telegram 語音訊息中的 file_id 欄位
//
// 回傳值：相對路徑字串（例如 "voice/file_123.oga"）
//          使用時需加上 https://api.telegram.org/file/bot{token}/ 前綴
// ============================================================
String getTelegramFilePath(String fileId) {

  WiFiSSLClient client;
  String filePath = "";
  String getAll = "", getBody = "";

  if (client.connect("api.telegram.org", 443)) {

    // 呼叫 getFile API 取得檔案資訊
    client.println("GET /bot" + telegrambotToken +
                   "/getFile?file_id=" + fileId + " HTTP/1.1");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();

    int     waitTime  = 5000;
    long    startTime = millis();
    boolean state     = false;

    // 讀取 HTTP 回應，略過標頭取得主體 JSON
    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (c == '\n') {
          if (getAll.length() == 0) state = true;  // 空白行 = 標頭結束
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state == true) getBody += String(c);

        startTime = millis();
      }

      if (getBody.length() > 0) break;
    }

    // 解析 JSON，提取 file_path 欄位
    DynamicJsonDocument doc(8192);
    deserializeJson(doc, getBody);
    filePath = doc["result"]["file_path"].as<String>();
  }

  return filePath;
}

// ============================================================
// Telegram Bot 訊息輪詢（Long Polling）
// 使用持久連線（keep-alive）不斷輪詢 Telegram 的 getUpdates API，
// 以取得使用者傳來的最新訊息（文字或語音）。
//
// 流程：
//   1. 若連線已斷開，重新建立 SSL 連線到 api.telegram.org
//   2. 發送 POST /getUpdates 請求（limit=1 每次只取最新一筆）
//   3. 解析回應 JSON，取得 message_id 和訊息內容
//   4. 比對 lastMessageId：若是新訊息則處理，否則跳過
//   5. 安全驗證：確認訊息來自設定的 telegrambotChatId
//   6. 根據訊息類型分流處理：
//      - 文字訊息：若以 / 開頭則呼叫 executeTool()，否則呼叫 geminiChatRequest()
//      - 語音訊息：呼叫 getTelegramFilePath() + downloadTelegramFile() + sendFileToGemini()
//                 進行語音轉文字，再處理轉錄結果
//   7. 每次處理後將對話歷史寫入 SD 卡（memory.md）
//   8. 若 WiFi 斷線，嘗試重新連線
// ============================================================
void getTelegramMessage() {

  const char* myDomain  = "api.telegram.org";
  String      getAll    = "";
  String      getTime   = "";
  String      getBody   = "";

  JsonObject          obj;
  DynamicJsonDocument doc(8192);

  String text        = "";
  String voiceFileId = "";
  long   message_id  = 0;

  // 重用現有連線；若已斷線則重新建立（節省 TLS 握手時間）
  if (!botClient.connected()) {
    if (lastMessageId == 0)
      Serial.println("Connect to " + String(myDomain));

    if (!botClient.connect(myDomain, 443))
      return;

    if (lastMessageId == 0)
      Serial.println("Connection successful");
  }

  while (botClient.connected()) {

    getAll  = "";
    getTime = "";
    getBody = "";

    // 發送 getUpdates 請求：取最新 1 筆，只訂閱 message 類型更新
    String request = "limit=1&offset=-1&allowed_updates=message";

    botClient.println("POST /bot" + telegrambotToken + "/getUpdates HTTP/1.1");
    botClient.println("Host: "           + String(myDomain));
    botClient.println("Content-Length: " + String(request.length()));
    botClient.println("Content-Type: application/x-www-form-urlencoded");
    botClient.println("Connection: keep-alive");  // 保持長連線，避免頻繁重建
    botClient.println();
    botClient.print(request);

    int           waitTime    = 5000;
    unsigned long startTime   = millis();
    bool          state       = false;
    bool          dataReceived = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);  // [看門狗修復] 讓出 CPU

      while (botClient.available()) {
        char c = botClient.read();

        if (c == '\n') {
          if (getAll.length() == 0)
            state = true;    // 空白行 = HTTP 標頭結束，開始讀取主體
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state) {
          getBody += String(c);    // 讀取回應主體（JSON）
        } else {
          // 在標頭中擷取 Date 欄位（用於 workId 的時間戳記）
          if      (getTime.indexOf("Date:") != -1)
            getTime  = "";
          else if (getTime.indexOf("Content-Type") != -1)
            getTime += "";
          else
            getTime += String(c);
        }

        startTime = millis();  // 重置逾時
      }

      if (getBody.length() > 0) {
        dataReceived = true;
        break;
      }
    }

    getTime.replace("Content-Type", "");

    // 建立工作 ID（格式：<BOT> + HTTP 回應日期時間）
    String workId = String(taskTags[1]) + " " + getTime;

    if (!dataReceived || getBody == "") return;

    // 解析 Telegram getUpdates 回應 JSON
    DeserializationError err = deserializeJson(doc, getBody);
    if (err) {
      Serial.println("[DEBUG] JSON parse failed: (getTelegramMessage)\n" + getBody);
      return;
    }
    obj = doc.as<JsonObject>();

    message_id = obj["result"][0]["message"]["message_id"].as<long>();

    if (message_id && message_id != lastMessageId) {

      long id_last  = lastMessageId;
      lastMessageId = message_id;

      // 安全驗證：拒絕來自未授權 Chat ID 的訊息
      String fromChatId = obj["result"][0]["message"]["chat"]["id"].as<String>();
      if (fromChatId != telegrambotChatId) {
        return;
      }

      if (id_last == 0) {
        // 第一次啟動：忽略初始訊息，設 message_id = 0 跳過處理
        message_id = 0;

      } else {
        // 正常訊息處理：更新 workId 為目前 RTC 時間
        workId = String(taskTags[1]) + " " + getRtcTimeString();

        if (obj["result"][0]["message"].containsKey("text")) {
          // ---- 處理文字訊息 ----
          text = obj["result"][0]["message"]["text"].as<String>();

          if (text == "help") {
            executeTool(workId, "/help", JsonObject());   // 顯示說明

          } else if (text == "null") {
            botClient.stop();  // 特殊指令：強制關閉連線

          } else if (text.startsWith("/")) {
            executeTool(workId, text, JsonObject());  // 內建指令

          } else {
            // 自然語言訊息：傳給 Gemini 處理
            text = geminiChatRequest(workId, text);
            handleAgentResponse(workId, text);
          }

          // 每次處理後持久化對話歷史到 SD 卡
          String localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory);

        } else if (doc["result"][0]["message"].containsKey("voice")) {
          // ---- 處理語音訊息（Telegram Voice Message）----
          voiceFileId = doc["result"][0]["message"]["voice"]["file_id"].as<String>();

          // 步驟 1：取得檔案下載路徑
          String   filePath  = getTelegramFilePath(voiceFileId);
          // 步驟 2：從 Telegram CDN 下載音訊檔案到堆積記憶體
          uint8_t* voiceFile = downloadTelegramFile(filePath);

          if (voiceFile && downloadedFileSize > 0) {
            // 步驟 3：傳送給 Gemini 進行語音轉文字（STT）
            text = sendFileToGemini(
              voiceFile, downloadedFileSize,
              "audio/ogg; codecs=opus",    // Telegram 語音訊息的 MIME 類型
              "Transcribe this audio to text exactly as spoken."
            );

            // 步驟 4：處理轉錄結果（指令或自然語言）
            if (text.startsWith("/")) {
              executeTool(workId, text, JsonObject());
            } else {
              text = geminiChatRequest(workId, text);
              handleAgentResponse(workId, text);
            }
          }

          // 釋放動態配置的音訊緩衝區
          if (voiceFile)
            free(voiceFile);

          // 持久化對話歷史
          String localHistory2 = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory2 = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory2);
        }
      }
    }

    vTaskDelay(5 / portTICK_PERIOD_MS);  // 讓出 CPU 避免 CPU 空轉
  }

  // ---- WiFi 重新連線（若連線中斷）----
  while (WiFi.status() != WL_CONNECTED) {
    WiFi.disconnect();
    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());

    unsigned long start = millis();

    while (WiFi.status() != WL_CONNECTED && millis() - start < 15000)
      vTaskDelay(500 / portTICK_PERIOD_MS);
  }

}

// ============================================================
// FreeRTOS 任務：HTTP 管理介面請求處理（埠 81）
// 這個任務持續監聽埠 81 的 HTTP 連線，
// 提供 fuClaw 系統的網頁設定管理介面。
//
// 支援的 HTTP GET 端點：
//   GET /                → 回傳設定管理頁面（index.html），填入目前設定值
//   GET /updateConfig?{} → 更新系統設定（WiFi/Telegram/Gemini 等），儲存到 env.json
//   GET /agent           → 回傳代理管理頁面（index_agent.html）
//   GET /getSoul         → 回傳目前的角色提示詞（geminiRole）
//   GET /updateSoul?     → 更新角色提示詞，儲存到 soul.md
//   GET /getDevice       → 回傳目前的裝置定義
//   GET /updateDevice?   → 更新裝置定義，儲存到 device.md
//   GET /getSkill        → 回傳目前的技能定義
//   GET /updateSkill?    → 更新技能定義，儲存到 skill.md
//   GET /chat            → 回傳 Gemini 聊天頁面（index_chat.html）
//   GET /schedule        → 回傳排程管理頁面（index_schedule.html）
//   GET /getScheduleTasks→ 回傳目前排程任務 JSON
//   GET /updateScheduleTasks?→ 更新排程任務清單，儲存到 schedule.json
//   GET /message?{msg}   → 接收使用者訊息，轉發給 Gemini 或執行工具，回傳回應
//
// 回應格式：HTTP/1.1 200 OK，Content-Type: text/html，以 512 位元組分批傳送
// ============================================================
void task_getRequest(void *param) {
  (void)param;
  while (1) {

    WiFiClient client = server.available();  // 等待客戶端連線

    if (client) {
      String currentLine = "";  // 用於累積目前讀取的 HTTP 請求行

      while (client.connected() || client.available()) {

        if (client.available()) {
          char c = client.read();

          if (c == '\n') {
            if (currentLine.length() == 0) {
              // 空白行 = HTTP 標頭結束，開始傳送回應

              // 取得並清空回應緩衝區（mainPageHTML）
              String pageToSend = mainPageHTML;
              mainPageHTML = "";

              // 傳送 HTTP 回應標頭
              client.println("HTTP/1.1 200 OK");
              client.println("Content-Type: text/html; charset=utf-8");
              client.println("Content-Length: " + String(pageToSend.length()));
              client.println("Access-Control-Allow-Origin: *");
              client.println("Cache-Control: no-cache");
              client.println();

              // 以 512 位元組分批傳送回應主體（避免大型 HTML 一次傳送失敗）
              const char* ptr = pageToSend.c_str();
              int total  = pageToSend.length();
              int sent   = 0;
              while (sent < total) {
                int chunk   = (total - sent) > 512 ? 512 : (total - sent);
                int written = client.write((const uint8_t*)(ptr + sent), chunk);
                if (written > 0) sent += written;
                else delay(5);
              }
              client.flush();

              break;  // 傳送完畢，結束 while 迴圈

            } else {
              currentLine = "";  // 非空白行：清除並繼續讀取
            }
          }
          else if (c != '\r') {
            currentLine += c;  // 累積目前行的字元
          }

          // ---- 根據請求路徑分流處理 ----

          // 首頁：回傳設定管理頁面（填入目前設定值）
          if ((currentLine.indexOf("GET / ") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

            mainPageHTML.replace("timeZone", timeZone);

          }
          // 更新系統設定（JSON 格式）
          else if ((currentLine.indexOf("GET /updateConfig?") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

            String workId = String(taskTags[0]) + " " + getRtcTimeString();

            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateConfig?", "");
            currentLine.replace(" HTTP/1.", "");

            if (currentLine.startsWith("{") && currentLine.endsWith("}")) {
              storeDataToFile(envFilename, currentLine);     // 儲存到 SD 卡
              setEnvironmentSettings(currentLine);           // 更新執行時設定

              // 重新組合裝置定義（加入最新的裝置名稱和時區）
              devicesDefinitionFinal = devicesDefinition;
              devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
              devicesDefinitionFinal += "\nDevice timezone: " + timeZone;

              systemContentReset();  // 重建系統提示詞（使用最新設定）

              mainPageHTML = "Configuration updated successfully.";
            }
            else
              mainPageHTML = "Configuration updated failed. JSON parse failed.";

            currentLine = "";

          }
          // 代理管理頁面
          else if ((currentLine.indexOf("GET /agent") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

            mainPageHTML = getStringFromFile(agentpageFilename);
            currentLine = "";

          // 取得角色提示詞（soul.md 內容）

            geminiRole = currentLine;                     // 更新執行時角色提示詞

            currentLine = "";

          }
          // 取得裝置定義（device.md 內容）
          else if ((currentLine.indexOf("GET /getDevice") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

            mainPageHTML = devicesDefinition;

              mainPageHTML = scheduleTasks;
              xSemaphoreGive(stateMutex);
            }

            currentLine = "";

          }
          // 更新排程任務清單（從網頁介面直接編輯）
          else if ((currentLine.indexOf("GET /updateScheduleTasks?") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

            currentLine = urldecode(currentLine);

                // 將排程更新事件記錄到對話歷史
                historicalMessages += buildGeminiMessage("user", "GET /updateScheduleTasks?<NEW SCHEDULE TASKS>");

                storeDataToFile(memoryFilename, localHistory);  // 同步持久化對話歷史
              }
            }
            else
              mainPageHTML = "Schedule updated failed. JSON parse failed.";

            currentLine = "";

          }

          else if ((currentLine.indexOf("GET /message?") != -1) && (currentLine.indexOf(" HTTP/1.") != -1)) {

              else {

          }
        }
        else {

        }
      }

      client.stop();
    }
    else {
      vTaskDelay(5 / portTICK_PERIOD_MS);  // 無連線時短暫休眠
    }
  }

// ============================================================
// FreeRTOS 任務：MJPEG 即時影像串流（埠 82）
// 持續監聽埠 82 的 HTTP 連線，提供相機的 MJPEG 即時串流。
// MJPEG 格式：使用 multipart/x-mixed-replace 持續推送 JPEG 幀。
// 每幀之間使用 boundary "Taiwan" 作為分隔。
// 瀏覽器連線後，相機會持續拍照並推送，直到客戶端斷線。
// ============================================================
void task_getRequestStream(void *param) {
  (void)param;

  while (1) {

    WiFiClient client = serverStream.available();
    uint32_t img_addr = 0;
    uint32_t img_len = 0;

    if (client) {
      String currentLine = "";

      while (client.connected() || client.available()) {

        if (client.available()) {
          char c = client.read();
          if (c == '\n') {
            if (currentLine.length() == 0) {
              // HTTP 標頭結束，開始串流

              String head = "--Taiwan\r\nContent-Type: image/jpeg\r\n\r\n";  // MJPEG 幀標頭

              // 傳送 HTTP 串流回應標頭
              client.println("HTTP/1.1 200 OK");
              client.println("Access-Control-Allow-Origin: *");
              client.println("Connection: keep-alive");
              client.println("Content-Type: multipart/x-mixed-replace; boundary=Taiwan");
              client.println();

              // 持續推送 JPEG 幀，直到客戶端斷線
              while(client.connected()) {

                Camera.getImage(0, &img_addr, &img_len);  // 取得最新影像幀
                uint8_t *fbBuf = (uint8_t*)img_addr;
                size_t fbLen = img_len;

                client.print(head);  // 傳送幀標頭

                // 以 1024 位元組分批傳送 JPEG 資料
                for (size_t n=0; n<fbLen; n=n+1024) {
                    if (n+1024<fbLen) {
                      client.write(fbBuf, 1024);
                      fbBuf += 1024;
                  }
                  else {
                    size_t remainder = fbLen - n;
                    if (remainder > 0)
                      client.write(fbBuf, remainder);
                  }
                }
                client.print("\r\n");  // 幀結束符

                vTaskDelay(10 / portTICK_PERIOD_MS);  // 短暫讓出 CPU（約 100 FPS 上限）
              }
              break;
            } else {
              currentLine = "";
            }
          }
          else if (c != '\r') {
            currentLine += c;
          }

          if (currentLine.indexOf(" HTTP/1.")!=-1) {
            currentLine="";  // 已讀取請求行，清空繼續處理標頭
          }
        }
        else {
          vTaskDelay(1);
        }
      }
      client.stop();
    }
    else {
      vTaskDelay(5 / portTICK_PERIOD_MS);
    }
  }
}

// FreeRTOS 任務：Telegram Bot 訊息接收（持續輪詢）
// 這個任務在背景持續執行，每 1 秒呼叫一次 getTelegramMessage()。
// 使用 botClientMutex 保護共享的 botClient SSL 連線，
// 防止與其他任務（排程任務、防盜偵測）同時使用造成衝突。
//
// 注意：看門狗計時器（WDT）的重置由 getTelegramMessage() 內部
// 的 vTaskDelay() 呼叫處理，不需要額外管理。
// ============================================================
void task_getTelegramMessage(void *param) {
  (void)param;
  while (1) {

    // 持有 botClientMutex 期間獨占 botClient 連線
    if (xSemaphoreTake(botClientMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      getTelegramMessage();   // 內部已有適當的 vTaskDelay 重置 WDT
      xSemaphoreGive(botClientMutex);
    }

    // [看門狗修復] 在互斥鎖釋放後稍作等待（getTelegramMessage 最多可能花費 5 秒）
    vTaskDelay(1000 / portTICK_PERIOD_MS);

  }
}

// FreeRTOS 任務：防盜偵測（每 5 分鐘執行一次）
// 這是一個定時觸發的技能（skill），透過 Gemini Vision 分析相機影像，
// 判斷是否有入侵者並發送 Telegram 警報。
//
// 設計細節：
//   - 5 分鐘等待被切割為 300 個 1 秒片段，讓 WDT 定期重置
//   - 執行前先取得 botClientMutex 並停止 Telegram 長輪詢連線，
//     確保此任務可以獨占使用網路資源
//   - 透過 evaluateWorkflowContinuation() 觸發 theft_detection 技能，
//     由 Gemini 決定具體的感應器讀取、視覺分析和通知動作
// ============================================================
void task_theft_detection(void *param) {
  (void)param;

  while (1) {

    // 將 5 分鐘（300 秒）等待切成每次 1 秒的片段，
    // 讓 FreeRTOS 看門狗計時器在等待期間也能正常重置
    for (int i = 0; i < 300000 / 1000; i++) {
      vTaskDelay(1000 / portTICK_PERIOD_MS);
    }

    // 取得 botClient 互斥鎖並停止 Telegram 長輪詢連線，
    // 讓此任務可以使用 SSL 網路資源執行 Gemini 分析
    if (xSemaphoreTake(botClientMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      botClient.stop();
      xSemaphoreGive(botClientMutex);
    }
    vTaskDelay(2000 / portTICK_PERIOD_MS);  // 等待 Telegram 任務完全釋放資源

    Serial.println("\n\nExecuting Skill: theft_detection\n\n");

    String workId = String(taskTags[4]) + " " + getRtcTimeString();

    // 觸發防盜偵測技能：強制要求 Gemini 執行 theft_detection 工作流程
    evaluateWorkflowContinuation(
      workId,
      true,
      "Must execute skill theft_detection. Return ONLY tool_call JSON."
    );
    // [看門狗修復] evaluateWorkflowContinuation 可能鏈式呼叫多次 Gemini+Vision，重置在完成後
  }
}

// ============================================================
// 整數補零格式化（用於時間顯示）
// 將個位數整數補前導零，確保時間格式為兩位數。
// 例：9 → "09"，12 → "12"
// ============================================================
String twoDigits(int value) {
  if (value < 10)
    return "0" + String(value);
  return String(value);
}

// ============================================================
// 檢查某個重複性排程任務今天是否已執行過
// 使用 |task| 格式在字串中搜尋任務名稱，避免誤判部分相符。
//
// 額外功能：當日期變更（跨過午夜）時，自動重置今日已執行記錄，
// 讓重複性任務在隔天可以再次執行。
// ============================================================
bool isExecutedToday(String task) {

  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  int today = now->tm_mday;

  // 若日期已變更（跨過午夜），自動重置今日執行記錄
  if (executedTodayDate != today) {
    executedTodayTasks = "";
    executedTodayDate = today;
  }

  // 以 |task| 格式搜尋，確保不會誤判子字串
  return executedTodayTasks.indexOf("|" + task + "|") != -1;
}

// ============================================================
// 將重複性排程任務標記為今日已執行
// 在任務執行後呼叫此函式，防止同一天重複執行。
// 同時更新 executedTodayDate 為當前日期（用於跨日重置邏輯）。
// ============================================================
void markExecutedToday(const String &task) {
  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  executedTodayDate = now->tm_mday;
  executedTodayTasks += "|" + task + "|";  // 以 | 包圍，防止子字串誤判
}

// ============================================================
// 取得未完成的排程任務清單（純文字格式）
// 篩選條件：
//   - executed = true 的任務跳過
//   - 重複性任務（year == 0）若今日已執行則跳過
// 回傳格式為純文字條列，方便顯示給使用者。
// ============================================================
String getUnfinishedScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);  // 排程清單可能較大，使用 16KB 緩衝區
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);

  if (err) {
      Serial.println("Schedule JSON parse failed");
      return "[]";
  }

  JsonArray tasks = doc.as<JsonArray>();

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();

  String result = "";

  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();

      if (executed) continue;  // 跳過已執行的任務

      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();

      // 組合唯一識別字串（排程時間 + 任務描述）用於今日執行記錄比對
      String scheduleStr;
      serializeJson(task["schedule"], scheduleStr);
      String compareTask = scheduleStr + " " + task["task"].as<String>();

      // 重複性任務若今日已執行則跳過
      if (year == 0 && isExecutedToday(compareTask))
          continue;

      // 格式化為人類可讀的條列格式
      result += "*** " + String(year) + "/" + String(month) + "/" + String(day) + " " +
                String(hour) + ":" + String(minute) + ":" + String(sec) +
                " ***\n" + task["task"].as<String>() + "\n";
  }

  if (result == "") result = "NULL";

  return result;
}

// ============================================================
// 取得目前應執行的排程任務 JSON 陣列
// 根據 RTC 目前時間，篩選出所有已到期但尚未執行的任務。
//
// 篩選規則：
//   1. executed = true 的任務跳過
//   2. 將 year/month/day 為 0 的欄位替換為目前的年/月/日（重複性任務處理）
//   3. 計算任務排程時間 taskTime，若 taskTime <= 目前時間（rawtime），視為到期
//   4. 若設有 scheduleTimeout（分鐘），超過容許延遲的任務跳過（補執行保護）
//   5. 重複性任務（year == 0）若今日已執行則跳過
//
// 回傳值：JSON 陣列字串（只含當前應執行的任務），或 "[]"
// ============================================================
String getExecuteScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);

  if (err) {
      Serial.println("Schedule JSON parse failed");
      return "[]";
  }

  JsonArray tasks = doc.as<JsonArray>();

  // 從硬體 RTC 取得目前時間
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();

  String result = "";

  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();
      if (executed) continue;  // 跳過已執行的任務

      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();

      // 將 0 值替換為目前的年/月/日（用於重複性任務：
      // year=0 → 今年；month=0 → 今月；day=0 → 今日）
      int resolvedYear  = (year  == 0) ? (now->tm_year + 1900) : year;
      int resolvedMonth = (month == 0) ? (now->tm_mon  + 1)    : month;
      int resolvedDay   = (day   == 0) ? (now->tm_mday)        : day;

      // 建立任務排程時間的 tm 結構
      struct tm tmTask = {};
      tmTask.tm_year = resolvedYear  - 1900;
      tmTask.tm_mon  = resolvedMonth - 1;
      tmTask.tm_mday = resolvedDay;
      tmTask.tm_hour = hour;
      tmTask.tm_min  = minute;
      tmTask.tm_sec  = sec;

      time_t taskTime = mktime(&tmTask);

      // 若任務排程時間已到（taskTime <= 目前時間）
      if (mktime(&tmTask) <= rawtime) {
        time_t diff = rawtime - taskTime;

        // 超過容許延遲（scheduleTimeout 分鐘）的任務跳過，防止無限補執行
        if ((scheduleTimeout > 0 && rawtime > taskTime && diff > (time_t)(scheduleTimeout * 60)))
            continue;

        // 重複性任務若今日已執行則跳過
        String scheduleStr;
        serializeJson(task["schedule"], scheduleStr);
        String compareTask = scheduleStr + " " + task["task"].as<String>();

        if (year == 0 && isExecutedToday(compareTask))
            continue;

        // 加入結果陣列（只包含 task 和 schedule 欄位）
        JsonObject item = resultArray.createNestedObject();
        item["task"] = task["task"].as<String>();
        item["schedule"] = schedule;
      }
  }

  serializeJson(resultDoc, result);
  return result;
}

// ============================================================
// FreeRTOS 任務：時間排程執行器（每 60 秒檢查一次）
// 這是系統排程引擎的核心任務，定時檢查並執行到期的排程任務。
//
// 執行流程：
//   1. 等待 60 秒（分 6 段，每段 10 秒，確保 WDT 定期重置）
//   2. 短暫停止 Telegram 長輪詢連線，取得網路資源
//   3. 確認 RTC 已初始化（rtcYear != 0），若未初始化則嘗試同步
//   4. 讀取目前排程任務清單，呼叫 getExecuteScheduleTasksJson() 篩選到期任務
//   5. 對每個到期任務：
//      - 建立提示詞要求 Gemini 執行對應的工具操作
//      - 呼叫 handleAgentResponse() 執行工具鏈
//      - 標記任務為「今日已執行」（防止重複執行）
//   6. 執行完所有任務後，更新 schedule.json 中的執行狀態
//   7. 持久化今日已執行記錄和對話歷史到 SD 卡
// ============================================================
void task_time_scheduling(void *param) {
  (void)param;
  while (1) {

    // [看門狗修復] 60 秒等待分 6 段（每段 10 秒），確保 WDT 定期重置
    for (int i = 0; i < 6; i++) {
      vTaskDelay(10000 / portTICK_PERIOD_MS);
    }

    // 停止 Telegram 長輪詢連線，讓此任務可以使用 SSL 資源
    if (xSemaphoreTake(botClientMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      botClient.stop();
      xSemaphoreGive(botClientMutex);
    }
    vTaskDelay(2000 / portTICK_PERIOD_MS);  // 等待 Telegram 任務完全釋放

    // 建立此次排程執行的工作 ID
    String workId = String(taskTags[3]) + " " + rtcFormatTime;

    // 確認 RTC 已初始化（rtcYear != 0 代表時間已同步）
    if (rtcYear == 0) {
      Serial.println("[DEBUG] RTC time is not initialized.");
      executeTool(workId, "/syncrtc", JsonObject(), false);  // 嘗試同步 RTC（不觸發 reCheck）
      if (rtcYear == 0)
        continue;  // 同步失敗，跳過本次排程檢查
    }

    // 讀取目前排程任務清單（持有 stateMutex）
    String localSchedule = "";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      // 確保 JSON 陣列格式完整
      if (scheduleTasks.startsWith("[") && scheduleTasks.indexOf("]") !=-1)
        scheduleTasks = scheduleTasks.substring(0, scheduleTasks.lastIndexOf("]") + 1);
      localSchedule = scheduleTasks;
      xSemaphoreGive(stateMutex);
    }

    if (localSchedule.startsWith("[") && localSchedule.indexOf("]") !=-1) {

      // 篩選目前應執行的排程任務
      String unfinishedScheduleTasksJson = getExecuteScheduleTasksJson(localSchedule);

      if (unfinishedScheduleTasksJson.startsWith("[") && unfinishedScheduleTasksJson.indexOf("]") !=-1) {
        unfinishedScheduleTasksJson = unfinishedScheduleTasksJson.substring(0, unfinishedScheduleTasksJson.lastIndexOf("]") + 1);

        String response = "";

        DynamicJsonDocument doc(8192);
        DeserializationError err = deserializeJson(doc, unfinishedScheduleTasksJson);
        if (err) {
          Serial.println("[DEBUG] JSON parse failed: (task_time_scheduling)\n" + unfinishedScheduleTasksJson);
          continue;   // 解析失敗：繼續下次迴圈，不 return（保持任務存活）
        }

        JsonArray tasks = doc.as<JsonArray>();

        // 逐一處理每個到期任務
        for (JsonObject obj : tasks) {

          String taskName = obj["task"].as<String>();
          String schedule = obj["schedule"].as<String>();
          String item = obj["task"].as<String>();

          if ((item.startsWith("{") && item.endsWith("}")) || (item.startsWith("[") && item.endsWith("]"))) {
			  handleAgentResponse(workId, item);
          } 
		  else {
			  // 建立嚴格的執行提示詞，強制 Gemini 執行工具操作（不允許自然語言回應）
			  String prompt =
				"This is a deterministic scheduling execution step. "

				"\n\nUnfinished scheduled tasks:\n" +
				item +

				// 強調：清單中的任務都是應立即執行的，不能跳過
				"\n\nThe task list above already contains ONLY tasks that have not been executed. "
				"Evaluate EVERY task in this list independently. "
				"If a task's scheduled time is less than or equal to the current time, "
				"it MUST be executed immediately. "
				"Do NOT skip any eligible task. "
				"More than one task may be eligible at the same time. "
				"If multiple tasks are eligible, execute ALL of them in the same response. "
				"Tasks whose scheduled time is still in the future must be ignored. "

				// 嚴格的輸出格式規定：只能輸出 tool_call JSON 或 NONE
				"Output rules: "
				"1. If no task is eligible for execution, return EXACTLY: NONE. "
				"2. If one or more tasks are eligible, return tool_call JSON for ALL eligible tasks. "
				"3. Never return natural language. "
				"4. Never explain. "
				"5. Never summarize. "
				"6. Never ask questions. "
				"7. Never claim success without tool execution results. "
				"8. Process every task in the provided task list. "
				"9. A task remains executable forever after its scheduled time has passed until it is marked executed=true. "
				"10. Do not stop after the first eligible task.";

			  response = geminiChatRequest(workId, prompt);
			  // [看門狗修復] geminiChatRequest 最多可能等待 20 秒，完成後重置

			  handleAgentResponse(workId, response);
			  // [看門狗修復] handleAgentResponse 可能觸發連鎖 Gemini 呼叫
		  }

          // 標記此任務為今日已執行（防止在下個 60 秒迴圈中重複執行）
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            markExecutedToday(schedule + " " + item);
            xSemaphoreGive(stateMutex);
          }
        }

        // 若有任務被執行，更新 schedule.json 中的執行狀態並持久化
        if (tasks.size()>0) {
          executeTool(workId, "/updateScheduleStatus", JsonObject(), false);  // 更新 executed 欄位

          String localExecuted = "";
          String localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localExecuted = executedTodayTasks;
            localHistory  = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(scheduleExecutedTodayTasksFilename, localExecuted);  // 持久化今日執行記錄
          storeDataToFile(memoryFilename, localHistory);                        // 持久化對話歷史
        }
      }
    }
  }
}

// ============================================================
// WiFi 初始化函式
// 同時啟用 STA 模式（連接到家用路由器）和 AP 模式（提供設定用熱點）。
// WiFi.enableConcurrent() 啟用雙模式並發（Concurrent WiFi）。
//
// 連線邏輯：
//   - 若 wifiSsid 為空，跳過 STA 連線（只使用 AP 模式）
//   - 最多重試 2 次，每次等待最多 15 秒
//   - 若連線超時，繼續執行（AP 模式仍可用）
// ============================================================
void initWiFi() {
  WiFi.enableConcurrent();  // 啟用 STA + AP 雙模式並發

  // 啟動 AP 模式（fuClaw 設定管理熱點）
  WiFi.apbegin((char*)apSsid.c_str(), (char*)apPassword.c_str(), channel_ap, 0);

  for (int i=0 ; i<2 ; i++) {  // 最多嘗試 2 次連線

    if (wifiSsid == "")
      break;  // 若無設定 SSID，跳過 STA 連線

    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());
    delay(1000);

    Serial.println();
    Serial.print("Connecting to ");
    Serial.println(wifiSsid);

    unsigned long StartTime=millis();

    while (WiFi.status() != WL_CONNECTED) {
      delay(500);

      if ((StartTime + 15000) < millis())
        break;  // 連線逾時（15 秒），跳出等待迴圈
    }
    
    if (WiFi.status() == WL_CONNECTED)
      break;    
  }
}

// ============================================================
// 從 JSON 字串更新系統設定（setEnvironmentSettings）
// 解析 env.json 的 JSON 物件，更新所有執行時設定變數。
// 在啟動時和網頁介面更新設定後呼叫。
// ============================================================
void setEnvironmentSettings(String jsonString) {

  DynamicJsonDocument doc(8192);
  DeserializationError error = deserializeJson(doc, jsonString);
  if (error) {
    Serial.println("[DEBUG] JSON parse failed : (setEnvironmentSettings)\n" + jsonString);
    return;
  }

  JsonObject obj = doc.as<JsonObject>();
  deviceName      = obj["device_name"].as<String>();   // 裝置名稱
  wifiSsid        = obj["wifi_ssid"].as<String>();     // WiFi SSID
  wifiPassword    = obj["wifi_pass"].as<String>();     // WiFi 密碼
  telegrambotToken  = obj["telegramBot_token"].as<String>();   // Telegram Bot Token
  telegrambotChatId = obj["telegramBot_chatID"].as<String>();  // 授權 Chat ID
  geminiApiKey    = obj["gemini_apikey"].as<String>(); // Gemini API 金鑰
  geminiModel     = obj["gemini_model"].as<String>();  // Gemini 模型名稱
  scheduleTimeout = obj["schedule_timeout"].as<int>(); // 排程容許延遲（分鐘）
  timeZone        = obj["timezone"].as<String>();      // 時區設定
}

// ============================================================
// 將 IPAddress 物件轉換為字串格式（例如 "192.168.1.100"）
// 用於在序列埠和 Telegram 訊息中顯示 IP 位址。
// ============================================================
String Ip2String(IPAddress ip) {
  return String(ip[0])+String(".")+String(ip[1])+String(".")+String(ip[2])+String(".")+String(ip[3]);
}

// ============================================================
// Arduino setup() 主要初始化函式
// 這是整個系統啟動的入口，依序執行以下初始化步驟：
//
//   1. 初始化序列埠（115200 baud）
//   2. 建立所有 FreeRTOS 互斥鎖（必須在任何使用它們的函式呼叫之前）
//   3. 設定狀態指示 LED 腳位
//   4. 從 SD 卡讀取 env.json 並套用系統設定
//   5. 連接 WiFi（STA + AP 雙模式）
//   6. 從 SD 卡讀取 soul.md（角色提示詞）、device.md（裝置定義）、
//      skill.md（技能定義）、schedule.json（排程任務）等設定檔
//   7. 組合三個系統提示詞預組裝字串
//   8. 從 memory.md 還原對話歷史記憶
//   9. 初始化相機
//  10. 同步 RTC 時間（透過 Gemini API）並傳送啟動訊息到 Telegram
//  11. 啟動 HTTP 伺服器（埠 81 管理介面、埠 82 影像串流）
//  12. 建立四個 FreeRTOS 任務：
//      - task_getRequest         → HTTP 管理介面（埠 81）
//      - task_getRequestStream   → MJPEG 影像串流（埠 82）
//      - task_getTelegramMessage → Telegram Bot 輪詢
//      - task_time_scheduling    → 排程任務執行
//  13. 若 WiFi 連線成功：LED 閃爍 3 次，傳送裝置 IP 到 Gemini 歷史
//
// 注意：若 soul.md 或 device.md 讀取失敗（空內容），系統將重啟。
// ============================================================
void setup() {
  Serial.begin(115200);

  // ---- 步驟 1：建立所有 FreeRTOS 互斥鎖 ----
  // 必須在任何可能使用這些鎖的函式（rtcInitialTime、replyUserMessage 等）呼叫之前建立
  botClientMutex = xSemaphoreCreateMutex();
  stateMutex     = xSemaphoreCreateMutex();
  sdMutex        = xSemaphoreCreateMutex();
  imageMutex     = xSemaphoreCreateMutex();

  // 若任何互斥鎖建立失敗，系統無法安全運作，立即重啟
  if (!botClientMutex || !stateMutex || !sdMutex || !imageMutex) {
    Serial.println("[DEBUG] Failed to create mutexes. Restarting the MCU...");
    delay(2000);
    NVIC_SystemReset();
  }

  // ---- 步驟 2：設定狀態指示 LED ----
  pinMode(ledPin, OUTPUT);

  // ---- 步驟 3：讀取環境設定檔（env.json）----
  String env = getStringFromFile(envFilename);
  Serial.println("env.json len: " + String(env.length()));
  if (env != "")
    setEnvironmentSettings(env);  // 套用 env.json 中的所有設定

  // ---- 步驟 4：連接 WiFi ----
  initWiFi();

  // ---- 步驟 5：讀取角色提示詞（soul.md）----
  String soul = getStringFromFile(soulFilename);
  Serial.println("Soul.md len: " + String(soul.length()));
  if (soul != "")
    geminiRole = soul;

  // ---- 步驟 6：讀取裝置定義（device.md）----
  String device = getStringFromFile(deviceFilename);
  Serial.println("device.md len: " + String(device.length()));
  if (device != "")
    devicesDefinition = device;

  // 組合最終版本裝置定義（加入裝置名稱和時區，供 Gemini 參考）
  devicesDefinitionFinal = devicesDefinition;
  devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
  devicesDefinitionFinal += "\nDevice timezone: " + timeZone;

  // 安全檢查：soul.md 和 device.md 是必要設定檔，若缺失則重啟
  if (geminiRole.length() == 0 || devicesDefinition.length() == 0) {
    Serial.println("System configuration failed. Restarting the MCU...");
    delay(5000);
    NVIC_SystemReset();
  }

  // ---- 步驟 7：讀取技能定義（skill.md）----
  String skill = getStringFromFile(skillFilename);
  Serial.println("skill.md len: " + String(skill.length()));
  if (skill != "")
    skillsDefinition = skill;

  // ---- 步驟 8：讀取排程任務（schedule.json）----
  String schedule = getStringFromFile(scheduleFilename);
  Serial.println("schedule.md len: " + String(schedule.length()));
  if (schedule != "")
    scheduleTasks = schedule;

  // ---- 步驟 9：讀取今日已執行排程記錄 ----
  String scheduleExecutedTodayTasks = getStringFromFile(scheduleExecutedTodayTasksFilename);
  Serial.println("scheduleTodayExecuted.md len: " + String(scheduleExecutedTodayTasks.length()));
  if (scheduleExecutedTodayTasks != "")
    executedTodayTasks = scheduleExecutedTodayTasks;

  // ---- 步驟 10：建立三個系統提示詞預組裝字串 ----
  systemContent = buildGeminiMessage("user", geminiRole, 0) + buildGeminiMessage("model", "OK");
  systemContentTools = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule + skillsDefinition + toolsDefinition, 0) + buildGeminiMessage("model", "OK");
  systemContentNoTools = buildGeminiMessage("user", geminiRole + devicesDefinitionFinal + devicesRule, 0) + buildGeminiMessage("model", "OK");

  // ---- 步驟 11：從 memory.md 還原對話歷史記憶 ----
  String memory = getStringFromFile(memoryFilename);
  Serial.println("memory.md len: " + String(memory.length()));
  if (memory != "")
    historicalMessages = memory;

  // ---- 步驟 12：初始化相機 ----
  config.setRotation(0);           // 不旋轉
  Camera.configVideoChannel(0, config);
  Camera.videoInit();
  Camera.channelBegin(0);

  // ---- 步驟 13：同步 RTC 時間（向 Gemini API 請求目前時間）----
  rtcInitialTime("RTC Initial Time");

  // 傳送啟動訊息到 Telegram（包含目前時間和快捷鍵盤）
  replyUserMessage(String(taskTags[1]) + " " + getRtcTimeString(), "RTC START: " + getRtcTimeString(), telegrambotKeyboard);

  // 同步 executedTodayDate（確保跨日重置邏輯使用正確的基準日期）
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);
  executedTodayDate = now->tm_mday;

  // ---- 步驟 14：啟動 HTTP 伺服器 ----
  server.begin();        // 埠 81：管理介面
  serverStream.begin();  // 埠 82：影像串流

  // ---- 步驟 15：建立 FreeRTOS 背景任務 ----
  // 每個任務分配 16KB 堆疊空間（排程任務因較簡單只需 6KB）

  // HTTP 管理介面任務
  if (xTaskCreate(
        task_getRequest,
        (const char *)"task_getRequest",
        16384,                    // 堆疊大小：16KB
        NULL,
        tskIDLE_PRIORITY + 1,    // 優先順序：略高於 idle
        NULL
      )!= pdPASS) {

    Serial.println("Create task_task_getRequest failed");
  }

  // MJPEG 影像串流任務
  if (xTaskCreate(
        task_getRequestStream,
        (const char *)"task_getRequestStream",
        16384,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_getRequestStream failed");
  }

  // Telegram Bot 輪詢任務
  if (xTaskCreate(
        task_getTelegramMessage,
        (const char *)"task_getTelegramMessage",
        16384,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_getTelegramMessage failed");
  }

  // 時間排程執行任務（堆疊需求較小，6KB 即可）
  if (xTaskCreate(
        task_time_scheduling,
        (const char *)"task_time_scheduling",
        6144,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_time_scheduling failed");
  }

  // ---- 步驟 16：輸出網路資訊到序列埠 ----
  // AP 模式資訊（固定可用）
  Serial.println("AP mode");
  Serial.println("fuClaw Manager: http://192.168.1.1:81");
  Serial.println("Video stream: http://192.168.1.1:82");
  Serial.println("AP ssid : " + apSsid);
  Serial.println("AP password : " + apPassword);
  Serial.println();

  // ---- 步驟 17：WiFi STA 模式成功連線後的處理 ----
  if (WiFi.status() == WL_CONNECTED) {
    // LED 閃爍 3 次，表示 WiFi 連線成功
    for (int i=0 ; i<3 ; i++) {
      digitalWrite(ledPin, 1);
      delay(300);
      digitalWrite(ledPin, 0);
      delay(300);
    }

    // 輸出 STA 模式的 IP 位址到序列埠
    Serial.println("STA mode");
    Serial.println("fuClaw Manager: http://" + Ip2String(WiFi.localIP()) + ":81");
    Serial.println("Video stream: http://" + Ip2String(WiFi.localIP()) + ":82");
    Serial.println();

    // 將裝置 IP 記錄到對話歷史，讓 Gemini 知道裝置的網路位址
    historicalMessages += buildGeminiMessage("user", "Device IP: " + Ip2String(WiFi.localIP()));
  }
}

// ============================================================
// Arduino loop() 主迴圈
// 所有業務邏輯都在 FreeRTOS 背景任務中執行，
// loop() 本身只需讓出 CPU 時間給排程器。
// 使用 vTaskDelay 而非 delay，確保 FreeRTOS 排程器可以正常運作。
// ============================================================
void loop() {
  // 主迴圈（loopTask）無需執行任何業務邏輯，
  // 所有工作均由以下四個 FreeRTOS 背景任務處理：
  //   - task_getRequest         → HTTP 管理介面
  //   - task_getRequestStream   → MJPEG 影像串流
  //   - task_getTelegramMessage → Telegram 訊息輪詢
  //   - task_time_scheduling    → 排程任務執行
  // 此任務不在 TWDT（看門狗）監控清單中，因為它不做任何阻塞操作。
  vTaskDelay(1000 / portTICK_PERIOD_MS);
}
