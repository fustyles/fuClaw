// ============================================================
// 【工具函式】URL 解碼
// 用途:將 HTTP 請求中的 URL-encoded 字串(如表單資料、查詢參數)還原成原始文字。
// 例如 "%E4%B8%AD%E6%96%87" 會被還原成對應的中文字元,"+" 還原成空白。
// ============================================================
String urldecode(const String& input) {
    String result = "";
    result.reserve(input.length());  // 預先配置記憶體,避免字串反覆重新配置造成的效能損耗
    for (int i = 0; i < (int)input.length(); i++) {
        if (input[i] == '%' && i + 2 < (int)input.length()) {
            // 讀取 % 後面兩個十六進位字元,轉換成對應的位元組值(例如 %E4 -> 0xE4)
            char hex[3] = { input[i+1], input[i+2], '\0' };
            uint8_t val = (uint8_t)strtol(hex, nullptr, 16);
            result.concat((char)val);
            i += 2;  // 跳過已處理的兩個十六進位字元
        } else if (input[i] == '+') {
            // application/x-www-form-urlencoded 格式中,+ 代表空白字元
            result += ' ';
        } else {
            result += input[i];
        }
    }
    return result;
}

// ============================================================
// 【工具函式】URL 編碼
// 用途:將字串轉成 URL-safe 格式,英數字與 -_.~ 保留原樣,其餘字元轉成 %XX 十六進位形式。
// 常用於組裝 GET 請求的查詢參數。
// ============================================================
String urlencode(String str) {
  const char *msg = str.c_str();
  const char *hex = "0123456789ABCDEF";
  String encodedMsg = "";
  while (*msg != '\0') {
    // 保留英文字母、數字與少數安全符號,其餘一律轉義
    if (('a' <= *msg && *msg <= 'z') || ('A' <= *msg && *msg <= 'Z') || ('0' <= *msg && *msg <= '9') || *msg == '-' || *msg == '_' || *msg == '.' || *msg == '~') {
      encodedMsg += *msg;
    } else {
      encodedMsg += '%';
      encodedMsg += hex[(unsigned char)*msg >> 4];  // 高位半位元組轉十六進位字元
      encodedMsg += hex[*msg & 0xf];                 // 低位半位元組轉十六進位字元
    }
    msg++;
  }
  return encodedMsg;
}

// ============================================================
// 【工具函式】JSON 字串跳脫處理
// 用途:將一般文字轉成可安全放進 JSON 字串值的形式(跳脫雙引號、反斜線、控制字元)。
// 注意 gemini 參數:Gemini API 對換行/Tab 等控制字元的容忍度與 OpenAI/Grok 不同,
// 因此依供應商決定是否要把 \r \n \t 也跳脫成 \\r \\n \\t,或直接原樣保留。
// ============================================================
String escapeForJson(const String &src, bool gemini) {
  unsigned int n = src.length();
  const char* buf = src.c_str();

  String out;
  out.reserve(2 * n + 1);  // 最壞情況下每個字元都要跳脫成兩個字元,預先配置足夠空間避免頻繁重新配置

  for (unsigned int i = 0; i < n; i++) {
    char c = buf[i];
    switch (c) {
      case '"':
        out += '\\'; out += '"';   // 雙引號一律要跳脫,否則會提早結束 JSON 字串
        break;
      case '\\':
        out += '\\'; out += '\\';  // 反斜線本身也要跳脫
        break;
      case '\r':
        if (gemini) out += c;               // Gemini：保留原始 \r
        else { out += '\\'; out += 'r'; }    // OpenAI/Grok：跳脫成 \\r
        break;
      case '\n':
        if (gemini) out += c;
        else { out += '\\'; out += 'n'; }
        break;
      case '\t':
        if (gemini) out += c;
        else { out += '\\'; out += 't'; }
        break;
      default:
        out += c;  // 一般字元原樣輸出
    }
  }
  return out;
}

// ============================================================
// 【核心函式】組裝一則對話訊息的 JSON 片段(供多供應商共用)
// 用途:把 role(user/model)與 message 內容包成 {"role": "...", "content": "..."} 的 JSON 片段,
// 用於累加進 historicalMessages 對話歷史字串。
// 依 llmType 決定跳脫規則,並且非 Gemini 供應商時把角色名稱 "model" 改成 OpenAI/Grok 慣用的 "system"。
// comma 參數控制開頭是否加上逗號,方便串接在既有 JSON 陣列後面。
// ============================================================
String buildLlmMessage(String role, String message, bool comma = true) {
  String esc;
  String jsonMessage;

  if (llmType == "gemini") {
    esc = escapeForJson(message, true);  // Gemini 格式跳脫規則
  } 
  else {
    esc = escapeForJson(message, false);  // OpenAI / Grok 格式跳脫規則
    role.replace("model", "system");      // Gemini 慣用 "model" 角色名稱，非 Gemini 供應商需改名對應
  }
  
  jsonMessage.reserve(esc.length() + role.length() + 48);  // 預先配置記憶體，減少字串重新配置次數
  
  jsonMessage  = comma ? ", {\"role\": \"" : "{\"role\": \"";  // 依需要決定開頭是否要接續逗號
  jsonMessage += role;
  jsonMessage += "\", \"content\": \"";
  jsonMessage += esc;
  jsonMessage += "\" }";
    
  return jsonMessage;
}

// ============================================================
// 【核心函式】取得目前的 GMT(格林威治)日期時間字串
// 用途:直接對 google.com 發出一次輕量 HTTPS 請求,從 HTTP 回應的 "Date:" 標頭擷取伺服器端時間。
// 這個作法刻意「不」依賴任何一家 LLM 供應商(Gemini/OpenAI/Grok),
// 因為裝置現在可能設定成任何一家，時間同步不該綁死在特定供應商的可用性上。
// 若連線失敗，則退回一段提示文字，讓後續呼叫 LLM 時改用「有依據的搜尋」去推算時間。
// ============================================================
String getGoogleDatetime() {

  WiFiSSLClient client;
  String getDatetime = "";

  if (client.connect("google.com", 443)) {
    // 送出最簡單的 HTTP GET 請求，目的只是為了拿到回應標頭裡的 Date 欄位，不需要頁面內容
    client.println("GET / HTTP/1.1");
    client.println("Host: google.com");
    client.println("Connection: close");    
    client.println();

    int waitTime = 5000;  // 最長等待 5 秒
    unsigned long startTime = millis();
    bool getStatus = false;  // 是否已經找到 "Date:" 標頭並開始收集其後內容

    // 逐字元讀取回應，直到抓到 Date 標頭那一行或逾時
    while ((startTime + waitTime) > millis()){
      vTaskDelay(100 / portTICK_PERIOD_MS);  // 讓出 CPU 時間片給其他 Task，避免忙等待獨佔處理器

      while (client.available()){
        char c = client.read();

        if (getStatus == true && c == '\n') {
          // 已經讀完 Date 標頭這一整行，提前結束等待迴圈
          waitTime = 0;
          break;
        }
        if (getDatetime.indexOf("Date:")!=-1) {
          // 剛好讀到 "Date:" 字樣，清空緩衝區重新開始收集，準備接收標頭的實際數值部分
          getDatetime = "";
          getStatus = true;
        }
        else
          getDatetime += String(c);

        startTime = millis();  // 每收到一個字元就重置逾時計時，避免資料傳輸中斷造成誤判逾時
      }
    }
    
    client.stop();
    
  } else {
    // 連線失敗（例如網路不通），改用文字提示，讓呼叫端的 LLM 之後改用搜尋工具自行查詢目前時間
    getDatetime = "Use grounded search to retrieve the current GMT date and time.";
  }

  if (getDatetime == "") {
    getDatetime = "Use grounded search to retrieve the current GMT date and time.";
  }

  return getDatetime;
  
}

// ============================================================
// 【工具函式】將硬體 RTC 目前的時間讀出並格式化成字串
// filename 參數決定輸出格式：
//   true  -> 適合當檔名的格式（不含斜線冒號），例如 20260807153045
//   false -> 一般可讀格式，例如 2026/8/7 15:30:45
// ============================================================
String getRtcTimeString(bool filename = false) {

  long long epoch = rtc.Read();  // 從硬體 RTC 讀出目前的 Unix epoch 時間戳記

  time_t rawtime = (time_t)epoch;

  struct tm *timeinfo = localtime(&rawtime);  // 轉換成年月日時分秒結構

  char buffer[32];

  if (filename == true) {
	  sprintf(
		buffer,
		"%04d%d%d%02d%02d%02d",
		timeinfo->tm_year + 1900,  // tm_year 是從 1900 年起算的偏移，需要加回 1900
		timeinfo->tm_mon + 1,      // tm_mon 是 0-11，需要加 1 才是一般認知的月份
		timeinfo->tm_mday,
		timeinfo->tm_hour,
		timeinfo->tm_min,
		timeinfo->tm_sec
	  );
  }
  else {
	  sprintf(
		buffer,
		"%04d/%d/%d %02d:%02d:%02d",
		timeinfo->tm_year + 1900,
		timeinfo->tm_mon + 1,
		timeinfo->tm_mday,
		timeinfo->tm_hour,
		timeinfo->tm_min,
		timeinfo->tm_sec
	  );
  }

  return String(buffer);
}

// ============================================================
// 【核心函式】RTC 初始校時流程
// 用途：開機時（或使用者呼叫 /syncrtc）執行的完整校時流程：
//   1. 呼叫 getGoogleDatetime() 取得不依賴任何 LLM 供應商的 GMT 時間字串
//   2. 組裝一段 prompt，請目前設定的 LLM（Gemini/OpenAI/Grok 三選一）
//      把 GMT 時間換算成裝置設定的 timeZone，並額外補償 4 秒的網路傳輸延遲
//   3. 要求 LLM「只能」輸出乾淨的 JSON（不含 markdown、不含說明文字），方便直接解析
//   4. 解析成功後寫入硬體 RTC
// workName 參數：紀錄這次呼叫是從哪個管道（Telegram/MQTT/排程等）觸發的，方便錯誤時回覆使用者
// ============================================================
void rtcInitialTime(String workName) {
	
  rtcUpdateStatus = true;  // 標記「校時進行中」，避免其他流程誤用尚未更新完成的 RTC 值
  
  // 組裝要求 LLM 進行時區換算與格式化輸出的完整 prompt
  String prompt =
    "Convert this GMT datetime to " + timeZone + ".\n"
    "GMT datetime: " + getGoogleDatetime() + "\n\n"
	
    "Before generating the JSON output, add exactly 4 seconds to the converted local datetime.\n"
    "Handle minute, hour, day, month, and year rollovers correctly if needed.\n\n" 

    "Output requirements:\n"
    "- Return ONLY a raw JSON object.\n"
    "- Do NOT use markdown.\n"
    "- Do NOT use code fences.\n"
    "- Do NOT explain anything.\n"
    "- Do NOT add extra text.\n"
    "- First character must be {.\n"
    "- Last character must be }.\n\n"

    "Required JSON format:\n"
    "{\n"
    "\"rtcYear\":2026,\n"
    "\"rtcMonth\":5,\n"
    "\"rtcDay\":28,\n"
    "\"rtcHour\":11,\n"
    "\"rtcMinute\":35,\n"
    "\"rtcSecond\":0\n"
    "}";

  // 呼叫多供應商統一入口 llmChatRequest()，tools 參數傳 -1 代表「角色限定模式」：
  // 不夾帶完整的工具（tool_call）定義，也不帶對話歷史，純粹只是一次單輪、無狀態的格式轉換請求，
  // 這樣可以把 token 用量壓到最低，也避免工具路由邏輯干擾這種單純的格式轉換任務。
  String message = llmChatRequest(String(taskTags[1]), prompt, -1);

  // 驗證回應是否為合法的 JSON 物件外觀（頭尾符合 { }），才進一步嘗試解析
  if (message.startsWith("{") && message.endsWith("}")) {

    DynamicJsonDocument doc(1024);
    DeserializationError error = deserializeJson(doc, message);

    if (error) {
      // JSON 語法錯誤（例如 LLM 意外夾帶了說明文字），記錄除錯訊息並中止本次校時
      Serial.println("[DEBUG] JSON parse failed\n" + message);
      replyUserMessage(workName, "RTC time update failed.", "");
      rtcUpdateStatus = false;
      return;
    }

    JsonObject obj = doc.as<JsonObject>();

    // 逐欄位讀取，若欄位不存在則用 | 0 提供預設值 0，避免解析失敗造成未定義行為
    rtcYear   = obj["rtcYear"]   | 0;
    rtcMonth  = obj["rtcMonth"]  | 0;
    rtcDay    = obj["rtcDay"]    | 0;
    rtcHour   = obj["rtcHour"]   | 0;
    rtcMinute = obj["rtcMinute"] | 0;
    rtcSecond = obj["rtcSecond"] | 0;

  } else {
    // 回應開頭/結尾不符合 JSON 物件外觀，直接視為失敗
    Serial.println("[DEBUG] JSON parse failed : (rtcInitialTime)\n" + message);
    replyUserMessage(workName, "RTC time update failed.", "");
  }

  // 將解析出來的年月日時分秒寫入硬體 RTC 晶片
  rtc.Init();
  long long initTime = rtc.SetEpoch(rtcYear, rtcMonth, rtcDay, rtcHour, rtcMinute, rtcSecond);
  rtc.Write(initTime);
}

// ============================================================
// 【核心函式】發送 Telegram 文字訊息
// 用途：透過 Telegram Bot API 的 sendMessage 端點，把 text 內容送給指定 chatid。
// keyboard 參數可選填，用來附上客製化的 Reply Keyboard JSON（讓使用者看到快捷按鈕）。
// 回傳值為 Telegram API 回應的 body 內容（成功或錯誤訊息皆包含在內，供上層判斷）。
// ============================================================
String telegramSendMessage(String token, String chatid, String text, String keyboard = "") {
  text.replace("\\n", "%0A");  // 把字面上的 \n 換行符號轉成 URL 編碼形式的換行，確保 Telegram 端正確斷行
  const char* myDomain = "api.telegram.org";
  String getAll="", getBody = "";
  // 組裝 application/x-www-form-urlencoded 格式的請求內容
  String request = "parse_mode=HTML&chat_id="+chatid+"&text="+text;

  if (keyboard!="")
    request += "&reply_markup="+keyboard;  // 有指定客製鍵盤才附加此參數

  WiFiSSLClient client;
  
  if (client.connect(myDomain, 443)) {
    // 手動組裝 HTTP POST 請求（Arduino 環境沒有高階 HTTP client 函式庫，需自行處理標頭與內容）
    client.println("POST /bot"+token+"/sendMessage HTTP/1.1");
    client.println("Host: " + String(myDomain));
    client.println("Content-Length: " + String(request.length()));
    client.println("Content-Type: application/x-www-form-urlencoded");
    client.println("Connection: close");
    client.println();
    client.print(request);

    int waitTime = 5000;  // 最長等待 5 秒回應
    unsigned long startTime = millis();
    bool state = false;  // 是否已經跳過 HTTP 標頭，開始收集回應 body 的旗標

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);
	  
      while (client.available())  {
        char c = client.read();

        if (state)
          getBody += String(c);  // 已進入 body 區段，逐字元累加

        if (c == '\n')  {
          // 遇到空白行（連續兩個換行）代表 HTTP 標頭區段結束，接下來就是 body 內容
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);  // 累積目前這一行的內容，用來判斷是否為空白行

        startTime = millis();  // 每收到資料就重置逾時計時
      }

      if (getBody.length()>0)
        break;  // 已經拿到部分 body 內容就提前跳出，加速回應（不強求把整個回應讀完）
    }
    client.stop();
  }
  else {
    getBody="Connected to api.telegram.org failed.";  // 連線失敗，回傳明確的錯誤訊息
  }
  
  return getBody;
}

// 用途：透過 LINE Messaging API 的 push 端點推播文字訊息給指定使用者/群組。

  // 組裝 LINE push message 所需的 JSON 請求內容
  String request = "{\"to\":\""+targetId+"\",\"messages\":[{\"type\":\"text\",\"text\":\""+message+"\"}]}";
	
  const char* myDomain = "api.line.me";

  WiFiSSLClient client;

  if (client.connect(myDomain, 443)) {
    client.println("POST /v2/bot/message/push HTTP/1.1");

    client.println("Host: api.line.me");

    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();
    client.println(request);
    client.println();
	
    boolean state = false;
    long startTime = millis();
	while ((startTime + 3000) > millis()) {  // 最長等待 3 秒
		while (client.available()) {
			char c = client.read();
			if (c == '\n') {
				if (getAll.length()==0) 
					state=true;
				getAll = "";

			break;
	}
	client.stop();
  }
  else {
    getBody="Connected to api.line.me failed.";
  }
  
  return getBody;

// ============================================================
// 【核心函式】將目前擷取到的相機影像傳送到 Telegram
// 用途：透過 sendPhoto 端點，以 multipart/form-data 格式上傳 JPEG 影像。
// frames 參數決定是否要「重新擷取一張新影像」：
//   true  -> 立即從相機重新抓一張新的影格
//   false -> 沿用「上一次」已經存在緩衝區裡的影像（例如 /still 之後接著 /vision 分析同一張）
// 因為影像緩衝區是全域共用資源，函式一開始就用 imageMutex 上鎖，避免多工同時存取造成資料錯亂或崩潰。
// ============================================================
String telegramSendCapturedImage(String token, String chat_id, bool frames) {
  const char* myDomain = "api.telegram.org";
  String getAll="", getBody = "";
  WiFiSSLClient client;

  // 嘗試取得影像緩衝區的互斥鎖，逾時（10 秒）仍拿不到就直接放棄，避免無限期卡住整個 Task
  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    return "Image buffer busy, please try again.";
  }

  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);  // 從相機硬體重新擷取一張影格，寫入全域緩衝區位址與長度
    else if (!frames && imageLength == 0) {
      // 使用者要求沿用舊影像，但緩衝區裡根本沒有任何影像資料，直接回錯誤避免傳送空內容
      client.stop();
      xSemaphoreGive(imageMutex);
      return "Previous image does not exist";
    }

    uint8_t *fbBuf = (uint8_t*)imageAddress;  // 影像資料的記憶體起始位址（轉型為位元組指標）
    size_t fbLen = imageLength;

    // multipart/form-data 格式需要的 boundary 前後包裝字串（"Taiwan" 是自訂的分界字串）
    String head =
      "--Taiwan\r\nContent-Disposition: form-data; name=\"chat_id\"; \r\n\r\n"
      + chat_id +
      "\r\n--Taiwan\r\nContent-Disposition: form-data; name=\"photo\"; filename=\"capture.jpg\"\r\nContent-Type: image/jpeg\r\n\r\n";

    String tail = "\r\n--Taiwan--\r\n";

    size_t imageLen = imageLength;
    size_t extraLen = head.length() + tail.length();
    size_t totalLen = imageLen + extraLen;  // Content-Length 必須精確等於整個 multipart body 的位元組數

    client.println("POST /bot"+token+"/sendPhoto HTTP/1.1");
    client.println("Host: " + String(myDomain));
    client.println("Content-Length: " + String(totalLen));
    client.println("Content-Type: multipart/form-data; boundary=Taiwan");
    client.println();

    client.print(head);

    // 分批傳送影像二進位資料，每次最多 1024 bytes，
    // 避免一次配置過大的傳輸緩衝區造成記憶體壓力（嵌入式環境 RAM 有限）
    for (size_t n=0;n<fbLen;n=n+1024) {
      if (n+1024<fbLen) {
        client.write(fbBuf, 1024);
        fbBuf += 1024;  // 指標前移，指向下一批資料的起點
      }
      else {
        // 最後一批可能不足 1024 bytes，只傳送剩餘的實際長度
        size_t remainder = fbLen - n;
        if (remainder > 0)
          client.write(fbBuf, remainder);
      }
    }

    client.print(tail);

    int waitTime = 10000;  // 影像上傳耗時較長，等待逾時拉長到 10 秒
    unsigned long startTime = millis();
    bool state = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (state)
          getBody += String(c);

        if (c == '\n') {
          if (getAll.length()==0)
            state=true;
          getAll = "";
        }
        else if (c != '\r')
          getAll += String(c);

        startTime = millis();
      }

      if (getBody.length()>0)
        break;
    }

    client.stop();
    Serial.println();

  } else {
    getBody="Connected to api.telegram.org failed.";
    Serial.println("Connected to api.telegram.org failed.");
  }

  xSemaphoreGive(imageMutex);  // 無論成功或失敗，函式結束前都要釋放互斥鎖，避免其他 Task 永久卡住

  return getBody;
}

// ============================================================
// 【工具函式】從 LLM 回應文字中移除時間戳記與內部標籤
// 用途：LLM 的回應歷史中可能夾帶了 workId、時間戳記字串、或內部事件標籤（taskTags），
// 這些是系統內部用來追蹤事件來源的資訊，不該原封不動顯示給使用者看，
// 這裡負責在「最終回覆使用者」前把它們清乾淨。
// ============================================================
String removeTimestamps(String workId, String timestamps, String text) {

    if (text.length() == 0 || text.startsWith("NONE")) return "NONE";  // 空字串或哨兵值 "NONE" 直接原樣回傳，代表無需再處理
	
    if (text.indexOf("&lt;") == -1 && text.indexOf("<") == -1) return text;  // 文字裡完全沒有任何角括號標籤，可以提早結束，省去不必要的字串處理

    // 先把 HTML 實體字元還原成一般角括號，才能正確比對後面的標籤字串
    text.replace("&lt;", "<");
    text.replace("&gt;", ">");

    text.replace(timestamps, "");  // 移除具體的時間戳記字串
    text.replace(workId, "");      // 移除具體的 workId 字串

    // 逐一檢查每一種事件標籤，若文字中出現任何一個標籤，
    // 就把該標籤（以及它之後的所有內容）截斷，只保留標籤之前的乾淨文字
    for (const auto& tag : taskTags) {
        int pos = text.indexOf(tag);
        if (pos != -1)
            text = text.substring(0, pos);
    }

    return text;
}

// ============================================================
// 【核心函式】回覆訊息給使用者(依來源管道分流)
// 用途:統一的「對外輸出」出口。所有工具執行結果、LLM 回應要送給使用者時都經過這裡。
// 依 workId 開頭的標籤判斷來源管道:
//   來自網頁介面(<PAGE>)-> 直接把文字累加進 mainPageHTML,等待 HTTP 回應時一併回傳
//   其他管道(Telegram 等)-> 呼叫 telegramSendMessage() 主動推播訊息
// ============================================================
void replyUserMessage(String workId, String text, String keyboard = "") {
	if (text.length() == 0 || text.startsWith("NONE")) return;  // 空字串或哨兵值 "NONE" 代表沒有內容要回覆,直接跳過
	
	if (workId.startsWith(String(taskTags[0])))  // taskTags[0] = "<PAGE>",判斷是否來自網頁介面
		mainPageHTML += text +"\n";
	else
		telegramSendMessage(telegrambotToken, telegrambotChatId, text, keyboard);
}

// ============================================================
// 【核心函式】回覆影像給使用者(依來源管道分流)
// 用途:與 replyUserMessage 概念相同,但處理的是影像資料。
// 網頁介面:把影像轉成 Base64 編碼的 <img> 標籤字串,直接內嵌進 HTML 回應中(不需額外檔案伺服器)
// Telegram:改用 telegramSendCapturedImage() 以 multipart 上傳原始 JPEG 二進位資料
// frames 參數:是否要重新從相機擷取新影格(否則沿用緩衝區裡既有的影像)
// ============================================================
String replyUserImage(String workId, bool frames) {
  if (workId.startsWith(String(taskTags[0]))) {  // 來自網頁介面

      // 影像緩衝區是全域共用資源,取用前先上鎖,逾時未取得直接放棄回傳錯誤,避免無限期卡住
      if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
        return "Image buffer busy, please try again.";
      }

      if (frames)
          Camera.getImage(0, &imageAddress, &imageLength);  // 重新擷取一張新影格

      if (imageLength == 0) {
        // 沒有指定要重新擷取,但緩衝區裡也沒有舊影像資料可用
        xSemaphoreGive(imageMutex);
        return "Previous image does not exist";
      }

      uint8_t* fbBuf = (uint8_t*)imageAddress;
      size_t   fbLen = imageLength;

      char *input = (char *)fbBuf;
      char output[base64_enc_len(3)];  // Base64 編碼固定以 3 bytes 一組轉成 4 個字元
                  
      // 預估編碼後字串的長度(Base64 每 3 bytes 膨脹成 4 字元),預先配置記憶體避免多次重新配置
      size_t estimatedSize = 23 + ((fbLen + 2) / 3) * 4 + 1;
      String imageFile = "<img src='data:image/jpeg;base64,";
      imageFile.reserve(estimatedSize);
      
      // 每次取 3 bytes 原始資料進行 Base64 編碼,逐段累加成完整的編碼字串
      for (size_t i = 0; i < fbLen; i += 3) {
          base64_encode(output, input, 3);
          input += 3;
          imageFile += String(output);
      }

      // 寫入全域的 mainPageHTML 前,用 stateMutex 保護,避免與其他 Task 同時寫入造成資料錯亂
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        mainPageHTML = imageFile + "' style='max-width:240px; height:auto; border-radius:8px;'><br>";
        xSemaphoreGive(stateMutex);
      }

      xSemaphoreGive(imageMutex);

	  return "Image file created.";
  }
  else
    return telegramSendCapturedImage(telegrambotToken, telegrambotChatId, frames);  // 非網頁管道,走 Telegram 上傳流程

  return "";
}

// ============================================================
// 【工具函式】從 SD 卡讀取整個檔案內容為字串
// 用途:通用的檔案讀取函式,soul.md / device.md / skill.md / memory.md / schedule.json 等
// 各種設定檔的載入都經過這裡。全程用 sdMutex 保護,避免與其他 Task 同時存取 SD 卡造成衝突。
// ============================================================
String getStringFromFile(String fileNname) {
  String data = "";

  // 嘗試取得 SD 卡存取的互斥鎖,逾時就放棄本次讀取(寧可讀取失敗,也不要讓多工同時碰 SD 卡)
  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] getStringFromFile: failed to take sdMutex");
    return "";
  }

  if (!fs.begin()) {
    Serial.println("Card Mount Failed");
    if (sdMutex) xSemaphoreGive(sdMutex);
    return "";
  }

  String path = String(fs.getRootPath()) + "/" + fileNname;

  file = fs.open(path);

  if (file) {
    uint32_t len = file.size();
    char *buf = (char*)malloc(len + 1);  // 動態配置剛好容納整個檔案內容 + 結尾 null 字元的緩衝區

    if (buf) {
      file.read(buf, len);
      buf[len] = '\0';  // 手動補上字串結尾字元,malloc 出來的記憶體不會自動初始化
      data = String(buf);
      free(buf);  // 轉成 String 物件後,原始緩衝區已無用處,立即釋放,避免記憶體佔用
    }

    file.close();
  }

  fs.end();

  if (sdMutex) xSemaphoreGive(sdMutex);
  
  return data;
}

// ============================================================
// 【核心函式】將資料寫入 SD 卡檔案(附帶自動備份機制)
// 用途:通用的檔案寫入函式,採用「兩段式改名再寫入」策略,避免斷電或寫入中斷造成資料損毀:
//   1. 若目前檔案已存在,先把舊檔改名成 .bak 備份(若已有舊備份則先刪除)
//   2. 再建立新檔案並寫入新內容
// 這樣即使寫入過程中斷電,至少還保有前一版的 .bak 備份可供救援,不會兩份都毀損。
// timestamp 參數:若為 true,備份檔名會加上 RTC 時間戳記,適合用於需要保留歷史版本的場合(如 memory.md)。
// ============================================================
void storeDataToFile(String filename, String data, bool timestamp = false) {

  if (sdMutex && xSemaphoreTake(sdMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    Serial.println("[DEBUG] storeDataToFile: failed to take sdMutex");
    return;
  }
  
  if (!fs.begin()) {
    Serial.println("Card Mount Failed");
    if (sdMutex) xSemaphoreGive(sdMutex);
    return;
  }
  
  String file_path = String(fs.getRootPath());
  String currentFile = file_path + "/" + filename;
  
  String backupFile = currentFile + ".bak";
  if (timestamp == true)
	  backupFile = currentFile + "_" + getRtcTimeString(true) + ".bak";  // 帶時間戳記的備份檔名,方便保留多個歷史版本
  
  if (fs.exists(currentFile)) {
    
    if (fs.exists(backupFile)) {
      
      fs.remove(backupFile);  // 若已存在同名備份(通常是非時間戳記模式),先刪除舊備份
    }
    delay(100);  // 短暫延遲,讓檔案系統操作有時間完成,降低連續操作互相干擾的風險
    
    fs.rename(currentFile, backupFile);  // 把現有檔案改名為備份檔,騰出原檔名給新內容使用
  }
      
  file = fs.open(currentFile);  // 以新檔名開啟(不存在則建立新檔)
  
  if (file) {
    
    file.println(data.c_str());
    file.close();
  }
  else
	  Serial.println("[DEBUG] File open failed: " + currentFile);
  
  fs.end();

  if (sdMutex) xSemaphoreGive(sdMutex);
}

// ============================================================
// 【工具函式】透過 TCP/HTTP 呼叫另一台裝置的 /message 端點
// 用途:實現「Agent 對 Agent」的多裝置協作通訊——本裝置可以主動把訊息傳給區網內另一台
// fuClaw(或任何實作相容端點的裝置),讓多台裝置之間能互相協調、通報事件。
// domain 參數為目標裝置的網域/IP,request 為要傳遞的訊息內容(會先 URL 編碼再送出)。
// ============================================================
String tcpSendMessage(String workId, String domain, String request) {
  
  WiFiClient client;
  
  if (client.connect(domain.c_str(), 81)) {  // 固定連往對方裝置的 81 port(主要 HTTP 設定伺服器)
	  
    client.setTimeout(20000);
	
    client.println("GET /message?" + urlencode(request) + " HTTP/1.1");
    client.println("Host: " + domain);
    client.println("Access-Control-Allow-Origin: *");
    client.println("Content-Length: 0");
    client.println();
    
    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    // 逐字元讀取回應,先跳過 HTTP 標頭區段,遇到空白行(headersEnded)後才開始收集 body 內容
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;  // 空白行代表標頭區段結束
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;  // 每收到 body 資料就重置逾時,避免資料傳輸中斷被誤判逾時
        }
      }
      vTaskDelay(1);  // 讓出極短時間片給其他 Task,避免忙等待完全佔用 CPU
    }
    
    client.stop();  
    
    // 成功連線,不論對方實際回應內容為何,一律包成統一的成功格式 JSON 回傳給呼叫端(通常是 executeTool)
    return 
      "{\"status\":\"success\","
      "\"method\":\"/tcpSendMessage\","
      "\"response\":\"" + body + "\","   
      "\"workId\":\"" + workId + "\"}"; 
  }

  // 連線失敗,回傳統一格式的錯誤 JSON
  return 
    "{\"status\":\"error\","
    "\"method\":\"/tcpSendMessage\","       
    "\"reason\":\"Connected to the device failed.\","
    "\"workId\":\"" + workId + "\"}";
      
}

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {

// ============================================================
// 【核心函式】Gemini 聊天請求實作(供 llmChatRequest() 派發呼叫)
// 用途:這是三家供應商(Gemini/OpenAI/Grok)中，Gemini 專屬的聊天請求實作，
// 也是整個多供應商架構的「範本」——後面的 openaiChatRequest()／grokChatRequest()
// 遵循幾乎相同的流程骨架，只是端點網址、請求 JSON 格式、回應解析路徑不同。
//
// 整體流程：
//   1. 把目前的 RTC 時間附加進訊息內容，讓 LLM 掌握「現在幾點」
//   2. 把這則使用者訊息（含時間戳記標籤）累加進全域對話歷史 historicalMessages
//   3. 依 tools 參數決定要用哪一種 system prompt（含工具／不含工具／純角色）
//   4. 組裝 Gemini generateContent API 所需的 JSON 請求本文
//   5. 手動建立 HTTPS 連線、送出 POST 請求、逐字元讀取回應
//   6. 解析回應 JSON，取出 LLM 產生的文字內容
//   7. 清除回應中的內部時間戳記標籤，並把回應也累加進對話歷史（讓下一輪對話延續上下文）
//
// tools 參數的三種模式：
//   0  -> 不含工具定義的 system prompt（單純聊天問答，不觸發硬體工具呼叫）
//   1  -> 含完整工具定義的 system prompt（允許 LLM 輸出 tool_call JSON 操作硬體）
//   -1 -> 「角色限定」模式（不帶對話歷史，也不帶工具定義，用於單次、無狀態的格式轉換任務，
//         例如 rtcInitialTime() 的時區換算請求）
// ============================================================
String geminiChatRequest(String workId, String message, int tools = 1) {
  String timestamps = "\n" + workId;  // 內部用時間戳記/來源標籤，之後會從最終回覆中移除

  // 附加目前 RTC 時間，讓 LLM 在推理時知道現在的實際時間（例如判斷排程是否該觸發、回答「現在幾點」等問題）
  message = message + "\n\nRTC current time: " + getRtcTimeString();

  // 把這輪使用者輸入累加進全域對話歷史，用 stateMutex 保護避免多工同時寫入造成競爭
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  // 依 tools 參數組裝實際要送出的 contents（system prompt + 對話歷史，或單次訊息）
  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    if (tools == 0)
      contents = systemContentNoTools + historicalMessages;  // 不含工具，但帶完整對話歷史
    else if (tools == 1)
      contents = systemContentTools + historicalMessages;    // 含工具定義，帶完整對話歷史（一般工作流程走這條）
    else if (tools == 2)
      contents = systemContent + buildLlmMessage("user", message);  // 純角色 + 單則訊息（不含歷史）
    else
      contents = systemContent + buildLlmMessage("user", message);  // tools == -1（或其他未列舉值）走相同的單次無狀態模式
    xSemaphoreGive(stateMutex);
  }

  // 組裝 Gemini generateContent API 所需的完整請求 JSON
  String request = "{\"contents\": [" + contents +
                   "],\"generationConfig\": {\"maxOutputTokens\": " +
                   llmMaxOutputTokens +
                   ", \"temperature\": " + llmTemperature + "}}";

  WiFiSSLClient client;
  String responseText = "";
	  
  client.setTimeout(10000);
		  
  if (client.connect("generativelanguage.googleapis.com", 443)) {

    // Gemini API 金鑰是以 URL query string 方式帶入（?key=...），而非 Authorization 標頭
    client.println("POST /v1beta/models/"+llmModel+":generateContent?key="+llmKey+" HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();
    
    // 分段送出請求本文，每次最多 1024 bytes，避免單次傳送過大資料造成緩衝區問題
    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    // 逐字元讀取回應：先跳過 HTTP 標頭，再收集 body 內容
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) { 
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } 
        else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }
    
    client.stop();

    // Gemini 回應可能是 chunked 編碼，body 開頭常夾帶區塊長度等雜訊，
    // 因此找出第一個 '{' 的位置，從那裡開始才是真正的 JSON 內容
    int jsonStart = body.indexOf('{'); 
    if (jsonStart != -1) { 
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      // 完全沒有收到任何回應內容（例如逾時），視為本輪工作流程結束，回傳哨兵值 "NONE"
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    // 解析 JSON 回應，取出實際的文字內容
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      // JSON 語法解析失敗（可能是回應被截斷或格式異常）
      Serial.println("[DEBUG] JSON parse failed: (geminiChatRequest)\n" + body);
      responseText = "JSON parse failed (geminiChatRequest). Please try again.";
    }  
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      // Gemini 正常回應的資料結構：candidates[0].content.parts[0].text
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    } 
    else if (doc["error"]) {
      // Gemini API 回傳明確的錯誤物件（例如金鑰無效、額度用盡等）
      responseText = "[DEBUG] Gemini API Error: " + doc["error"]["message"].as<String>();
	  Serial.println(responseText);
	  responseText = "Gemini API Error";
    } 
    else {
      // 回應格式不符預期的任何結構，屬於未知狀況
      responseText = "Unexpected response from Gemini.";
      Serial.println("Unknown response format.");
    }

  } else {
    Serial.println("Failed to connect to Gemini API");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "Gemini did not respond. Please try again.";
  }
  
  // 清除回應文字中可能夾帶的內部標籤/時間戳記，才是真正要呈現給使用者的乾淨內容
  responseText = removeTimestamps(workId, timestamps, responseText);

  // 把 LLM 的回應（含時間戳記標籤，方便下次比對移除）累加進對話歷史，讓後續對話能延續上下文
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
  
}

// ============================================================
// 【核心函式】Gemini「含即時搜尋」請求實作
// 用途：與 geminiChatRequest() 幾乎相同的流程，差別只在於請求 JSON 多帶了
// "tools": [{"google_search": {}}] 參數，啟用 Gemini 內建的 Google 即時搜尋能力，
// 讓 LLM 可以查詢訓練資料截止後的最新資訊（例如即時新聞、天氣、股價等）再回答。
// 對應 /search 工具呼叫時使用，與一般聊天請求（llmChatRequest）分開，
// 避免每次一般對話都額外耗用搜尋功能的成本與延遲。
// ============================================================
String geminiSearchRequest(String workId, String message, int tools = 1) {

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {

    else if (tools == 0)
      contents = systemContentNoTools + historicalMessages;  // 不含工具定義 + 完整歷史
    xSemaphoreGive(stateMutex);
  }

  // 與 geminiChatRequest 的關鍵差異：多帶了 google_search 工具宣告，啟用即時網路搜尋
  String request = "{\"contents\": [" + contents +
                   "],\"tools\": [{\"google_search\": {}}],\"generationConfig\": {\"maxOutputTokens\": " +
                   llmMaxOutputTokens +
                   ", \"temperature\": " + llmTemperature + "}}";

  WiFiSSLClient client;
  String responseText = "";
	  
  client.setTimeout(10000);
	
  if (client.connect("generativelanguage.googleapis.com", 443)) {

    client.println("POST /v1beta/models/"+llmModel+":generateContent?key="+llmKey+" HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: generativelanguage.googleapis.com");
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();
    
    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    // 以下讀取回應、解析 JSON 的流程與 geminiChatRequest 完全相同
    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }
    
    client.stop();  

    int jsonStart = body.indexOf('{'); 
    if (jsonStart != -1) { 
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (geminiSearchRequest)\n" + body);
      responseText = "JSON parse failed (geminiSearchRequest). Please try again.";
    } 
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    } 
    else if (doc["error"]) {
      responseText = "[DEBUG] Gemini API (Search) Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
	  responseText = "Gemini API (Search) Error";
    } 
    else {
      responseText = "Unexpected response from Gemini Search.";
    }

  } else {
    Serial.println("Failed to connect to Gemini API (Search)");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "Gemini Search did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// ============================================================
// 【核心函式】Gemini 多模態視覺分析請求實作
// 用途：把相機擷取到的 JPEG 影像以 Base64 編碼後，連同文字提示一起送給 Gemini 進行圖像理解，
// 對應 /vision 工具呼叫。與聊天/搜尋請求最大的不同：
//   - 這是「無狀態」的單次呼叫（不帶對話歷史 historicalMessages，只有一句 message + 一張圖）
//   - 需要先鎖定 imageMutex，確保讀取影像緩衝區時不會被其他 Task 同時修改
//   - Base64 編碼過程會佔用可觀的 CPU 與暫存記憶體，因此編碼完成、組好請求 JSON 後
//     立刻釋放 imageMutex（xSemaphoreGive），不用等到整個 HTTP 請求結束才放鎖，
//     讓其他需要用相機的流程能盡快接續執行
// frames 參數：是否要重新擷取新影格，或沿用緩衝區裡既有的影像
// ============================================================
String geminiVisionRequest(String workId, String message, bool frames = true) {
  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  // 仍然把這次視覺分析的請求文字記錄進對話歷史，方便後續對話能引用「剛才看到了什麼」
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  WiFiSSLClient client;
  String responseText = "";
  const char* myDomain = "generativelanguage.googleapis.com";
	  
  client.setTimeout(10000);

  // 影像緩衝區為全域共用資源，取用前先上鎖，逾時就直接放棄本次視覺分析
  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    responseText = "Image buffer busy, please try again.";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      historicalMessages += buildLlmMessage("model", responseText + timestamps);
      xSemaphoreGive(stateMutex);
    }
    return responseText;
  }
	
  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);  // 重新擷取新影格
    else if (!frames && imageLength == 0) {
      // 要求沿用舊影像，但緩衝區是空的
      client.stop();
      xSemaphoreGive(imageMutex);

      responseText = "Previous image does not exist";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", responseText + timestamps);
        xSemaphoreGive(stateMutex);
      }

      return responseText;
    }
    
    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    char *input = (char *)fbBuf;
    char output[base64_enc_len(3)];
    String imageFile = "";

    // 逐 3 bytes 進行 Base64 編碼，累加成完整的編碼字串（Gemini inline_data 需要純 Base64，不含 data: 前綴）
    for (size_t i = 0; i < fbLen; i += 3) {
      base64_encode(output, input, 3);
      input += 3;
      imageFile += String(output);
    }

    // 組裝 Gemini 多模態請求：parts 陣列裡同時包含「文字提示」與「inline_data 圖片」兩個元素
    String Data = "{\"contents\": [{\"parts\": [{\"text\": \"" + message + 
                  "\"}, {\"inline_data\": {\"mime_type\":\"image/jpeg\",\"data\":\"" + 
                  imageFile + "\"}}]}]}";

    // 請求 JSON 已經組裝完成，不再需要存取原始影像緩衝區，提早釋放鎖，讓其他流程可以盡快使用相機
    xSemaphoreGive(imageMutex);

    client.println("POST /v1beta/models/"+llmModel+":generateContent?key="+llmKey+" HTTP/1.0");
    client.println("Host: " + String(myDomain));
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(Data.length()));
    client.println("Connection: close");
    client.println();
    
    for (size_t i = 0; i < Data.length(); i += 1024) {
      client.print(Data.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }
    
    client.stop();   

    int jsonStart = body.indexOf('{'); 
    if (jsonStart != -1) { 
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed (geminiVisionRequest):\n" + body);
      responseText = "JSON parse failed (geminiVisionRequest). Please try again.";
    } 
    else if (doc["candidates"][0]["content"]["parts"][0]["text"]) {
      responseText = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    } 
    else if (doc["error"]) {
      responseText = "[DEBUG] Gemini API (Vision) Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
	  responseText = "Gemini API (Vision) Error";
    } 
    else {
      responseText = "Unexpected response from Gemini Vision.";
    }

  } else {
    Serial.println("Failed to connect to Gemini API (Vision)");
    responseText = "Connection failed";
    xSemaphoreGive(imageMutex);  // 連線失敗也要記得釋放鎖，避免其他 Task 永久卡住
  }

  if (responseText == "") {
    responseText = "Gemini Vision did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;
}

// ============================================================
// 【核心函式】OpenAI 聊天請求實作(供 llmChatRequest() 派發呼叫)
// 用途：與 geminiChatRequest() 功能完全對應，但改用 OpenAI 的 Chat Completions API 格式。
// 與 Gemini 版本的關鍵差異：
//   - 端點：api.openai.com/v1/chat/completions（Gemini 是 generateContent）
//   - 驗證方式：Authorization: Bearer <llmKey> 標頭（Gemini 是把金鑰放在 URL query string）
//   - 請求欄位：{"model":..., "messages":[...]}（Gemini 是 {"contents":[...]}）
//   - Token 上限欄位名稱：max_completion_tokens（Gemini 是 maxOutputTokens）
//   - 回應解析路徑：choices[0].message.content（Gemini 是 candidates[0].content.parts[0].text）
// 其餘流程（附加 RTC 時間、寫入對話歷史、tools 參數決定 system prompt 版本、
// 手動組 HTTP 請求、逐字元讀取回應、清除時間戳記後寫回歷史）與 Gemini 版本完全相同。

    else if (tools == 1)
      contents = systemContentTools + historicalMessages;
    else if (tools == 2)
      contents = systemContent + buildLlmMessage("user", message);
    else
      contents = systemContent + buildLlmMessage("user", message);
    xSemaphoreGive(stateMutex);
  }

  // OpenAI Chat Completions 格式：model + messages 陣列 + max_completion_tokens + temperature
  String request = "{\"model\":\""+llmModel+"\",\"messages\":[" + contents + "], \"max_completion_tokens\": " + llmMaxOutputTokens + ", \"temperature\": " + llmTemperature + "}";

  WiFiSSLClient client;
  String responseText = "";

  client.setTimeout(10000);

  if (client.connect("api.openai.com", 443)) {

    client.println("POST /v1/chat/completions HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: api.openai.com");
    client.println("Authorization: Bearer " + llmKey);  // OpenAI 慣用 Bearer Token 驗證方式，與 Gemini 的 URL 帶金鑰不同
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        }
        else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {

        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (openaiChatRequest)\n" + body);
      responseText = "JSON parse failed (openaiChatRequest). Please try again.";
    }
    else if (doc["choices"][0]["message"]["content"]) {
      // OpenAI Chat Completions 的回應結構：choices[0].message.content（與 Gemini 的路徑不同）
      responseText = doc["choices"][0]["message"]["content"].as<String>();
    }
    else if (doc["error"]) {
      responseText = "[DEBUG] OpenAI API Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "OpenAI API Error";
    }
    else {
      responseText = "Unexpected response from OpenAI.";
      Serial.println("Unknown response format.");
    }

  } else {
    Serial.println("Failed to connect to OpenAI API");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "OpenAI did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

// ============================================================
// 【核心函式】OpenAI「含即時搜尋」請求實作
// 用途：對應 Gemini 版的 geminiSearchRequest()，但 OpenAI 走的是不同的 API 家族——
// 使用 /v1/responses 端點（Responses API），而非 /v1/chat/completions，
// 並在請求中宣告 {"type": "web_search"} 工具啟用即時網路搜尋。
// 回應結構也隨之不同：Responses API 回傳的是 output 陣列，
// 需要先篩選出 type=="message" 的項目，再從其 content 陣列中挑出 type=="output_text" 的片段組合成完整文字，
// 不像 Chat Completions 那樣可以直接用固定路徑一次取到文字。
// ============================================================
String openaiSearchRequest(String workId, String message, int tools = 1) {

  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    if (tools == 0)
      contents = systemContentNoTools + historicalMessages;
    else if (tools == 1)
      contents = systemContentTools + historicalMessages;
    else if (tools == 2)
      contents = systemContent + buildLlmMessage("user", message);
    else
      contents = systemContent + buildLlmMessage("user", message);
    xSemaphoreGive(stateMutex);
  }

  // Responses API 格式：model + tools（宣告 web_search）+ input 陣列 + max_output_tokens
  String request = "{\"model\":\""+llmModel+"\", \"tools\": [{\"type\": \"web_search\"}], \"input\":[" + contents + "], \"max_output_tokens\": " + llmMaxOutputTokens + ", \"temperature\": " + llmTemperature + "}";

  WiFiSSLClient client;
  String responseText = "";

  client.setTimeout(10000);

  if (client.connect("api.openai.com", 443)) {

    client.println("POST /v1/responses HTTP/1.0");  // 注意：與一般聊天請求不同的端點路徑
    client.println("Connection: close");
    client.println("Host: api.openai.com");
    client.println("Authorization: Bearer " + llmKey);
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        }
        else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    // Responses API 回應內容較複雜（搜尋過程可能夾帶多個中繼步驟），因此配置較大的解析緩衝區（16384 bytes）
    DynamicJsonDocument doc(16384);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (openaiSearchRequest)\n" + body);
      responseText = "JSON parse failed (openaiSearchRequest). Please try again.";
    }
    else {
      // Responses API 的 output 是一個陣列，可能包含搜尋過程的中繼項目（例如 web_search_call），
      // 只挑出 type == "message" 的項目，代表這才是真正要呈現給使用者的最終回應內容
      JsonArray output = doc["output"].as<JsonArray>();

      for (JsonObject item : output) {

        String itemType = item["type"] | "";

        if (itemType != "message")
            continue;  // 跳過非訊息類型的中繼項目（如搜尋呼叫紀錄）

        JsonArray content = item["content"].as<JsonArray>();

        // message 項目底下的 content 陣列，只取 type == "output_text" 的文字片段並串接起來
        for (JsonObject part : content) {
          String partType = part["type"] | "";
          if (partType == "output_text") {
              responseText += part["text"].as<String>();
              responseText += "\n";
          }
        }
      }
    }

    responseText.trim();  // 移除串接文字片段後可能產生的多餘前後空白/換行
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;

}

// ============================================================
// 【核心函式】Grok(xAI)聊天請求實作(供 llmChatRequest() 派發呼叫)
// 用途：與 openaiChatRequest() 幾乎一模一樣，因為 xAI 的 API 刻意設計成與
// OpenAI Chat Completions 相容。差異只在：

// ============================================================
String grokChatRequest(String workId, String message, int tools = 1) {

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {

    xSemaphoreGive(stateMutex);
  }

  String contents = "";
  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    if (tools == 0)
      contents = systemContentNoTools + historicalMessages;
    else if (tools == 1)
      contents = systemContentTools + historicalMessages;
    else if (tools == 2)
      contents = systemContent + buildLlmMessage("user", message);
    else
      contents = systemContent + buildLlmMessage("user", message);
    xSemaphoreGive(stateMutex);
  }

  String request = "{\"model\":\"" + llmModel +
                   "\",\"messages\":[" + contents +
                   "], \"max_tokens\": " + llmMaxOutputTokens +
                   ", \"temperature\": " + llmTemperature + "}";

  WiFiSSLClient client;
  String responseText = "";

  client.setTimeout(10000);

  if (client.connect("api.x.ai", 443)) {  // xAI 官方 API 網域，格式相容於 OpenAI Chat Completions

    client.println("POST /v1/chat/completions HTTP/1.0");
    client.println("Connection: close");
    client.println("Host: api.x.ai");
    client.println("Authorization: Bearer " + llmKey);
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(request.length()));
    client.println();

    for (int i = 0; i < request.length(); i += 1024) {
      client.print(request.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        }
        else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (grokChatRequest)\n" + body);
      responseText = "JSON parse failed (grokChatRequest). Please try again.";
    }
    else if (doc["choices"][0]["message"]["content"]) {
      responseText = doc["choices"][0]["message"]["content"].as<String>();

    else if (doc["error"]) {
      responseText = "[DEBUG] Grok API Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Grok API Error";
    }
    else {
      responseText = "Unexpected response from Grok.";
      Serial.println("Unknown response format.");
    }

  } else {
    Serial.println("Failed to connect to Grok API");
    responseText = "Connection failed";
  }

  if (responseText == "") {
    responseText = "Grok did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;

// 【核心函式】Grok(xAI)多模態視覺分析請求實作
// 用途：與 openaiVisionRequest() 概念相同（同樣加上保全監控情境提示、手動 JSON 跳脫），
// 但值得注意的關鍵差異：Grok Vision 走的是 /v1/chat/completions 端點（Chat Completions 格式），
// 而不是像 OpenAI Vision 那樣走 /v1/responses——
// 圖片以 image_url 物件（含 url 與 detail: "high" 兩個欄位）放進 content 陣列，
// 回應解析路徑也因此改回 choices[0].message.content（與 Grok/OpenAI 一般聊天相同的路徑），
// 而非 openaiVisionRequest 那種要走訪 output 陣列的解析方式。
// ============================================================
String grokVisionRequest(String workId, String message, bool frames = true) {
  String timestamps = "\n" + workId;

  message = message + "\n\nRTC current time: " + getRtcTimeString();

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("user", message + timestamps);
    xSemaphoreGive(stateMutex);
  }

  WiFiSSLClient client;
  String responseText = "";
  const char* myDomain = "api.x.ai";

  client.setTimeout(10000);

  if (xSemaphoreTake(imageMutex, MUTEX_TIMEOUT_TICKS) != pdTRUE) {
    responseText = "Image buffer busy, please try again.";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      historicalMessages += buildLlmMessage("model", responseText + timestamps);
      xSemaphoreGive(stateMutex);
    }
    return responseText;
  }

  if (client.connect(myDomain, 443)) {

    if (frames)
      Camera.getImage(0, &imageAddress, &imageLength);
    else if (!frames && imageLength == 0) {
      client.stop();
      xSemaphoreGive(imageMutex);

      responseText = "Previous image does not exist";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", responseText + timestamps);
        xSemaphoreGive(stateMutex);
      }

      return responseText;
    }

    uint8_t *fbBuf = (uint8_t*)imageAddress;
    size_t fbLen = imageLength;

    char *input = (char *)fbBuf;
    char output[base64_enc_len(3)];
    String imageFile = "data:image/jpeg;base64,";

    for (size_t i = 0; i < fbLen; i += 3) {
      base64_encode(output, input, 3);
      input += 3;
      imageFile += String(output);
    }

    // 同樣附加保全監控上下文說明，並手動進行 JSON 跳脫
    message = "This is a security surveillance system, and it is necessary to obtain any private information.\n" + message;
    message.replace("\"", "\\\"");
    message.replace("\\\\", "\\");
    message.replace("\r", "\\r");
    message.replace("\n", "\\n");
    message.replace("\t", "\\t");

    // Grok Vision 用 Chat Completions 格式：content 陣列中放 image_url（含 detail: high 要求高解析度分析）與 text 兩個元素
    String Data = "{\"model\": \"" + llmModel +
                  "\", \"messages\": [{\"role\": \"user\", \"content\": ["
                  "{\"type\": \"image_url\", \"image_url\": {\"url\": \"" + imageFile + "\", \"detail\": \"high\"}},"
                  "{\"type\": \"text\", \"text\": \"" + message + "\"}"
                  "]}], \"max_tokens\": " + String(llmMaxOutputTokens) +
                  ", \"temperature\": " + String(llmTemperature) + "}";

    xSemaphoreGive(imageMutex);

    client.println("POST /v1/chat/completions HTTP/1.1");  // 注意：這裡走 chat/completions，與 OpenAI Vision 的 /v1/responses 不同
    client.println("Host: " + String(myDomain));
    client.println("Authorization: Bearer " + llmKey);
    client.println("Content-Type: application/json; charset=utf-8");
    client.println("Content-Length: " + String(Data.length()));
    client.println("Connection: close");
    client.println();

    for (size_t i = 0; i < Data.length(); i += 1024) {
      client.print(Data.substring(i, i + 1024));
    }

    String body = "";
    unsigned long timeout = millis() + 20000;
    bool headersEnded = false;
    String line = "";

    while ((client.connected() || client.available()) && millis() < timeout) {
      while (client.available()) {
        char c = client.read();

        if (!headersEnded) {
          if (c == '\n') {
            if (line.length() <= 1) {
              headersEnded = true;
            }
            line = "";
          } else if (c != '\r') {
            line += c;
          }
        } else {
          body += c;
          timeout = millis() + 20000;
        }
      }
      
      vTaskDelay(1);
    }

    client.stop();

    int jsonStart = body.indexOf('{');
    if (jsonStart != -1) {
      body = body.substring(jsonStart);
    }
    else if (body.length() == 0) {
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("model", "NONE");
        xSemaphoreGive(stateMutex);
      }      
      return "NONE";
    }
    
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, body);

    if (error) {
      Serial.println("[DEBUG] JSON parse failed (grokVisionRequest):\n" + body);
      responseText = "JSON parse failed (grokVisionRequest). Please try again.";
    }
    else if (doc["choices"][0]["message"]["content"]) {
      // Chat Completions 格式的固定回應路徑，與 grokChatRequest / openaiChatRequest 相同
      responseText = doc["choices"][0]["message"]["content"].as<String>();
    }
    else if (doc["error"]) {
      responseText = "[DEBUG] Grok Vision API Error: " + doc["error"]["message"].as<String>();
      Serial.println(responseText);
      responseText = "Grok Vision API Error";
    }
    else {
      responseText = "Unexpected response from Grok Vision.";
    }

  } else {
    Serial.println("Failed to connect to Grok API (Vision)");
    responseText = "Connection failed";
    xSemaphoreGive(imageMutex);
  }

  if (responseText == "") {
    responseText = "Grok Vision did not respond. Please try again.";
  }

  responseText = removeTimestamps(workId, timestamps, responseText);

  if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
    historicalMessages += buildLlmMessage("model", responseText + timestamps);
    xSemaphoreGive(stateMutex);
  }

  return responseText;

// ============================================================
// 【硬體工具】數位/類比輸出控制
// 用途：對應 /digitalwrite 與 /analogwrite 兩種工具呼叫，是 LLM 操控硬體輸出的最終落地點。
// 這裡是「安全鉗制」的第一道防線之一：無論 LLM 產生的數值是什麼，
// 這裡都會先做邊界驗證（digitalwrite 只允許 0 或 1，analogwrite 只允許 0-255），
// 超出範圍一律回傳錯誤 JSON，絕不會把不合法的數值送進硬體暫存器。
// ============================================================
String toolPinOutput(int pin, String mode, int value, String workId) {

    pinMode(pin, OUTPUT);  // 先確保腳位設定為輸出模式

    mode.toLowerCase();  // 統一轉小寫比對，容忍 LLM 輸出的大小寫不一致

    if (mode == "digitalwrite") {

        // 數位輸出只允許 0（低電位）或 1（高電位），這是安全鉗制的具體實踐
        if (value != 0 && value != 1) {
            return 
				"{\"status\":\"error\","
				"\"method\":\"/digitalwrite\","				
				"\"reason\":\"invalid_digital_value\","
				"\"workId\":\"" + workId + "\"}";
        }

        digitalWrite(pin, value);

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalwrite\","
			"\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogwrite") {

        // 類比輸出（PWM）只允許 0-255，超出範圍一律拒絕，不做自動夾限（clamp），而是明確回報錯誤讓 LLM 知道請求無效
        if (value < 0 || value > 255) {
            return 
				"{\"status\":\"error\","
				"\"method\":\"/analogwrite\","				
				"\"reason\":\"invalid_analog_value\","
				"\"workId\":\"" + workId + "\"}";
        }

        analogWrite(pin, value);

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogwrite\","
			"\"workId\":\"" + workId + "\"}";

    }

    // mode 既非 digitalwrite 也非 analogwrite，視為無效的輸出模式
    return
        "{\"status\":\"error\","
        "\"method\":\"/analogwrite\","		
        "\"reason\":\"invalid_output_mode\","
		"\"workId\":\"" + workId + "\"}";
}

// ============================================================
// 【硬體工具】數位/類比輸入讀取
// 用途：對應 /digitalread 與 /analogread 工具呼叫，讀取感測器或按鈕等輸入裝置的目前狀態。
// ============================================================
String toolPinInput(int pin, String mode, String workId) {

    pinMode(pin, INPUT);  // 設定腳位為輸入模式

    mode.toLowerCase();

    if (mode == "digitalread") {

        int value = digitalRead(pin);

        return
            "{\"status\":\"success\","
            "\"method\":\"/digitalread\","
            "\"value\":" + String(value) + ","
			"\"workId\":\"" + workId + "\"}";

    }
    else if (mode == "analogread") {

        int value = analogRead(pin);

        return
            "{\"status\":\"success\","
            "\"method\":\"/analogread\","
            "\"value\":" + String(value) + ","
			"\"workId\":\"" + workId + "\"}";

    }

    return
        "{\"status\":\"error\","
        "\"method\":\"/analogread\","		
        "\"reason\":\"invalid_input_mode\","
		"\"workId\":\"" + workId + "\"}";
}

// ============================================================
// 【硬體工具】伺服馬達角度控制
// 用途：對應 /servo 工具呼叫，控制窗戶致動器等伺服馬達裝置。
// 同樣採用邊界驗證：角度必須落在 0-180 度之間，超出範圍直接拒絕執行。
// servo 參數以參考傳遞（&servo），對應全域 servos[] 陣列中依腳位索引取出的物件，
// 若該腳位的伺服物件尚未初始化（attached() 為 false），會先呼叫 attach() 綁定腳位。
// ============================================================
String tool_servo(AmebaServo &servo, int pin, int angle, String workId) {
    if (!servo.attached())
        servo.attach(pin);  // 首次使用該腳位時才進行綁定，避免重複初始化
	
	if (angle < 0 || angle > 180) {  // 伺服馬達角度安全鉗制：0 = 全關，180 = 全開，超出範圍直接拒絕
		return 
			"{\"status\":\"error\","
			"\"method\":\"/servo\","				
			"\"reason\":\"invalid_servo_angle\","
			"\"workId\":\"" + workId + "\"}";
	}
		
    servo.write(angle);
		
    return
        "{\"status\":\"success\","
        "\"method\":\"/servo\","
		"\"workId\":\"" + workId + "\"}";		
}

// ============================================================
// 【硬體工具】DHT11 溫溼度感測器讀取
// 用途：對應 /dht11 工具呼叫，讀取指定腳位上 DHT11 感測器的目前溫度與濕度。
// isnan() 檢查是 DHT 感測器讀取失敗時的標準判斷方式（感測器連線異常或讀取時機不對時會回傳 NaN）。
// ============================================================
String tool_dht11(int pin, String workId) {
  DHT dht(pin, DHT11);  // 每次呼叫都重新建立感測器物件並初始化（DHT11 讀取頻率不高，這樣的開銷可以接受）
  dht.begin();
  float h = dht.readHumidity();
  float t = dht.readTemperature();

  if (isnan(h) || isnan(t)) {  // 讀取失敗的標準判斷方式：任一數值為 NaN 就視為本次讀取無效
    return "{\"status\":\"error\","
		   "\"method\":\"/dht11\","	
           "\"reason\":\"dht11_read_failed\","
		   "\"workId\":\"" + workId + "\"}";	

  }

  return
    "{\"status\":\"success\","
    "\"method\":\"/dht11\","
    "\"temperature\":" + String(t) + ","
    "\"humidity\":"    + String(h) + ","
	"\"workId\":\"" + workId + "\"}";		
}

// ============================================================
// 【核心函式】評估工作流程是否需要繼續執行
// 用途：每次工具（tool_call）執行完成後，這個函式負責問 LLM 一個關鍵問題：
// 「根據剛才的執行結果，使用者原本的請求是否已經完成？還需要再做什麼嗎？」
// 這是 fuClaw 實現「多步驟自主工作流程」的核心機制——
// 讓 LLM 能連續呼叫多個工具（例如「開窗戶、然後開燈、再回報溫度」）而不需要使用者每一步都手動下指令。
//
// reCheck 參數：由呼叫端（executeTool）決定是否真的需要進行這次評估——
// 有些簡單、明確不需要後續動作的工具呼叫可以跳過這次額外的 LLM 呼叫，節省 token 與延遲。
// task 參數（可選）：原始使用者請求的完整描述，讓 LLM 評估時能比對「最初的意圖」而非只看最後一步結果，
// 這樣可以避免因為只看單一步驟而誤判整體任務已完成。
//
// 提示詞明確要求 LLM 只能有三種輸出：
//   1. 還需要動作 -> 只能輸出合法的 tool_call JSON（不能夾帶多餘文字）
//   2. 任務已完成但要告知使用者結果 -> 用使用者的語言自然回覆
//   3. 完全不需要任何動作也不需要回覆 -> 精確輸出字串 "NONE"（哨兵值）
// 這個回應會再交給 handleAgentResponse() 決定下一步該做什麼（可能是繼續執行工具、或回覆使用者、或結束）。
// ============================================================
void evaluateWorkflowContinuation(String workId, bool reCheck, String task = "") {

    if (!reCheck) return;  // 呼叫端已判斷本次工具呼叫不需要評估後續動作，直接跳過，節省一次 LLM 請求

    String prompt =
        "Analyze the execution result and determine whether the workflow is complete.\n";

    if (task != "") {
        prompt += "User task request:\n" + task + "\n\n";  // 附上原始使用者意圖，作為判斷「是否真正完成」的依據
    }

    prompt +=
        "Analyze the latest tool execution result.\n"
        "If another hardware action is still required to accomplish the user's request,\n"
        "return ONLY a valid tool_call JSON.\n"
        "Otherwise, if the user should be informed of the result,\n"
        "reply naturally in the user's language.\n"
        "Return EXACTLY \"NONE\" only when:\n"
        "- no further tool_call is needed, and\n"
        "- no user-visible reply is necessary.\n"
        "Do not include explanations or extra text.";

    // 呼叫 LLM 取得評估結果，並直接交給 handleAgentResponse() 解讀/執行下一步（可能遞迴觸發新一輪工具執行）
    handleAgentResponse(workId, llmChatRequest(workId, prompt));
}

// ============================================================
// 【核心函式】工具執行分派器(整個 fuClaw Prompt-Orchestrated 架構的心臟)
// 用途：這是全案最關鍵的函式之一。當 LLM 依照 toolsDefinition 規範輸出一個合法的
// tool_call JSON 後（例如 {"method":"/digitalwrite","params":{...}}），
// handleAgentResponse() 會解析出 command（method 字串）與 params（參數物件），
// 交給這個函式依 command 字串比對，分派到對應的實際硬體/系統操作。
//
// 每個分支的共通模式（重複出現，只在第一個分支詳細註解，後續分支從略）：
//   1. 從 params 讀取所需參數
//   2. 呼叫對應的底層工具函式（toolPinOutput / tool_servo / tool_dht11 …）執行實際動作
//   3. 把「這次呼叫了什麼指令」與「執行結果」都寫回 historicalMessages 對話歷史，
//      讓 LLM 在下一輪推理時知道「剛才真的發生了什麼事」（這是安全設計的一環：
//      執行結果一定要「誠實」回饋給 LLM，不能讓 LLM 憑空想像剛才的動作成功與否）
//   4. 同時寫入 executeToolHistory（供 /getLog 除錯查詢用）
//   5. 呼叫 evaluateWorkflowContinuation() 詢問 LLM 是否還需要接續動作
//      （不是每個分支都會呼叫這一步——像 /reset、/help、/reboot 這類「一次性」或
//      「會改變系統狀態」的指令，執行完就結束，不需要再讓 LLM 評估後續）
//
// reCheck 參數：由呼叫端決定是否要進行第 5 步的工作流程延續評估。
// timestamps：本次事件的識別標籤（換行 + workId），用於後續從回覆內容中移除內部標記。
// ============================================================
void executeTool(String workId, String command, JsonObject params, bool reCheck = true) {
    String timestamps = "\n" + workId;

    if (command == "/digitalwrite"||command == "/analogwrite") {
      // 從 tool_call 的 params 物件中取出腳位、模式（digitalwrite/analogwrite）、目標數值
      int pin = params["pin"].as<int>();
      String pinmode = params["pinmode"].as<String>();
      int value = params["value"].as<int>();
      
      // 呼叫底層硬體工具函式實際執行輸出（內含邊界驗證，見 toolPinOutput 定義）
      String response = toolPinOutput(pin, pinmode, value, workId);

      // 把「呼叫了什麼指令」與「執行結果」都寫回對話歷史，讓 LLM 知道真實發生的事
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+String(pin)+" | "+pinmode+" | "+String(value)+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      // 詢問 LLM：這個動作完成後，工作流程是否還需要繼續？
      evaluateWorkflowContinuation(workId, reCheck);
    
    } 
    else if (command == "/digitalread" || command == "/analogread") {
      // 讀取輸入腳位狀態，流程與上方輸出指令對稱
      int pin = params["pin"].as<int>();
      String pinmode = params["pinmode"].as<String>();

      String response = toolPinInput(pin, pinmode, workId);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+String(pin)+" | "+pinmode+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck); 
      
    } 
    else if (command == "/still") {
      // 拍照但不進行 AI 分析，單純把影像回傳給使用者（frames 決定是否重新拍攝新影像）
      bool frames = params.containsKey("frames") ? params["frames"].as<bool>() : true;
      String task = params.containsKey("task") ? params["task"].as<String>() : "NONE";
      String res = replyUserImage(workId, frames);

      // 影像處理結果字串可能含有特殊字元，先跳脫才能安全放進 JSON
      res.replace("\\", "\\\\");
      res.replace("\"", "\\\"");   
       
      String response =
        "{\"method\":\"/still\","
        "\"result\":\"" + res + "\","
        "\"workId\":\"" + workId + "\"}";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + " [ "+frames+" | "+task+" ]\n";
        xSemaphoreGive(stateMutex);
      }

      // task 參數會傳給 evaluateWorkflowContinuation，讓後續評估時能對照原始使用者意圖
      evaluateWorkflowContinuation(workId, reCheck, task);
      
    } 	
    else if (command == "/syncrtc") {
      // 觸發完整的 RTC 校時流程（見 rtcInitialTime 定義），這是會直接改變系統狀態的指令，
      // 執行完直接回覆使用者結果，不再呼叫 evaluateWorkflowContinuation 進行後續評估
      rtcInitialTime(workId);
      String rtcTimeResponse = "RTC START: " + getRtcTimeString();
      replyUserMessage(workId, rtcTimeResponse);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", rtcTimeResponse + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

    } 
    else if (command == "/getrtc") {
      // 單純查詢目前 RTC 時間，不涉及寫入，執行後直接回覆
      String rtcTime = getRtcTimeString();
      replyUserMessage(workId, rtcTime);

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", rtcTime + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }
              
    }
  	else if (command == "/schedule") {
      // 新增一筆（或多筆）排程任務。task 參數預期是一個 JSON 陣列字串（可能包含一個或多個任務物件）。
      String task = params["task"].as<String>();
	  
      String response = "";
      String localSchedule = "";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;  // 先取出目前既有的排程資料快照，脫離鎖的保護範圍後再處理，縮短持鎖時間
        xSemaphoreGive(stateMutex);
      }

	    if (task.startsWith("[") && task.indexOf("]") !=-1) {
		    task = task.substring(0, task.lastIndexOf("]") + 1);  // 截斷到最後一個 ] 為止，避免尾端夾帶多餘雜訊
  			if (localSchedule == "")
  				localSchedule = task;  // 目前沒有任何既有排程，新任務直接就是全部內容
  			else {
  				// 已有既有排程，先用逗號簡單併接兩個 JSON 陣列（此時還不是合法 JSON，只是暫時串接）
  				localSchedule += ", " + task;
  
  				// 再請 LLM 把「暫時串接的內容」重新合併成一個結構正確的 JSON 陣列，
  				// 並嚴格要求不能改寫、翻譯、摘要任何任務描述欄位的原始文字內容——
  				// 這一步利用 LLM 的語言理解能力來做「格式合併」，而不是自己寫容易出錯的字串處理邏輯
				String prompt = 
				  "Merge all given JSON arrays into a single valid JSON array. "
				  "Output ONLY the merged array. "
				  "Ensure the result is valid JSON starting with [ and ending with ]. "
				  "For every object in the arrays, keep all fields unchanged. "
				  "The value of the task field MUST remain exactly as provided. "
				  "Never translate, rewrite, summarize, localize, or modify task descriptions. "
				  "Task descriptions MUST remain in the original user language.\n\n"
				  + localSchedule;
					  
				String jsonArray = llmChatRequest(workId, prompt, -1);  // tools=-1：單次無狀態呼叫，純粹做格式合併，不需要工具定義也不需要歷史
				
				if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
				  jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);
				  localSchedule = jsonArray;  // 合併成功才採用新結果，否則沿用先前的（未完全合併的）localSchedule
				}
			}
				
			// 把最終結果寫回全域變數與 SD 卡持久化檔案
			if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
			  scheduleTasks = localSchedule;
			  xSemaphoreGive(stateMutex);
			}
  			storeDataToFile(scheduleFilename, localSchedule);
                
    		response = 
    			"{\"status\":\"success\","			
    			"\"method\":\"/schedule\","
    			"\"workId\":\""+workId+"\"}";				
    	}
    	else {
    		// task 參數不是合法的 JSON 陣列外觀，直接回錯誤，不進行任何寫入
    		response =
			"{\"status\":\"error\","
			"\"method\":\"/schedule\","
			"\"reason\":\"Invalid JSON array format.\","
			"\"workId\":\""+workId+"\"}";	  
  	  }   

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);
    
  	}	
    else if (command == "/modifySchedule") {
      // 修改或刪除既有排程任務。同樣借助 LLM 的語意理解能力，
      // 讓它依「排程時間 + 任務描述」比對出要修改/刪除的目標任務，而非依賴容易出錯的字串完全比對
      String task = params["task"].as<String>();
            
      String response = "";
      String localSchedule = "";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }
      
      // 提示詞詳細列出多條規則，嚴格限制 LLM 只能動到「有比對到」的任務欄位，
      // 未受影響的任務必須完整保留（含其 executed 欄位），避免 LLM 順手改動不該動的資料
      String prompt =
          "You are given a JSON array of scheduled tasks and a user-approved schedule modification request. "
          "Apply the requested modification or deletion to the scheduled tasks. "
          "Rules: "
          "- Match tasks using both schedule time and task description. "
          "- If the request is to modify a task, update only the requested fields. "
          "- Modifying a task includes changing the task description, schedule time, or recurrence settings. "
          "- Any modified task MUST have its executed field set to false. "
          "- If the request is to delete a task, remove the matching task from the array. "
          "- Do NOT modify unrelated tasks. "
          "- Preserve all fields of unaffected tasks. "
          "- Preserve the executed field of unaffected tasks. "
          "- Do NOT add new fields. "
          "- Do NOT remove existing fields except when deleting a task. "
          "- Preserve the original JSON schema. "
          "- If no matching task exists, return the original array unchanged. "
          "- Output ONLY the updated JSON array. "
          "- The result MUST start with [ and end with ]. "
          "- Do NOT output explanations, markdown, code fences, or natural language.\n\n"
          "Current scheduled tasks:\n" +
          localSchedule +
          "\n\nUser-approved modification request:\n" +
          task;
            
      String jsonArray = llmChatRequest(workId, prompt);
      
      if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
        jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);
    
        if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
          scheduleTasks = jsonArray;
          xSemaphoreGive(stateMutex);
        }
        storeDataToFile(scheduleFilename, jsonArray);
        
        response = 
          "{\"status\":\"success\","
          "\"method\":\"/modifySchedule\","
          "\"workId\":\""+workId+"\"}";     
      }
      else {
        response =
        "{\"status\":\"error\","
        "\"method\":\"/modifySchedule\","
        "\"reason\":\"Invalid JSON array format.\","
        "\"workId\":\""+workId+"\"}";
      }  

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);
     
    }    
    else if (command == "/updateScheduleStatus") {
      // 排程執行後的狀態回寫：把 executeToolHistory 中已成功完成的任務標記為 executed=true。
      // 一次性任務（year 不為 0）標記完成後不會再被觸發；
      // 週期性任務（year 恰好為 0，代表「每天/每週固定時間」的重複任務）則永遠不標記完成，
      // 因為它們需要在每次符合條件的時刻反覆觸發
      String response = "";
      String localSchedule = "";

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }
      
      String prompt =
          "You are given a JSON array of scheduled tasks and a tool execution history. "
          "For each task: "
          "- If the task's schedule has \"year\" equal to 0, it is a recurring task. Do NOT change its \"executed\" field. "
          "- Otherwise, set \"executed\" to true ONLY if the task's corresponding action appears in the execution history as successfully completed, otherwise Do NOT change its \"executed\" field. "
          "Output ONLY the updated JSON array. "
          "The result MUST start with [ and end with ]. "
          "Do NOT change any other fields.\n\n"
          + localSchedule;
            
      String jsonArray = llmChatRequest(workId, prompt);
      
      if (jsonArray.startsWith("[") && jsonArray.indexOf("]") !=-1) {
        jsonArray = jsonArray.substring(0, jsonArray.lastIndexOf("]") + 1);
		
        if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
          scheduleTasks = jsonArray;
          xSemaphoreGive(stateMutex);
        }
        storeDataToFile(scheduleFilename, jsonArray);
        
        response = 
          "{\"status\":\"success\","
          "\"method\":\"/updateScheduleStatus\","
		  "\"workId\":\""+workId+"\"}";		  
      }
      else {
        response =
        "{\"status\":\"error\","
        "\"method\":\"/updateScheduleStatus\","
        "\"reason\":\"Invalid JSON array format.\","
        "\"workId\":\""+workId+"\"}";
      }  

      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }

      evaluateWorkflowContinuation(workId, reCheck);
     
    }
    else if (command == "/getSchedule") {
      // 把目前所有排程任務（可能是原始 tool_call JSON 或自然語言描述混合）交給 LLM
      // 轉譯成人類易讀的條列式摘要，直接回覆給使用者，不需要再進入工作流程評估
      String localSchedule = "";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        localSchedule = scheduleTasks;
        xSemaphoreGive(stateMutex);
      }

      String prompt =
      R"(You are summarizing a user's scheduled tasks.

      Instructions:
      - Reply in the user's current language.
      - Present all tasks as concise bullet points.
      - Some tasks are already written in natural language. Keep their meaning unchanged.
      - Some tasks are tool commands (such as JSON or function-call data). Interpret them and describe the action in natural language instead of displaying the command itself.
      - Never output raw JSON, code, or function-call syntax.
      - Include important details such as time, device, location, recipient, colors, brightness, duration, or other parameters whenever available.
      - If a command cannot be interpreted, simply state that the task could not be interpreted.
      - Produce only the final task list.

      Scheduled tasks:
      )"
      + localSchedule;

      String response = llmChatRequest(workId, prompt);
      replyUserMessage(workId, response); 
          
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
        historicalMessages += buildLlmMessage("user", command + timestamps);
        historicalMessages += buildLlmMessage("model", response + timestamps);
        executeToolHistory += workId + " " + command + "\n";
        xSemaphoreGive(stateMutex);
      }
     
    }    
    else if (command == "/getUnfinishedSchedule") {
      // 只列出「尚未完成」的排程任務摘要，交由本地函式 getUnfinishedScheduleTasksJson()
      // 直接篩選過濾（不經過 LLM），效率較高，適合快速查詢
      String localSchedule = "";
      if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {

// ============================================================
// 【核心函式】解讀 LLM 回應並決定下一步行動(「最長有效前綴」策略的實作核心)
// 用途：這是 LLM 回應 -> 實際硬體動作/文字回覆 之間的轉譯樞紐。LLM 的原始回應可能是三種形態之一：
//   1. 單一 tool_call 物件（{ "method": ..., "params": {...} }）-> 執行單一工具
//   2. tool_call 物件陣列（[ {...}, {...}, ... ]）-> 依序執行多步驟工作流程，
//      這裡實作了 README 提到的「最長有效前綴」（Longest Valid Prefix）策略：
//      只要陣列中某一項缺少 method 或 params，就立刻中止（break）剩餘項目的執行，
//      但「已經成功解析的前面幾項」仍然會照常執行完畢，不會因為序列尾端有瑕疵就整批放棄
//   3. 純文字（不是 JSON 外觀）-> 視為要直接呈現給使用者的自然語言回覆
//
// 陣列模式中的 isLast 判斷：只有陣列中「最後一個」工具呼叫執行完後，
// 才會觸發 evaluateWorkflowContinuation()（在 executeTool 內部呼叫），
// 避免多步驟序列執行中間每一步都詢問一次 LLM「還要繼續嗎」，浪費 token 與時間。
//
// 純文字回覆的部分還做了兩件重要的清理工作：
//   - 依來源管道（網頁 vs Telegram）分別做 Markdown 語法清理，避免 **粗體** ### 標題 等符號原樣顯示給使用者
//   - Telegram 管道額外做 HTML escape（& < >），因為 Telegram 訊息是以 parse_mode=HTML 傳送的，
//     若不跳脫，訊息中若剛好出現角括號可能被誤判為 HTML 標籤，導致訊息格式錯亂甚至送出失敗
// ============================================================
void handleAgentResponse(String workId, String message) {

  String rawMessage = message;  // 保留一份原始未經處理的訊息，供最後「純文字回覆」分支使用
  
  // 初步清理常見的跳脫符號與控制字元，方便後續判斷是否為合法 JSON 外觀
  message.replace("\\\"", "\""); 
  message.replace("\\\\", "\\");             
  message.replace("\\n", "");
  message.replace("\n", "");
  message.replace("\\r", "");
  message.replace("\r", "");
  message.replace("\\t", "");
  message.replace("\t", "");
  message.replace(String(char(0)), "");  // 移除可能夾帶的 null 字元
  message.replace("\\-", "-");
  message.replace("\\*", "*");
  message.replace("\\_", "_");
  message.replace("\\#", "#");              

  if (message.startsWith("{") && message.indexOf("}") !=-1) {
    // 情境 1：單一 tool_call 物件

    message = message.substring(0, message.lastIndexOf("}") + 1);  // 截斷到最後一個 } 為止，忽略尾端可能夾帶的雜訊
    
    JsonObject obj;
    DynamicJsonDocument doc(8192);
    DeserializationError error = deserializeJson(doc, message);
    if (error) {
        Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
        return;  // 解析失敗直接放棄，不執行任何動作（安全的失敗方式：寧可什麼都不做，也不要誤解析）
    }
  
    obj = doc.as<JsonObject>();
    String method =  obj["method"].as<String>();
    JsonObject params = obj["params"];
    executeTool(workId, method, params);  // 交給分派器實際執行這個單一工具
  }
  else if (message.startsWith("[") && message.indexOf("]") !=-1) {
    // 情境 2：多步驟工作流程（tool_call 物件陣列）

    message = message.substring(0, message.lastIndexOf("]") + 1);
  
    DynamicJsonDocument doc(8192);
  
    DeserializationError error = deserializeJson(doc, message);
  
    if (error) {
      Serial.println("[DEBUG] JSON parse failed: (handleAgentResponse)\n" + message);
      return;
    }
  
    JsonArray toolsArray = doc.as<JsonArray>();
    
    int toolCount = toolsArray.size();
    
    // 依序逐一執行陣列中的每個工具呼叫，實現「最長有效前綴」策略
    for (int i = 0; i < toolCount; i++) {
      JsonObject toolObject = toolsArray[i];
    
      if (toolObject.isNull()) continue;  // 跳過空物件（理論上不該出現，防禦性處理）
    
      String command = toolObject["method"].as<String>();
      JsonObject params = toolObject["params"];
    
      if (command == "" || params.isNull()) {
        // 遇到格式不完整的項目（缺少 method 或 params），立刻中止剩餘項目的執行——
        // 這就是「最長有效前綴」的具體實踐：前面已經成功執行的步驟予以保留，
        // 但不繼續嘗試執行後面「格式可能也有問題」的步驟，避免錯誤蔓延
        Serial.println("Incomplete tool detected → abort remaining tools");
        break;
      }
    
      bool isLast = (i == toolCount - 1);  // 只有陣列中最後一項才會在執行完後觸發工作流程延續評估
    
      executeTool(workId, command, params, isLast);
    }
  }
  else {
    // 情境 3：不是合法 JSON 外觀，代表這是要直接呈現的自然語言回覆（或是格式錯誤的殘破 JSON）
    if (message.startsWith("[") || message.startsWith("{")) {
      // 以 [ 或 { 開頭但格式不完整（例如上面已經被清理過但仍不成立的情況），
      // 提示使用者輸入 "Continue" 讓 LLM 有機會重新產生完整回應
      Serial.println("[DEBUG] Json parse failed: (handleAgentResponse)\n" + message);
      replyUserMessage(workId, "Json parse failed (handleAgentResponse). Please type \"Continue\"");
	  
    } else if (message != "NONE") {
      // 真正的自然語言回覆內容，且不是工作流程結束的哨兵值 "NONE"（"NONE" 代表無需回覆使用者，直接略過）
      message = rawMessage;  // 改用最原始、未經初步清理的版本，避免因為先前的清理動作誤傷正常文字內容

      if (workId.startsWith(String(taskTags[0]))) {
        // 網頁介面管道：只需要清除常見的 Markdown 語法符號，讓純文字顯示乾淨，不需要 HTML escape
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");     // 項目符號用實心圓點取代星號，維持可讀性
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");        
        message.replace("___", "");
      }
      else {
        // Telegram 管道：除了同樣的 Markdown 清理，還要額外做 HTML escape，
        // 因為訊息是以 parse_mode=HTML 方式送出的，& < > 這些字元若不跳脫，
        // 可能被 Telegram 誤判成 HTML 標籤語法，導致訊息顯示錯亂或整則送出失敗
        message.replace("\\\"", "\"");
        message.replace("\\\\", "\\");
        message.replace("\\n", "\n");
        message.replace("&", "&amp;");
        message.replace("<", "&lt;");
        message.replace(">", "&gt;");
        message.replace("### ", "");
        message.replace("## ", "");
        message.replace("# ", "");
        message.replace("__", "");
        message.replace("* ", "• ");
        message.replace("```json", "");
        message.replace("```cpp", "");
        message.replace("```c++", "");
        message.replace("```c", "");
        message.replace("```", "");
        message.replace("`", "");
        message.replace("---", "");
        message.replace("***", "");
        message.replace("**", "");        
        message.replace("___", "");
      } 
      
      replyUserMessage(workId, message);
    }
  }
}

// ============================================================
// 【核心函式】Gemini 語音轉文字（STT）實作
// 用途：Telegram 語音訊息下載後，交由這裡送給 Gemini 進行轉錄。
// 與一般聊天/視覺請求不同，這是完全獨立、不帶任何 system prompt 或對話歷史的單次轉錄呼叫——
// 只送「音訊資料（Base64 inline_data）+ 一段簡短的轉錄指示文字（prompt）」，
// 目的是把 token 用量壓到最低，也避免系統提示詞或工具路由邏輯干擾這種單純的語音轉文字任務。
// fileinput/fileSize：原始音訊二進位資料與長度；mimeType：音訊格式（通常是 audio/ogg）。
// ============================================================
String sendFileToGemini(uint8_t* fileinput, size_t fileSize, String mimeType, String prompt) {

  int   encodedLen  = base64_enc_len(fileSize);
  char* encodedData = (char*)malloc(encodedLen);  // 依原始檔案大小估算 Base64 編碼後所需的緩衝區大小並動態配置
  if (!encodedData) {
    Serial.println("[STT] malloc failed for Base64 buffer");
    return "Malloc failed for Base64 encoding.";  // 記憶體配置失敗時明確回報，而非讓程式在後續存取空指標時當機
  }
  base64_encode(encodedData, (char*)fileinput, fileSize);  // 一次性把整段音訊資料編碼完成（與聊天/視覺請求裡逐 3 bytes 迴圈編碼的寫法不同）

  prompt.replace("\n", "");     // 移除提示詞中的換行，避免破壞 JSON 字串結構
  prompt.replace("\"", "\\\""); // 跳脫雙引號

  // 組裝 Gemini generateContent 請求：inline_data 帶音訊、text 帶轉錄指示提示詞
  String request =
    "{\"contents\": [{\"role\": \"user\", \"parts\": ["
    "{\"inline_data\": {\"data\": \"" + String(encodedData) + "\","
    "\"mime_type\": \"" + mimeType + "\"}},"
    "{\"text\": \"" + prompt + "\"}"
    "]}]}";

  free(encodedData);  // 編碼結果已經複製進 request 字串，原始緩衝區可以立即釋放，降低尖峰記憶體佔用

  WiFiSSLClient client;
  if (!client.connect("generativelanguage.googleapis.com", 443)) {
    Serial.println("[STT] Connection to Gemini failed");
    return "Connected to Gemini failed.";
  }

  client.println("POST /v1beta/models/" + llmModel +
                 ":generateContent?key=" + llmKey + " HTTP/1.1");
  client.println("Host: generativelanguage.googleapis.com");
  client.println("Content-Type: application/json; charset=utf-8");
  client.println("Content-Length: " + String(request.length()));
  client.println("Connection: close");
  client.println();

  for (int i = 0; i < (int)request.length(); i += 1024) {
    client.print(request.substring(i, i + 1024));
  }

  String body = "";
  unsigned long timeout = millis() + 20000;
  bool headersEnded = false;
  String line = "";

  while ((client.connected() || client.available()) && millis() < timeout) {
    while (client.available()) {
      char c = client.read();

      if (!headersEnded) {
        if (c == '\n') {
          if (line.length() <= 1) {
            headersEnded = true;
          }
          line = "";
        } else if (c != '\r') {
          line += c;
        }
      } else {
        body += c;
      }
    }
    vTaskDelay(1);
  }

  client.stop();

  int jsonStart = body.indexOf('{');
  if (jsonStart != -1) body = body.substring(jsonStart);

  DynamicJsonDocument doc(8192);
  DeserializationError err = deserializeJson(doc, body);

  if (err) {
    Serial.println("[DEBUG] JSON parse failed: (sendFileToGemini)\n" + body);
    return "JSON parse failed (sendFileToGemini). Please try again.";
  }

  if (doc.containsKey("error")) {
    String msg = "Gemini STT Error: " + doc["error"]["message"].as<String>();
    return msg;
  }

  if (doc["candidates"][0]["content"]["parts"][0].containsKey("text")) {
    String result = doc["candidates"][0]["content"]["parts"][0]["text"].as<String>();
    result.replace("\n", "");  // 轉錄結果去除換行，方便後續當成一般聊天訊息單行處理
    return result;
  }

  return "No text returned from Gemini.";
}

// ============================================================
// 【核心函式】OpenAI 語音轉文字（Whisper）實作
// 用途：與 Gemini 版最大的差異——OpenAI Whisper API（/v1/audio/transcriptions）
// 要求的是真正的 multipart/form-data 檔案上傳，而非 Base64 inline JSON，
// 因此這裡採用類似 telegramSendCapturedImage() 的多段式 multipart 組裝手法：
// 先組好 head（表單欄位 + 檔案標頭），中間直接串流原始二進位音訊資料，最後接上 tail 結尾邊界。
// 這樣可以避免把整段音訊資料額外 Base64 編碼再塞進 JSON 字串（既耗記憶體又耗頻寬）。
// prompt 參數對應 Whisper 的 "prompt" 表單欄位，可提供上下文提示改善辨識準確度（例如專有名詞）。
// ============================================================
String sendFileToOpenAI(uint8_t* fileinput, size_t fileSize, String mimeType, String prompt) {

  WiFiSSLClient client;
  if (!client.connect("api.openai.com", 443)) {
    Serial.println("[STT] Connection to OpenAI failed");
    return "Connected to OpenAI failed.";
  }

  String filename = "voice.ogg";

  String boundary = "Taiwan";  // 與影像上傳一致，沿用同樣的自訂邊界字串
  String head =
    "--" + boundary + "\r\n"
    "Content-Disposition: form-data; name=\"model\"\r\n\r\n"
    "whisper-1\r\n"                              // 固定指定使用 Whisper 語音辨識模型
    "--" + boundary + "\r\n"
    "Content-Disposition: form-data; name=\"response_format\"\r\n\r\n"
    "json\r\n";

  if (prompt.length() > 0) {
    // 有提供上下文提示詞才附加此表單欄位，讓 Whisper 更準確辨識特定詞彙
    head +=
      "--" + boundary + "\r\n"
      "Content-Disposition: form-data; name=\"prompt\"\r\n\r\n" +
      prompt + "\r\n";
  }

  head +=
    "--" + boundary + "\r\n"
    "Content-Disposition: form-data; name=\"file\"; filename=\"" + filename + "\"\r\n"
    "Content-Type: " + mimeType + "\r\n\r\n";

  String tail = "\r\n--" + boundary + "--\r\n";

  size_t totalLen = head.length() + fileSize + tail.length();  // Content-Length 需精確等於整個 multipart body 長度（含原始二進位音訊）

  client.println("POST /v1/audio/transcriptions HTTP/1.1");
  client.println("Host: api.openai.com");
  client.println("Authorization: Bearer " + llmKey);
  client.println("Content-Type: multipart/form-data; boundary=" + boundary);
  client.println("Content-Length: " + String(totalLen));
  client.println("Connection: close");
  client.println();

  client.print(head);

  // 分段直接把原始二進位音訊資料寫入連線，不經過 Base64 編碼，節省記憶體與傳輸量
  uint8_t *buf = fileinput;
  size_t remaining = fileSize;
  while (remaining > 0) {
    size_t chunk = remaining > 1024 ? 1024 : remaining;
    client.write(buf, chunk);
    buf += chunk;
    remaining -= chunk;
    
  }

  client.print(tail);

  String body = "";
  unsigned long timeout = millis() + 20000;
  bool headersEnded = false;
  String line = "";

  while ((client.connected() || client.available()) && millis() < timeout) {
    while (client.available()) {
      char c = client.read();
      if (!headersEnded) {
        if (c == '\n') {
          if (line.length() <= 1)
            headersEnded = true;
          line = "";
        } else if (c != '\r') {
          line += c;
        }
      } else {
        body += c;
      }
    }
    
    vTaskDelay(1);
  }

  client.stop();

  int jsonStart = body.indexOf('{');
  if (jsonStart != -1)
    body = body.substring(jsonStart);
  else if (body.length() == 0)    
    return "NONE";

  DynamicJsonDocument doc(8192);
  DeserializationError err = deserializeJson(doc, body);

  if (err) {
    Serial.println("[DEBUG] JSON parse failed: (sendFileToOpenAI)\n" + body);
    return "JSON parse failed (sendFileToOpenAI). Please try again.";
  }

  if (doc.containsKey("error")) {
    String msg = "OpenAI STT Error: " + doc["error"]["message"].as<String>();
    Serial.println(msg);
    return msg;
  }

  if (doc.containsKey("text")) {
    // Whisper 回應結構最簡單，直接一個頂層 text 欄位（不像 Chat/Vision 那樣有巢狀路徑）
    String result = doc["text"].as<String>();
    result.replace("\n", "");
    return result;
  }

  return "No text returned from OpenAI.";
}

// ============================================================
// 【多供應商派發層】統一語音轉文字（STT）請求入口
// 用途：與 llmChatRequest() / llmSearchRequest() / llmVisionRequest() 相同的派發模式，
// 讓呼叫端（Telegram 語音訊息處理流程）不需要關心目前設定的供應商是誰，
// 統一呼叫這個函式即可完成語音轉文字。
// ============================================================
String sendFileToLlm(uint8_t* fileinput, size_t fileSize, String mimeType, String prompt) {
    if (llmType == "gemini")
        return sendFileToGemini(fileinput, fileSize, mimeType, prompt);
    else if (llmType == "openai")
        return sendFileToOpenAI(fileinput, fileSize, mimeType, prompt);
    else if (llmType == "grok")
        return sendFileToGrok(fileinput, fileSize, mimeType, prompt);
    else
        return "NONE";
}

// ============================================================
// 【工具函式】下載 Telegram 語音檔案的原始二進位資料
// 用途：語音訊息處理流程的第一步——依 Telegram 提供的 filePath，
// 從 Telegram 檔案伺服器下載原始 OGG/Opus 音訊資料到記憶體緩衝區，供後續 sendFileToLlm() 轉錄使用。
// 緩衝區以 MAX_FILE_SIZE（256 KB）為上限動態配置，超過此大小的檔案會被截斷（不會溢位寫入）。
// 回傳值為指向下載資料的指標，實際下載到的位元組數記錄在全域變數 downloadedFileSize 中。
// 注意：呼叫端使用完畢後需要自行 free() 這塊記憶體，此函式本身不負責釋放。
// ============================================================
uint8_t* downloadTelegramFile(String filePath) {

  uint8_t* voiceFile = (uint8_t*)malloc(MAX_FILE_SIZE);
  if (!voiceFile) return NULL;  // 記憶體配置失敗，回傳空指標，呼叫端需自行檢查

  downloadedFileSize = 0;
  WiFiSSLClient client;

  if (client.connect("api.telegram.org", 443)) {

    client.println("GET /file/bot" + telegrambotToken + "/" + filePath + " HTTP/1.0");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();

    String header    = "";
    long   startTime = millis();

    // 先讀取並丟棄 HTTP 標頭部分，直到偵測到標頭結尾的空白行（\r\n\r\n）
    while (client.connected() || client.available()) {
      if (millis() - startTime > 10000) break;
      if (client.available()) {
        char c = client.read();
        header += c;
        if (header.endsWith("\r\n\r\n")) break;
      }
    }

    startTime = millis();
    // 逐位元組讀取檔案主體內容，直接寫入緩衝區（不經過字串物件，避免大檔案處理時的額外記憶體開銷）
    while ((client.connected() || client.available()) &&
           downloadedFileSize < MAX_FILE_SIZE) {  // 嚴格限制不超過緩衝區容量，避免緩衝區溢位
      if (millis() - startTime > 10000) break;
      if (client.available()) {
        voiceFile[downloadedFileSize++] = client.read();
        startTime = millis();
      }
    }

    client.stop();
  }

  return voiceFile;
}

// ============================================================
// 【工具函式】依 file_id 查詢 Telegram 檔案的實際下載路徑
// 用途：Telegram Bot API 的兩段式檔案下載流程第一步——
// 語音/圖片訊息只會先給一個 file_id，需要額外呼叫 getFile 端點換取真正的 file_path，
// 才能在 downloadTelegramFile() 中組出完整下載網址。
// ============================================================
String getTelegramFilePath(String fileId) {

  WiFiSSLClient client;
  String filePath = "";
  String getAll = "", getBody = "";

  if (client.connect("api.telegram.org", 443)) {

    client.println("GET /bot" + telegrambotToken +
                   "/getFile?file_id=" + fileId + " HTTP/1.1");
    client.println("Host: api.telegram.org");
    client.println("Connection: close");
    client.println();

    int     waitTime  = 5000;
    long    startTime = millis();
    boolean state     = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (client.available()) {
        char c = client.read();

        if (c == '\n') {
          if (getAll.length() == 0) state = true;
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state == true) getBody += String(c);

        startTime = millis();
      }

      if (getBody.length() > 0) break;
    }

    DynamicJsonDocument doc(8192);
    deserializeJson(doc, getBody);  // 注意：這裡沒有檢查 DeserializationError，若解析失敗 filePath 會維持空字串（呼叫端會據此判斷下載失敗）
    filePath = doc["result"]["file_path"].as<String>();
  }

  return filePath;
}

// ============================================================
// 【核心函式】Telegram Long Polling 主迴圈：接收並處理使用者訊息
// 用途：這是 Telegram Bot 版本的訊息接收核心，採用 Long Polling 方式
//（持續呼叫 getUpdates，limit=1&offset=-1 表示「只要最新的一則訊息」），
// 而非 Webhook 方式（不需要對外開放的公開網址，適合部署在家用網路 NAT 後面的嵌入式裝置）。
//
// 這個函式同時身兼兩個重要職責：
//   1. 訊息接收與分派：判斷收到的是文字指令、一般聊天，還是語音訊息，分別導向不同處理路徑
//   2. 時間來源之一：從 HTTP 回應標頭擷取 Date 欄位（getTime），
//      這個時間戳記完全搭載在「本來就必要」的 Telegram 通訊上，不需要額外的網路請求成本，
//      與是否使用 Gemini/OpenAI/Grok 無關（詳見 README 對此設計的說明）
//
// id_last == 0 的特殊處理：裝置每次重新連線後的「第一次」輪詢結果會被視為「舊訊息」而捨棄
//（message_id 被強制設為 0，不會進入實際處理分支）——這是避免裝置重開機/斷線重連後，
// 把「斷線期間堆積的舊訊息」當成新訊息重新處理一次的關鍵防呆機制。
// ============================================================
void getTelegramMessage() {

  const char* myDomain  = "api.telegram.org";
  String      getAll    = "";
  String      getTime   = "";
  String      getBody   = "";

  JsonObject          obj;
  DynamicJsonDocument doc(8192);

  String text        = "";
  String voiceFileId = "";
  long   message_id  = 0;

  // botClient 是全域共用的長連線物件，只有在尚未連線時才重新建立連線，
  // 避免每次輪詢都重新握手 TLS（成本較高），改用長連線重複發送請求
  if (!botClient.connected()) {
    if (lastMessageId == 0)
      Serial.println("Connect to " + String(myDomain));

    if (!botClient.connect(myDomain, 443))
      return;

    if (lastMessageId == 0)
      Serial.println("Connection successful");
  }

  while (botClient.connected()) {

    getAll  = "";
    getTime = "";
    getBody = "";

    // getUpdates 帶 offset=-1 只取得最新一筆待處理訊息（配合 lastMessageId 判斷是否為「新」訊息）
    String request = "limit=1&offset=-1&allowed_updates=message";

    botClient.println("POST /bot" + telegrambotToken + "/getUpdates HTTP/1.1");
    botClient.println("Host: "           + String(myDomain));
    botClient.println("Content-Length: " + String(request.length()));
    botClient.println("Content-Type: application/x-www-form-urlencoded");
    botClient.println("Connection: keep-alive");  // 保持連線不關閉，供下一輪迴圈重複使用（Long Polling 精神）
    botClient.println();
    botClient.print(request);

    int           waitTime    = 5000;
    unsigned long startTime   = millis();
    bool          state       = false;
    bool          dataReceived = false;

    while ((startTime + waitTime) > millis()) {
      vTaskDelay(100 / portTICK_PERIOD_MS);

      while (botClient.available()) {
        char c = botClient.read();

        if (c == '\n') {
          if (getAll.length() == 0)
            state = true;  // 空白行，代表 HTTP 標頭區段已結束
          getAll = "";
        } else if (c != '\r') {
          getAll += String(c);
        }

        if (state) {
          getBody += String(c);  // 標頭結束後，累積實際的 JSON body
        } else {
          // 尚在標頭區段內，順便擷取 Date 標頭作為時間來源（與是否使用哪家 LLM 供應商完全無關）
          if      (getTime.indexOf("Date:") != -1)
            getTime  = "";
          else if (getTime.indexOf("Content-Type") != -1)
            getTime += "";
          else
            getTime += String(c);
        }

        startTime = millis();
      }

      if (getBody.length() > 0) {
        dataReceived = true;
        break;
      }
    }

    getTime.replace("Content-Type", "");  // 清除可能誤夾帶進來的下一個標頭名稱殘留字元

    String workId = String(taskTags[1]) + " " + getTime;  // 預設先用 Telegram 回應時間戳記組成 workId（若後續判斷為有效新訊息會被覆寫成 RTC 時間）

    if (!dataReceived || getBody == "") return;  // 逾時或空回應，直接結束本輪，等待下次呼叫再重試

    DeserializationError err = deserializeJson(doc, getBody);
    if (err) {
      Serial.println("[DEBUG] JSON parse failed: (getTelegramMessage)\n" + getBody);
      return;
    }
    obj = doc.as<JsonObject>();

    message_id = obj["result"][0]["message"]["message_id"].as<long>();

    if (message_id && message_id != lastMessageId) {  // 有訊息，且與上次處理過的不是同一則（避免重複處理）

      long id_last  = lastMessageId;
      lastMessageId = message_id;  // 立刻更新，即使本次訊息最終被判定為「首次連線後的舊訊息」而略過，也要記錄避免下輪重複判斷
	  
      // 安全檢查：只處理來自預先設定的 chatId 的訊息，防止其他人（若 Bot Token 外洩）冒用本 Bot 發送指令
      String fromChatId = obj["result"][0]["message"]["chat"]["id"].as<String>();
      if (fromChatId != telegrambotChatId) {
        return;
      }	  	  

      if (id_last == 0) {
        // 裝置剛開機/剛重新連線後的第一次輪詢——這則訊息很可能是斷線期間累積的舊訊息，
        // 強制將 message_id 歸零，讓它不會進入下方的實際處理分支，只是單純「同步」到最新狀態
        message_id = 0;

      } else {
        // 真正的新訊息，改用精準的硬體 RTC 時間重新組裝 workId（比 HTTP Date 標頭更精確）
        workId = String(taskTags[1]) + " " + getRtcTimeString();

        if (obj["result"][0]["message"].containsKey("text")) {
          // 文字訊息分支
          text = obj["result"][0]["message"]["text"].as<String>();

          if (text == "help") {
            executeTool(workId, "/help", JsonObject());  // 純文字 "help"（非以 / 開頭）也視為求助指令的別名

          } else if (text == "null") {
            botClient.stop();  // 特殊除錯用途：手動送出 "null" 可強制中斷目前的長連線（用於測試重連機制）

          } else if (text.startsWith("/")) {
            executeTool(workId, text, JsonObject());  // 以 / 開頭視為內建系統指令（/help、/getMemory 等），直接分派，不經過 LLM

          } else {
            // 一般自然語言訊息，送給目前設定的 LLM 供應商處理，取得的回應再交給 handleAgentResponse 解讀
            text = llmChatRequest(workId, text);
            handleAgentResponse(workId, text);
          }

          // 每次處理完文字訊息後，把最新的完整對話歷史寫回 memory.md，確保重開機後仍能接續對話
          String localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory);

        } else if (doc["result"][0]["message"].containsKey("voice")) {
          // 語音訊息分支：先取得 file_id，換取實際檔案路徑，下載原始音訊，再送去轉錄
          voiceFileId = doc["result"][0]["message"]["voice"]["file_id"].as<String>();

          String   filePath  = getTelegramFilePath(voiceFileId);
          uint8_t* voiceFile = downloadTelegramFile(filePath);

          if (voiceFile && downloadedFileSize > 0) {
            // 呼叫多供應商 STT 派發層，取得轉錄後的純文字
            text = sendFileToLlm(
              voiceFile, downloadedFileSize,
              "audio/ogg; codecs=opus",
              "Transcribe this audio to text exactly as spoken."
            );

            // 轉錄後的文字走與一般文字訊息完全相同的處理管線：
            // 以 / 開頭視為指令，否則送給 LLM 進行一般對話理解，
            // 語音輸入與文字輸入之後沒有任何特殊分支差異
            if (text.startsWith("/")) {
              executeTool(workId, text, JsonObject());
            } else {
              text = llmChatRequest(workId, text);
              handleAgentResponse(workId, text);
            }
          }

          if (voiceFile)
            free(voiceFile);  // downloadTelegramFile() 配置的緩衝區，使用完畢後在此釋放，避免記憶體洩漏

          String localHistory2 = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localHistory2 = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(memoryFilename, localHistory2);
        }
      }
    }

    vTaskDelay(5 / portTICK_PERIOD_MS);  // 短暫讓出 CPU，避免這個內層 while 迴圈忙迴圈佔用過多處理器資源
  }

  // 跳出上面的 while (botClient.connected()) 迴圈，代表連線已中斷（可能是網路異常），
  // 這裡負責偵測 WiFi 連線狀態並在斷線時主動重新連線，確保裝置能自我恢復，不需要人工重啟
  while (WiFi.status() != WL_CONNECTED) {
    WiFi.disconnect();
    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());

    unsigned long start = millis();

    while (WiFi.status() != WL_CONNECTED && millis() - start < 15000)  // 最長等待 15 秒讓 WiFi 重新連線
      vTaskDelay(500 / portTICK_PERIOD_MS);
  }

}

// ============================================================
// 【FreeRTOS Task】主要 HTTP 設定伺服器（監聽 81 port）
// 用途：這是整個 Web 管理介面（設定頁、Agent 頁、聊天頁、排程頁）與 API 端點
//（/updateConfig、/message、/getScheduleTasks 等）的統一入口。
// 因為 Arduino 環境沒有高階 HTTP 伺服器框架，這裡採用最原始的手法：
// 逐字元讀取 HTTP 請求的第一行（GET 開頭那行），比對 URL 路徑字串前綴，手動分派到對應的處理邏輯。
//
// 整體運作邏輯：
//   1. 持續等待新的用戶端連線（server.available()）
//   2. 逐字元讀取請求內容，累積成 currentLine，直到偵測到請求行的結束（\n）
//   3. 依 currentLine 的內容比對是哪個路徑（GET / 、GET /updateConfig?... 等），
//      把對應的回應內容準備好放進全域變數 mainPageHTML
//   4. 讀到 HTTP 標頭區段結束的空白行時（currentLine.length() == 0），
//      統一在這裡組裝並送出 HTTP 回應（把 mainPageHTML 的內容當作回應本文），
//      這種「先準備內容、最後統一送出」的寫法讓每個路徑的處理邏輯只需要專心設定 mainPageHTML 即可，
//      不需要各自處理 HTTP 回應標頭的組裝細節
//
// 這個 Task 是無窮迴圈（while(1)），是 FreeRTOS 排程器管理的多個平行任務之一。
// ============================================================
void task_getRequest(void *param) {
  (void)param;  // 未使用的參數，明確標示避免編譯器警告
  while (1) {
    
    WiFiClient client = server.available();  // 檢查是否有新的用戶端連線進來

    if (client) {
      String currentLine = "";
      
      while (client.connected() || client.available()) { 
        
        if (client.available()) {
          char c = client.read();

          if (c == '\n') {
            if (currentLine.length() == 0) {
              // 收到空白行，代表 HTTP 請求標頭區段已經結束，開始組裝並送出回應
            
              String pageToSend = mainPageHTML;  // 先取出目前累積好的回應內容
              mainPageHTML = "";  // 立刻清空全域變數，避免殘留內容污染下一次請求
            
              client.println("HTTP/1.1 200 OK");
              client.println("Content-Type: text/html; charset=utf-8");
              client.println("Content-Length: " + String(pageToSend.length()));
              client.println("Access-Control-Allow-Origin: *");  // 允許跨來源請求（CORS），方便從其他網頁前端呼叫這些 API
              client.println("Cache-Control: no-cache");
              client.println("Connection: close");
              client.println();
            
              // 分段傳送回應內容（每次最多 512 bytes），避免大型頁面一次性寫入造成緩衝區壓力
              const char* ptr = pageToSend.c_str();
              int total  = pageToSend.length();
              int sent   = 0;
              while (sent < total) {
                int chunk   = (total - sent) > 512 ? 512 : (total - sent);
                int written = client.write((const uint8_t*)(ptr + sent), chunk);
                if (written > 0) sent += written;
                else delay(5);  // 寫入暫時失敗（緩衝區滿），短暫等待後重試，而非直接放棄
              }
              client.flush();
            
              break;  // 回應已送出，跳出內層讀取迴圈，準備關閉這個連線

            } else {
              currentLine = "";  // 一般標頭行結束，清空準備讀取下一行
            }
          }
          else if (c != '\r') {
            currentLine += c;  // 忽略 \r，只累積實際內容字元
          }
          
          // ------------------------------------------------------------
          // 以下依序比對 currentLine 是否符合各個已知路徑，逐一列舉路由規則
          // ------------------------------------------------------------

          if (currentLine.startsWith("GET / ") && currentLine.endsWith(" HTTP/1.")) {
            // 根路徑：回傳主設定頁，並把頁面中的預留字串（如 "wifiSsid"）替換成目前實際的設定值，
            // 讓瀏覽器打開頁面時能直接看到目前的裝置設定（含目前的 llmType/llmKey/llmModel）
            mainPageHTML = getStringFromFile(configpageFilename);
			
            mainPageHTML.replace("deviceName", deviceName);
            mainPageHTML.replace("wifiSsid", wifiSsid);
            mainPageHTML.replace("wifiPassword", wifiPassword);
            mainPageHTML.replace("telegrambotToken", telegrambotToken);
            mainPageHTML.replace("telegrambotChatId", telegrambotChatId);
            mainPageHTML.replace("scheduleTimeout", String(scheduleTimeout));            
            mainPageHTML.replace("llmType", llmType);            
            mainPageHTML.replace("llmKey", llmKey);
            mainPageHTML.replace("llmModel", llmModel);
            mainPageHTML.replace("timeZone", timeZone);		

            currentLine = "";            
          }
          else if (currentLine.startsWith("GET /updateConfig?") && currentLine.endsWith(" HTTP/1.")) {
            // 接收網頁介面送出的新設定（含 llmType/llmKey/llmModel 等），寫入 SD 卡並立即套用到全域變數
            String workId = String(taskTags[0]) + " " + getRtcTimeString();
            
            currentLine = urldecode(currentLine);  // 查詢字串是 URL 編碼過的，先解碼還原成原始 JSON 字串
            currentLine.replace("GET /updateConfig?", "");
            currentLine.replace(" HTTP/1.", "");
            
            if (currentLine.startsWith("{") && currentLine.endsWith("}")) {
              storeDataToFile(envFilename, currentLine);  // 持久化寫入 env.json
			  setEnvironmentSettings(currentLine);        // 立即套用到執行期全域變數，不需重開機生效
			  
              // 硬體定義需要重新併入裝置名稱與時區資訊（因為這兩個值可能剛被使用者改過）
              devicesDefinitionFinal = devicesDefinition;
              devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
              devicesDefinitionFinal += "\nDevice timezone: " + timeZone;
			  
			  systemContentReset();  // 重新組裝 system prompt，讓新設定（例如換了 LLM 供應商）立即生效
			
              mainPageHTML = "Configuration updated successfully.";
			}
            else
              mainPageHTML = "Configuration updated failed. JSON parse failed.";
		  
			currentLine = "";
			
          }
          else if (currentLine.startsWith("GET /agent") && currentLine.endsWith(" HTTP/1.")) {
            // Agent 性格/技能設定頁
            mainPageHTML = getStringFromFile(agentpageFilename);

            currentLine = "";

          }
          else if (currentLine.startsWith("GET /getSoul") && currentLine.endsWith(" HTTP/1.")) {
            // 查詢目前的人格設定內容（soul.md 內容，已載入記憶體的 llmRole）
            mainPageHTML = llmRole;

            currentLine = "";

          }
          else if (currentLine.startsWith("GET /updateSoul?") && currentLine.endsWith(" HTTP/1.")) {
            // 更新人格設定：寫入 soul.md 並重建 system prompt
            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateSoul?", "");
            currentLine.replace(" HTTP/1.", "");
            
            storeDataToFile(soulFilename, currentLine);
            llmRole = currentLine;
			
            systemContentReset();
			
            mainPageHTML = "Soul updated successfully.";
            
            currentLine = "";  

          }				  
          else if (currentLine.startsWith("GET /getDevice") && currentLine.endsWith(" HTTP/1.")) {
            // 查詢目前的硬體腳位定義（device.md 內容）
            mainPageHTML = devicesDefinition;

            currentLine = "";

          }
		  else if (currentLine.startsWith("GET /updateDevice?") && currentLine.endsWith(" HTTP/1.")) {
            // 更新硬體腳位定義（白名單）：寫入 device.md 並重建 system prompt，
            // 讓新的硬體對應規則立即生效（不需重開機）
            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateDevice?", "");
            currentLine.replace(" HTTP/1.", "");
            
            storeDataToFile(deviceFilename, currentLine);
            devicesDefinition = currentLine;
			
            devicesDefinitionFinal = devicesDefinition;
            devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
            devicesDefinitionFinal += "\nDevice timezone: " + timeZone;
			
			systemContentReset();
			
            mainPageHTML = "Device updated successfully.";
			
            currentLine = "";

          }
          else if (currentLine.startsWith("GET /getSkill") && currentLine.endsWith(" HTTP/1.")) {
            // 查詢目前的技能巨集定義（skill.md 內容）
            mainPageHTML = skillsDefinition;

            currentLine = "";

          } 		  
          else if (currentLine.startsWith("GET /updateSkill?") && currentLine.endsWith(" HTTP/1.")) {
            // 更新技能巨集定義：寫入 skill.md 並重建 system prompt
            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateSkill?", "");
            currentLine.replace(" HTTP/1.", "");
            
            storeDataToFile(skillFilename, currentLine);
			skillsDefinition = currentLine;
			
			systemContentReset();
			
            mainPageHTML = "Skill updated successfully.";
            
            currentLine = "";

          }		  
          else if (currentLine.startsWith("GET /chat") && currentLine.endsWith(" HTTP/1.")) {
            // 網頁聊天介面（HTTP 版本）
            mainPageHTML = getStringFromFile(chatpageFilename);

            currentLine = "";

          }
          else if (currentLine.startsWith("GET /schedule") && currentLine.endsWith(" HTTP/1.")) {
            // 排程管理頁面；若排程檔案不存在或是空的，回傳空陣列字面值，避免前端 JavaScript 解析 undefined 而出錯
            mainPageHTML = getStringFromFile(schedulepageFilename);
			if (mainPageHTML == "")
				mainPageHTML = "[]";
			
            currentLine = "";

          }
          else if (currentLine.startsWith("GET /getScheduleTasks") && currentLine.endsWith(" HTTP/1.")) {
            // 提供目前排程資料的 API 端點（供排程管理頁面的 JavaScript 以 AJAX 方式讀取）
            if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
              mainPageHTML = scheduleTasks;
              xSemaphoreGive(stateMutex);
            }

            currentLine = "";

          }                                            
          else if (currentLine.startsWith("GET /updateScheduleTasks?") && currentLine.endsWith(" HTTP/1.")) {
            // 從排程管理頁面直接送出整批修改後的排程 JSON 陣列，覆寫既有排程
            String workId = String(taskTags[0]) + " " + getRtcTimeString();
            
            currentLine = urldecode(currentLine);
            currentLine.replace("GET /updateScheduleTasks?", "");
            currentLine.replace(" HTTP/1.", "");
            
            if (currentLine.startsWith("[") && currentLine.endsWith("]")) {
              storeDataToFile(scheduleFilename, currentLine);

              mainPageHTML = "Schedule updated successfully.";

              if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
                scheduleTasks = currentLine;
                // 這次的排程修改動作也記錄進對話歷史，讓 LLM 之後若被問及排程狀態時能有上下文可循
                historicalMessages += buildLlmMessage("user", "GET /updateScheduleTasks?<NEW SCHEDULE TASKS>");
                historicalMessages += buildLlmMessage("model", mainPageHTML);
                String localHistory = historicalMessages;
                xSemaphoreGive(stateMutex);
                storeDataToFile(memoryFilename, localHistory);
              }
            }
            else
              mainPageHTML = "Schedule updated failed. JSON parse failed.";

            currentLine = "";        
            
          }
          else if (currentLine.startsWith("GET /message?") && currentLine.endsWith(" HTTP/1.")) {
            // 網頁聊天介面（與 /tcpSendMessage 的目標端點相同）的訊息接收入口，
            // 是網頁聊天、Agent 對 Agent 通訊共用的統一訊息處理端點
            mainPageStatus = true;  // 標記目前正在處理網頁訊息中（可供其他流程判斷是否應暫緩存取 mainPageHTML）

            mainPageHTML = "";
            
            String workId = String(taskTags[0]) + " " + getRtcTimeString();       

            currentLine.replace("GET /message?", "");
            currentLine.replace(" HTTP/1.", "");

            if (currentLine != "") {
              currentLine = urldecode(currentLine);           
      				
    				if (currentLine.startsWith("/")) 
    				  executeTool(workId, currentLine, JsonObject());  // 以 / 開頭視為系統指令
    				else {
    				  currentLine = llmChatRequest(workId, currentLine);  // 一般訊息送給 LLM 處理
    				  handleAgentResponse(workId, currentLine);
    				}

              String localHistory = "";
              if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
                localHistory = historicalMessages;
                xSemaphoreGive(stateMutex);
              }
              storeDataToFile(memoryFilename, localHistory);
            }
            
            mainPageStatus = false;  // 處理完畢，解除標記

            currentLine = "";

    			}      
        }
		else {
          vTaskDelay(1);  // 目前沒有新資料可讀，短暫讓出處理器，避免忙迴圈
        }
      }

      client.stop();  // 本次連線處理完畢，關閉連線（配合上面 HTTP 標頭的 "Connection: close" 一致）
    }
    else {
      vTaskDelay(5 / portTICK_PERIOD_MS);  // 沒有新的用戶端連線，稍微延遲後再檢查，避免無意義地忙碌輪詢
    }
  }
}

// ============================================================
// 【FreeRTOS Task】相機串流伺服器（監聽 82 port）
// 用途：提供 MJPEG（Motion JPEG）格式的即時影像串流，讓使用者能在瀏覽器中直接看到連續畫面
//（而非只有 /still 那種單張快照）。採用 multipart/x-mixed-replace 格式——
// 每一張新影格都以 boundary 分隔持續推送給用戶端，瀏覽器原生就能把這種格式當成連續影片播放。
// ============================================================
void task_getRequestStream(void *param) {
  (void)param;
  
  while (1) {
    
    WiFiClient client = serverStream.available();
    uint32_t img_addr = 0;
    uint32_t img_len = 0;	
    
    if (client) {
      String currentLine = "";

      while (client.connected() || client.available()) { 

        if (client.available()) {
          char c = client.read();
          if (c == '\n') {
            if (currentLine.length() == 0) {
             // HTTP 標頭讀取完畢，開始進入持續推流迴圈
             String head = "--Taiwan\r\nContent-Type: image/jpeg\r\n\r\n";
            client.println("HTTP/1.1 200 OK");
            client.println("Access-Control-Allow-Origin: *");
            client.println("Connection: keep-alive");  // 保持連線持續開啟，才能不斷推送新影格
            client.println("Content-Type: multipart/x-mixed-replace; boundary=Taiwan");  // 瀏覽器辨識此格式後會自動以「連續替換影像」方式顯示
            client.println();
            while(client.connected()) {
              // 只要用戶端還連著，就不斷擷取新影格並送出，形成持續串流
              Camera.getImage(0, &img_addr, &img_len);
              uint8_t *fbBuf = (uint8_t*)img_addr;
              size_t fbLen = img_len;
              client.print(head);  // 每一影格前都要重複送出 boundary + Content-Type 標頭
              for (size_t n=0;n<fbLen;n=n+1024) {
                  if (n+1024<fbLen) {
                    client.write(fbBuf, 1024);
                    fbBuf += 1024;
                }
                else {
                  size_t remainder = fbLen - n;
                  if (remainder > 0)
                    client.write(fbBuf, remainder);
                }
              }
              client.print("\r\n");
              
              vTaskDelay(10 / portTICK_PERIOD_MS);  // 控制推流速率，避免佔滿頻寬與 CPU（間接決定串流的大略 FPS 上限）
            }
            break;
            } else {
              currentLine = "";
            }
          }
          else if (c != '\r') {
            currentLine += c;
          }

          if (currentLine.indexOf(" HTTP/1.")!=-1) {
            currentLine="";  // 此伺服器不需要判斷不同路徑（只提供單一串流功能），偵測到請求行結束就直接清空即可
          }
        }
        else {
          vTaskDelay(1); 
        }
      }
      client.stop();
    }
    else {
      vTaskDelay(5 / portTICK_PERIOD_MS);
    }
  }
}

// ============================================================
// 【FreeRTOS Task】Telegram 訊息輪詢排程器
// 用途：這個 Task 是一個簡單的排程外殼——每隔 1 秒呼叫一次 getTelegramMessage()。
// 因為 botClient 是全域共用資源（可能同時被其他流程存取，如 /telegramSendMessage 工具），
// 這裡用 botClientMutex 保護整段呼叫過程，確保同一時間只有一個地方在操作這個連線物件。
// ============================================================
void task_getTelegramMessage(void *param) {
  (void)param;
  while (1) {

// ============================================================
// 【工具函式】判斷某週期性任務今天是否已經執行過
// 用途：週期性排程任務（例如「每天早上 8 點提醒澆花」）在同一天內只應該觸發一次，
// 這個函式用一個簡單的字串包含比對（以 | 符號當作分隔標記）來檢查該任務是否已在今天執行過。
// 若偵測到日期已經換日（executedTodayDate 與今天不同），就清空整個紀錄重新開始計算，
// 這是避免「今日已執行」清單無限增長、也確保跨日後週期任務能重新觸發的關鍵機制。
// ============================================================
bool isExecutedToday(String task) {

  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  int today = now->tm_mday;  // 取得今天的日期數字（1-31）

  if (executedTodayDate != today) {
    // 日期已經改變（跨日），清空舊紀錄重新開始
    executedTodayTasks = "";
    executedTodayDate = today;
  }

  return executedTodayTasks.indexOf("|" + task + "|") != -1;  // 以 | 包夾比對，避免子字串誤判（例如任務 A 剛好是任務 B 名稱的一部分）
}

// ============================================================
// 【工具函式】標記某週期性任務今天已經執行過
// 用途：與 isExecutedToday() 搭配使用，任務執行完成後呼叫此函式記錄，避免同一天內重複觸發。
// ============================================================
void markExecutedToday(const String &task) {
  time_t rawtime;
  time(&rawtime);
  struct tm *now = localtime(&rawtime);
  executedTodayDate = now->tm_mday;
  executedTodayTasks += "|" + task + "|";
}

// ============================================================
// 【核心函式】篩選出所有「尚未完成」的排程任務，格式化成可讀文字
// 用途：對應 /getUnfinishedSchedule 指令，純本地端邏輯（不經過 LLM），
// 逐一檢查每筆任務的 executed 欄位，並額外處理週期性任務（year==0）今天是否已執行的情況，
// 把符合條件的任務組成一段人類可讀的文字（含時間戳記與任務描述）回傳。
// ============================================================
String getUnfinishedScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);
  
  if (err) {
      Serial.println("Schedule JSON parse failed");
      return "[]";
  }
  
  JsonArray tasks = doc.as<JsonArray>();

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();  // 宣告但實際上此函式最終用字串拼接輸出，這個陣列變數目前未被使用（保留供未來擴充或除錯用途）

  String result = "";
  
  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();
       
      if (executed) continue;  // 已標記完成的任務直接跳過
  
      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();
                
      String scheduleStr;
      serializeJson(task["schedule"], scheduleStr);
      String compareTask = scheduleStr + " " + task["task"].as<String>();  // 組合出可唯一識別這筆任務的比對字串

      // year == 0 代表週期性任務，若今天已經執行過就不列入「未完成」清單（因為它今天其實已經觸發過了）
      if (year == 0 && isExecutedToday(compareTask))
          continue;

      result += "*** " + String(year) + "/" + String(month) + "/" + String(day) + " " + String(hour) + ":" + String(minute) + ":" + String(sec) + " ***\n" + task["task"].as<String>() + "\n";
  }
  
  if (result == "") result = "NULL";  // 沒有任何未完成任務時，明確回傳 "NULL" 字串（而非空字串），讓呼叫端容易判斷

  return result;
}

// ============================================================
// 【核心函式】篩選出目前「已到期、應該執行」的排程任務
// 用途：這是排程系統的核心判斷邏輯，由 task_time_scheduling() 定期呼叫，
// 逐一檢查每筆任務的預定時間是否已經到達（或剛過去但仍在容錯時間內），
// 篩選出這次應該被觸發執行的任務清單（回傳 JSON 陣列字串）。
//
// 關鍵邏輯——時間欄位為 0 代表「萬用」比對：
//   year==0  -> 不指定年份，代表每年重複（配合 month/day 可組成「每年固定某天」的任務）
//   month==0 -> 不指定月份，代表每月重複
//   day==0   -> 不指定日期，代表每天重複
//   若 year/month/day 皆為 0，就是最常見的「每天固定時間」型任務
// 這三個欄位為 0 時，會直接代入「今天的年/月/日」進行比對，讓任務能在對應的週期時間點觸發。
//
// 逾時容錯機制（scheduleTimeout）：
//   若目前時間已經超過任務預定時間，但差距在 scheduleTimeout 分鐘之內，仍視為「應該執行」
//  （因為裝置可能短暫離線、或這個檢查迴圈剛好晚了幾秒才跑到），
//   但若差距超過這個容錯範圍，就視為「已經錯過，不再補跑」，直接跳過，避免任務在數小時後才被誤觸發。
// ============================================================
String getExecuteScheduleTasksJson(const String &scheduleTasksJson) {
  DynamicJsonDocument doc(16384);
  DeserializationError err = deserializeJson(doc, scheduleTasksJson);
  
  if (err) {
      Serial.println("Schedule JSON parse failed");
      return "[]";
  }
  
  JsonArray tasks = doc.as<JsonArray>();
  
  long long epoch = rtc.Read();  // 從硬體 RTC 讀取目前真實時間，作為比對基準
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);

  DynamicJsonDocument resultDoc(8192);
  JsonArray resultArray = resultDoc.to<JsonArray>();

  String result = "";
  
  for (JsonObject task : tasks) {
      bool executed = task["executed"].as<bool>();
       
      if (executed) continue;
  
      JsonObject schedule = task["schedule"];
      int year   = schedule["year"].as<int>();
      int month  = schedule["month"].as<int>();
      int day    = schedule["day"].as<int>();
      int hour   = schedule["hour"].as<int>();
      int minute = schedule["minute"].as<int>();
      int sec    = schedule["second"].as<int>();

      // 欄位為 0 時代入「今天」的實際年/月/日，實現週期性任務的比對邏輯
      int resolvedYear  = (year  == 0) ? (now->tm_year + 1900) : year;
      int resolvedMonth = (month == 0) ? (now->tm_mon  + 1)    : month;
      int resolvedDay   = (day   == 0) ? (now->tm_mday)        : day;

      struct tm tmTask = {};
      tmTask.tm_year = resolvedYear  - 1900;
      tmTask.tm_mon  = resolvedMonth - 1;
      tmTask.tm_mday = resolvedDay;
      tmTask.tm_hour = hour;
      tmTask.tm_min  = minute;
      tmTask.tm_sec  = sec;

      time_t taskTime = mktime(&tmTask);  // 把任務的預定時間轉換成 epoch 時間戳記，方便與目前時間直接比較大小
      
      if (mktime(&tmTask) <= rawtime) {  // 預定時間已經到達（或已經過去）
        time_t diff = rawtime - taskTime;  // 已經超過預定時間多久（秒）
        
        // 超過容錯時間（scheduleTimeout 分鐘），視為已經錯過此次觸發時機，不再補跑，避免延遲數小時才誤觸發
        if ((scheduleTimeout > 0 && rawtime > taskTime && diff > (time_t)(scheduleTimeout * 60)))
            continue;
                
        String scheduleStr;
        serializeJson(task["schedule"], scheduleStr);
        String compareTask = scheduleStr + " " + task["task"].as<String>();

        // 週期性任務（year==0）今天若已經觸發過，就不重複觸發
        if (year == 0 && isExecutedToday(compareTask))
            continue;

        // 通過以上所有檢查，這筆任務確定要在本次排程週期中被執行，加入結果陣列
        JsonObject item = resultArray.createNestedObject();
        item["task"] = task["task"].as<String>();
        item["schedule"] = schedule;

      }
  }

  serializeJson(resultDoc, result);  // 把篩選結果序列化成 JSON 字串
  return result;
}

// ============================================================
// 【FreeRTOS Task】排程觸發引擎
// 用途：每 60 秒（6 次 x 10 秒延遲）檢查一次是否有排程任務到期需要執行，
// 是整個「時間到自動觸發動作」機制的主迴圈。
//
// 執行流程：
//   1. 每輪先中斷 Telegram 長連線（與防盜偵測相同的考量：避免搶用連線資源）
//   2. 確認 RTC 是否已經初始化（rtcYear==0 代表尚未校時），若尚未校時則先嘗試校時，
//      仍失敗就跳過本輪（continue），避免用未校正的錯誤時間去判斷任務是否到期
//   3. 呼叫 getExecuteScheduleTasksJson() 篩選出「本輪應該觸發」的任務清單
//   4. 逐一處理每筆到期任務：
//      - 若任務內容本身就是合法的 tool_call JSON（LLM 當初判斷「不需臨場推理」而直接存了結構化指令），
//        直接呼叫 handleAgentResponse() 執行，完全不需要再打一次 API（省 token、省延遲、離線也能跑）
//      - 若任務內容是自然語言描述（需要臨場推理才能決定怎麼做），
//        才組裝提示詞呼叫 LLM，讓它即時判斷該怎麼執行
//   5. 執行完畢後標記今日已執行（週期性任務用），並呼叫 /updateScheduleStatus 把
//      一次性任務的 executed 狀態同步寫回排程檔案，避免重複觸發
//
// 這個設計正是 README 中提到的「排程任務可離線觸發」機制的實際落地：
// 一次性、明確的動作（如「7:30 開燈」）在建立當下就已經固化成 tool_call JSON，
// 到期觸發時完全不需要網路連上 LLM，只有需要臨場判斷的任務才會消耗一次 API 呼叫。
// ============================================================
void task_time_scheduling(void *param) {
  (void)param;
  while (1) {
    
    // 分段延遲累積 60 秒（6 x 10 秒），作為排程檢查的輪詢週期
    for (int i = 0; i < 6; i++) {
      vTaskDelay(10000 / portTICK_PERIOD_MS);
      
    }
    
    if (xSemaphoreTake(botClientMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      botClient.stop();
      xSemaphoreGive(botClientMutex);
    }
    vTaskDelay(2000 / portTICK_PERIOD_MS);

    String workId = String(taskTags[3]) + " " + rtcFormatTime;  // taskTags[3] = "<TIME_SCHEDULING>"，標記本次事件來源

    if (rtcYear == 0) {
      // RTC 尚未初始化（開機後第一次進入此迴圈時常見的狀態），先嘗試立即校時
      Serial.println("[DEBUG] RTC time is not initialized.");
      executeTool(workId, "/syncrtc", JsonObject(), false);
      if (rtcYear == 0)
        continue;  // 校時仍然失敗（例如網路不通），本輪直接放棄，等下一輪再試，避免用錯誤時間誤判任務是否到期
    }

    String localSchedule = "";
    if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
      if (scheduleTasks.startsWith("[") && scheduleTasks.indexOf("]") !=-1)
        scheduleTasks = scheduleTasks.substring(0, scheduleTasks.lastIndexOf("]") + 1);
      localSchedule = scheduleTasks;
      xSemaphoreGive(stateMutex);
    }

    if (localSchedule.startsWith("[") && localSchedule.indexOf("]") !=-1) {
 
      String unfinishedScheduleTasksJson = getExecuteScheduleTasksJson(localSchedule);  // 篩選出本輪應該觸發的任務

      if (unfinishedScheduleTasksJson.startsWith("[") && unfinishedScheduleTasksJson.indexOf("]") !=-1) {
        unfinishedScheduleTasksJson = unfinishedScheduleTasksJson.substring(0, unfinishedScheduleTasksJson.lastIndexOf("]") + 1);

        String response = "";

        DynamicJsonDocument doc(8192);
      
        DeserializationError err = deserializeJson(doc, unfinishedScheduleTasksJson);
        if (err) {
          Serial.println("[DEBUG] JSON parse failed: (task_time_scheduling)\n" + unfinishedScheduleTasksJson);
          continue;
        }  

        JsonArray tasks = doc.as<JsonArray>();
        
        for (JsonObject obj : tasks) {

          String taskName = obj["task"].as<String>();

          String schedule = obj["schedule"].as<String>();
          String item = obj["task"].as<String>();           

          // 判斷任務內容本身是否已經是合法的 tool_call JSON 外觀（物件或陣列）
          if ((item.startsWith("{") && item.endsWith("}")) || (item.startsWith("[") && item.endsWith("]"))) {
			  // 已經是結構化指令，直接執行，完全不需要呼叫 LLM——這是離線觸發能力的關鍵
			  handleAgentResponse(workId, item);
          } 
		  else {
			  // 任務內容是自然語言描述，需要 LLM 臨場判斷並產生對應的 tool_call
			  String prompt =
				"This is a deterministic scheduling execution step. "
			  
				"\n\nUnfinished scheduled tasks:\n" +
				item +
			  
				"\n\nThe task list above already contains ONLY tasks that have not been executed. "
				"Evaluate EvERY task in this list independently. "
				"If a task's scheduled time is less than or equal to the current time, "
				"it MUST be executed immediately. "
				"Do NOT skip any eligible task. "
				"More than one task may be eligible at the same time. "
				"If multiple tasks are eligible, execute ALL of them in the same response. "
				"Tasks whose scheduled time is still in the future must be ignored. "
			  
				"Output rules: "
				"1. If no task is eligible for execution, return EXACTLY: NONE. "
				"2. If one or more tasks are eligible, return tool_call JSON for ALL eligible tasks. "
				"3. Never return natural language. "
				"4. Never explain. "
				"5. Never summarize. "
				"6. Never ask questions. "
				"7. Never claim success without tool execution results. "
				"8. Process every task in the provided task list. "
				"9. A task remains executable forever after its scheduled time has passed until it is marked executed=true. "
				"10. Do not stop after the first eligible task.";

			  response = llmChatRequest(workId, prompt);

			  handleAgentResponse(workId, response);
				 
          }	

          // 無論是離線直接執行、還是透過 LLM 執行，都要標記「今日已執行」，供週期性任務的重複觸發判斷使用
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            markExecutedToday(schedule + " " + item);
            xSemaphoreGive(stateMutex);
          }
        }
        
        if (tasks.size()>0) {
          // 本輪確實有任務被觸發過，才需要更新排程狀態並持久化，避免每輪都做不必要的寫入
          executeTool(workId, "/updateScheduleStatus", JsonObject(), false);

          String localExecuted = "";
          String localHistory = "";
          if (xSemaphoreTake(stateMutex, MUTEX_TIMEOUT_TICKS) == pdTRUE) {
            localExecuted = executedTodayTasks;
            localHistory  = historicalMessages;
            xSemaphoreGive(stateMutex);
          }
          storeDataToFile(scheduleExecutedTodayTasksFilename, localExecuted);
          storeDataToFile(memoryFilename, localHistory);
        }
      }
      
    }
  }
}

// ============================================================
// 【核心函式】初始化 WiFi 連線
// 用途：開機時執行的網路初始化流程，同時啟用兩種模式：
//   1. SoftAP 模式（enableConcurrent + apbegin）：裝置自己開一個熱點（fuclaw），
//      即使尚未設定家中 WiFi，使用者仍可用手機直接連上這個熱點進行初始設定
//   2. Station 模式（WiFi.begin）：嘗試以已儲存的帳密連上既有的 WiFi 網路
// 最多重試 2 次，每次最長等待 15 秒，任一次成功連線就提前跳出。
// 若 wifiSsid 為空字串（裝置從未設定過），直接跳過連線嘗試，只保留 SoftAP 模式可用。
// ============================================================
void initWiFi() {
  WiFi.enableConcurrent();  // 啟用 AP + Station 並行模式，讓裝置同時扮演熱點與客戶端兩種角色
  WiFi.apbegin((char*)apSsid.c_str(), (char*)apPassword.c_str(), channel_ap, 0);
    
  for (int i=0 ; i<2 ; i++) {

    if (wifiSsid == "")
      break;  // 尚未設定任何 WiFi 帳密，不需要嘗試連線，直接結束

    WiFi.begin((char*)wifiSsid.c_str(), (char*)wifiPassword.c_str());
    delay(1000);

    Serial.println();
    Serial.print("Connecting to ");
    Serial.println(wifiSsid);

    unsigned long StartTime=millis();

    while (WiFi.status() != WL_CONNECTED) {
      delay(500);

      if ((StartTime + 15000) < millis())
        break;  // 單次嘗試最長等待 15 秒
    }
    
    if (WiFi.status() == WL_CONNECTED)
      break;  // 連線成功，不需要進行第二次重試
  }
  
}

// ============================================================
// 【核心函式】套用環境設定 JSON 到執行期全域變數
// 用途：解析 env.json 的內容，並把每個欄位的值寫入對應的全域變數。
// 這是開機初始化與 /updateConfig 端點都會呼叫的共用函式，
// 確保「檔案內容」與「執行期記憶體中實際生效的設定」隨時保持一致。
// 特別注意 llm_type/llm_key/llm_model 三個欄位——這正是多供應商架構得以運作的資料來源。
// ============================================================
void setEnvironmentSettings(String jsonString) {
  
  DynamicJsonDocument doc(8192);
  DeserializationError error = deserializeJson(doc, jsonString);
  if (error) {
    Serial.println("[DEBUG] JSON parse failed : (setEnvironmentSettings)\n" + jsonString);
    return;  // 解析失敗，直接放棄本次套用，維持原本的舊設定值不變（安全的失敗方式，不會讓系統套用到一半的錯誤設定）
  }

  JsonObject obj = doc.as<JsonObject>();
  deviceName =  obj["device_name"].as<String>();  
  wifiSsid =  obj["wifi_ssid"].as<String>();
  wifiPassword =  obj["wifi_pass"].as<String>();
  telegrambotToken =  obj["telegramBot_token"].as<String>();
  telegrambotChatId =  obj["telegramBot_chatID"].as<String>();
  llmType =  obj["llm_type"].as<String>();     // 多供應商派發的關鍵欄位："gemini" / "openai" / "grok"
  llmKey =  obj["llm_key"].as<String>();
  llmModel =  obj["llm_model"].as<String>();
  scheduleTimeout = obj["schedule_timeout"].as<int>();  
  timeZone = obj["timezone"].as<String>();  

}

// ============================================================
// 【工具函式】把 IPAddress 物件格式化成點分十進位字串
// 用途：例如把內部的 IP 位址物件轉換成 "192.168.1.100" 這種常見的可讀字串格式，供除錯輸出使用。
// ============================================================
String Ip2String(IPAddress ip) {
  return String(ip[0])+String(".")+String(ip[1])+String(".")+String(ip[2])+String(".")+String(ip[3]);
}

// ============================================================
// 【核心函式】Arduino 標準進入點：初始化流程
// 用途：裝置開機後只會執行一次，負責建立整個系統運作所需的一切前置條件：
// 建立同步鎖 -> 載入所有持久化設定檔 -> 連接網路 -> 初始化相機 -> 首次校時 ->
// 啟動 HTTP/串流伺服器 -> 建立各個 FreeRTOS Task -> 最後回報啟動狀態。
// 任何一個關鍵初始化步驟失敗（互斥鎖建立失敗、人格或硬體定義缺失），都會選擇直接重啟裝置，
// 而非帶著不完整的狀態繼續運行，這是嵌入式系統常見的「失敗就重來」防禦策略。
// ============================================================
void setup() {
  Serial.begin(115200);  // 開啟序列埠除錯輸出，鮑率 115200

  // 建立所有 FreeRTOS 互斥鎖（見全域變數區的說明）
  botClientMutex = xSemaphoreCreateMutex();
  stateMutex     = xSemaphoreCreateMutex();
  sdMutex        = xSemaphoreCreateMutex();
  imageMutex     = xSemaphoreCreateMutex();
  
  if (!botClientMutex || !stateMutex || !sdMutex || !imageMutex) {
    // 互斥鎖是整個多工架構運作的基礎，若建立失敗代表系統處於不可信任的狀態，
    // 與其嘗試在沒有鎖保護的情況下繼續運行（可能導致資料損毀或難以除錯的當機），不如直接重啟
    Serial.println("[DEBUG] Failed to create mutexes. Restarting the MCU...");
    delay(2000);
    NVIC_SystemReset();
  }
  
  pinMode(ledPin, OUTPUT);  // 初始化板載 LED 腳位為輸出模式，用於後續狀態指示
  
  // 依序載入所有持久化設定檔，若檔案存在則覆寫對應的預設值
  String env = getStringFromFile(envFilename);
  Serial.println("env.json len: " + String(env.length())); 
  if (env != "")
    setEnvironmentSettings(env);  // 套用 WiFi、Telegram、LLM 供應商等所有環境設定

  initWiFi();      // 連接 WiFi（含 SoftAP 備援模式）

  String soul = getStringFromFile(soulFilename);
  Serial.println("Soul.md len: " + String(soul.length()));
  if (soul != "")
    llmRole = soul;

  String device = getStringFromFile(deviceFilename);
  Serial.println("device.md len: " + String(device.length()));
  if (device != "")
    devicesDefinition = device;
  devicesDefinitionFinal = devicesDefinition;
  devicesDefinitionFinal += "\n\nDevice Name: " + deviceName;
  devicesDefinitionFinal += "\nDevice timezone: " + timeZone;
  
  if (llmRole.length() == 0 || devicesDefinition.length() == 0) {
    // 人格設定（soul.md）或硬體定義（device.md）任一缺失，代表系統無法安全運作
    // （沒有硬體白名單，LLM 就沒有邊界可循），直接重啟並期望下次啟動時 SD 卡狀態正常
	  Serial.println("System configuration failed. Restarting the MCU...");
	  delay(5000);
	  NVIC_SystemReset();
  }

  String skill = getStringFromFile(skillFilename);
  Serial.println("skill.md len: " + String(skill.length()));
  if (skill != "")
    skillsDefinition = skill;

  String schedule = getStringFromFile(scheduleFilename);
  Serial.println("schedule.md len: " + String(schedule.length()));
  if (schedule != "")
    scheduleTasks = schedule;

  String scheduleExecutedTodayTasks = getStringFromFile(scheduleExecutedTodayTasksFilename);
  Serial.println("scheduleTodayExecuted.md len: " + String(scheduleExecutedTodayTasks.length()));
  if (scheduleExecutedTodayTasks != "")
    executedTodayTasks = scheduleExecutedTodayTasks;
  
  // 組裝三種版本的 system prompt（與 llmChatReset/systemContentReset 內的邏輯一致），供之後所有 LLM 呼叫使用
  systemContent = buildLlmMessage("user", llmRole, 0) + buildLlmMessage("model", "OK");
  systemContentTools = buildLlmMessage("user", llmRole + devicesDefinitionFinal + devicesRule + skillsDefinition + toolsDefinition, 0) + buildLlmMessage("model", "OK");
  systemContentNoTools = buildLlmMessage("user", llmRole + devicesDefinitionFinal + devicesRule, 0) + buildLlmMessage("model", "OK");

  // 載入先前保存的對話歷史，實現「重開機後仍能接續對話」的持久化記憶能力
  String memory = getStringFromFile(memoryFilename);
  Serial.println("memory.md len: " + String(memory.length()));
  if (memory != "")
    historicalMessages = memory;

  // 相機初始化
  config.setRotation(0);
  Camera.configVideoChannel(0, config);
  Camera.videoInit();
  Camera.channelBegin(0);  

  // 開機首次校時，並把校時結果主動回報給使用者
  rtcInitialTime("RTC Initial Time");
  replyUserMessage(String(taskTags[1]) + " " + getRtcTimeString(), "RTC START: " + getRtcTimeString(), telegrambotKeyboard);

  // 記錄開機當下的日期，作為「今日已執行任務」紀錄的初始基準日
  long long epoch = rtc.Read();
  time_t rawtime = (time_t)epoch;
  struct tm *now = localtime(&rawtime);
  executedTodayDate = now->tm_mday;

  server.begin();        // 啟動主要 HTTP 設定伺服器（81 port）
  serverStream.begin();  // 啟動影像串流伺服器（82 port）

  // 建立各個 FreeRTOS Task，讓多工併發運作。每個 Task 各自負責一項獨立職責：
  // 堆疊大小（第三參數）依任務複雜度分配，HTTP 相關 Task 給較大的 16384 bytes，排程 Task 給較小的 6144 bytes

  if (xTaskCreate(
        task_getRequest,
        (const char *)"task_getRequest",
        16384,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_task_getRequest failed");
  }        

  if (xTaskCreate(
        task_getRequestStream,
        (const char *)"task_getRequestStream",
        16384,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_getRequestStream failed");
  } 
  
  if (xTaskCreate(
        task_getTelegramMessage,
        (const char *)"task_getTelegramMessage",
        16384,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_getTelegramMessage failed");
  } 

  if (xTaskCreate(
        task_time_scheduling,
        (const char *)"task_time_scheduling",
        6144,
        NULL,
        tskIDLE_PRIORITY + 1,
        NULL
      )!= pdPASS) {

    Serial.println("Create task_time_scheduling failed");
  }  
  // 注意：task_theft_detection 在此並未透過 xTaskCreate 建立，
  // 代表防盜偵測背景任務目前不會隨開機自動啟動（僅函式定義存在，供未來啟用或其他觸發方式使用）

  // 開機狀態回報：先假設處於 AP（熱點）模式印出對應連線資訊
  Serial.println("AP mode"); 
  Serial.println("fuClaw Manager: http://192.168.1.1:81");
  Serial.println("Video stream: http://192.168.1.1:82"); 
  Serial.println("AP ssid : " + apSsid);
  Serial.println("AP password : " + apPassword);
  Serial.println(); 

  if (WiFi.status() == WL_CONNECTED) {
    // 若確實成功連上既有 WiFi（Station 模式），額外閃爍 LED 三次作為視覺提示，並印出實際區網 IP 位址
    for (int i=0 ; i<3 ; i++) {
      digitalWrite(ledPin, 1);
      delay(300);
      digitalWrite(ledPin, 0);
      delay(300);      
    }
	
    Serial.println("STA mode"); 
    Serial.println("fuClaw Manager: http://" + Ip2String(WiFi.localIP()) + ":81"); 
    Serial.println("Video stream: http://" + Ip2String(WiFi.localIP()) + ":82");            
    Serial.println();

    // 把裝置目前的區網 IP 位址也寫入對話歷史，讓 LLM 之後若被問到裝置網址時能夠正確回答
    historicalMessages += buildLlmMessage("user", "Current Device IP: " + Ip2String(WiFi.localIP()));
  }  
   
}

// ============================================================
// 【核心函式】Arduino 標準進入點：主迴圈
// 用途：所有實際工作都已經交給 setup() 中建立的各個 FreeRTOS Task 並行處理，
// 因此這個主迴圈本身幾乎沒有事情要做，只需要定期讓出處理器時間即可
//（FreeRTOS 架構下，loop() 本身也是被排程器視為一個低優先度的背景 Task 執行）。
// ============================================================
void loop() {
  vTaskDelay(1000 / portTICK_PERIOD_MS);
}
